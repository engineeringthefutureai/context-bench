```python
"""Parsing of weights typed in by warehouse staff.

Weights are stored and reported as whole grams (see
``shopkit.shipping.packaging`` for the formatting side).
"""

import re

# Exact gram equivalents of the units staff are allowed to type in.
_GRAMS_PER_UNIT: dict[str, float] = {
    "g": 1.0,
    "kg": 1000.0,
    "lb": 453.59237,
    "oz": 28.349523125,
}

# number (integer or "." decimal), optional whitespace, unit
# then one of the known units, case-insensitive.
_WEIGHT_RE = re.compile(
    r"\s*([0-9]+(?:\.[0-9]*)?|\.[0-9]+)\s*(g|kg|lb|oz)\s*",
    re.IGNORECASE,
)


def parse_weight(text: str) -> int:
    """Parse a weight like ``"1.5 kg"`` and return it in whole grams.

    The input is an integer or a decimal number using ``.`` as the
    separator, optional whitespace, and a unit of ``g``, ``kg``,
    ``lb`` or ``oz`` (case-insensitive). Leading and trailing
    whitespace is ignored and the result is rounded to the nearest
    gram.

    Raises ``ValueError`` for a missing or unknown unit, a negative
    value, a non-string input, and any other malformed text (for
    example ``"1,5 kg"``).
    """
    if not isinstance(text, str):
        raise ValueError(f"invalid weight: {text!r}")
    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")
    value = float(match.group(1))
    unit = match.group(2).lower()
    return round(value * _GRAMS_PER_UNIT[unit])
```
