"""Utilities for parsing and formatting time durations."""

import re

_DURATION_RE = re.compile(
    r"^\s*(?:(\d+)\s*h)?(?:(\d+)\s*m)?\s*$",
    re.IGNORECASE,
)

def parse_duration(text: str) -> int:
    """Parse a duration string into whole minutes.

    Accepts hours and/or minutes in the form ``NhMm`` or ``Nh Mm``,
    where the hours portion precedes the minutes portion and each unit
    appears at most once.  Units are case-insensitive and surrounding
    whitespace is ignored.

    Examples:
        >>> parse_duration("2h")
        120
        >>> parse_duration("45m")
        45
        >>> parse_duration("1h 30m")
        90

    Raises:
        ValueError: if the text is empty, contains unrecognized units,
            is missing a required number, places minutes before hours,
            or contains negative numbers.
    """
    if not isinstance(text, str) or not text.strip():
        raise ValueError("duration text must be a non-empty string")

    match = _DURATION_RE.match(text)
    if not match:
        raise ValueError(f"invalid duration: {text!r}")

    hours_str, minutes_str = match.groups()

    # Ensure at least one unit is present
    if hours_str is None and minutes_str is None:
        raise ValueError("duration must include at least one of hours or minutes")

    hours = int(hours_str) if hours_str is not None else 0
    minutes = int(minutes_str) if minutes_str is not None else 0

    return hours * 60 + minutes
