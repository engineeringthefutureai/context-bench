"""Refunds against captured payments. Partial refunds are allowed up to the captured amount."""
from dataclasses import dataclass, field
from datetime import date, datetime, timedelta

from shopkit.errors import Conflict
from shopkit.money import format_money
from shopkit.payments.models import Payment
from shopkit.utils.dates import utc_now


class RefundTooLarge(Conflict):
    pass


@dataclass(frozen=True)
class Refund:
    payment_id: str
    amount_cents: int
    reason: str
    at: datetime = field(default_factory=utc_now)


def refund(payment: Payment, amount_cents: int, reason: str) -> Refund:
    if amount_cents <= 0:
        raise ValueError("refund amount must be positive")
    if amount_cents > payment.refundable_cents:
        raise RefundTooLarge(
            f"refund {format_money(amount_cents, payment.currency)} exceeds refundable "
            f"{format_money(payment.refundable_cents, payment.currency)}",
            payment_id=payment.id, requested=amount_cents, refundable=payment.refundable_cents)
    payment.refunded_cents += amount_cents
    return Refund(payment.id, amount_cents, reason)


# Business days the banks need to settle a refund to the customer's card.
SETTLEMENT_DAYS = 5

# Annual days the shop is closed, as (month, day). Together with weekends these
# make up the shop's usual business-day calendar.
SHOP_CLOSED = frozenset({(1, 1), (12, 25)})


def is_business_day(day: date) -> bool:
    """Monday through Friday that is not a day the shop is closed."""
    return day.weekday() < 5 and (day.month, day.day) not in SHOP_CLOSED


def expected_arrival(refund: Refund) -> date:
    """Business day the refunded money is expected to reach the customer's card.

    Banks need five business days, counted from the day the refund was issued;
    weekends and days the shop is closed don't count.
    """
    day = refund.at.date()
    counted = 0
    while True:
        if is_business_day(day):
            counted += 1
            if counted == SETTLEMENT_DAYS:
                return day
        day += timedelta(days=1)
