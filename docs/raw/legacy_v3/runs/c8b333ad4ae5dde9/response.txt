from dataclasses import dataclass, field


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"
    discount_percent: int = 0

    @property
    def subtotal_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

    @property
    def discount_cents(self) -> int:
        # Whole-cent discount, rounded down.
        return self.subtotal_cents * self.discount_percent // 100

    @property
    def total_cents(self) -> int:
        return self.subtotal_cents - self.discount_cents

import json

from shopkit.orders.models import Order


def export_orders_json(orders: list[Order]) -> str:
    """Raw JSON export for the warehouse integration."""
    return json.dumps(
        [
            {
                "id": order.id,
                "currency": order.currency,
                "discount_percent": order.discount_percent,
                "total_cents": order.total_cents,
                "lines": [
                    {
                        "sku": line.sku,
                        "qty": line.qty,
                        "unit_cents": line.unit_cents,
                        "total_cents": line.total_cents,
                    }
                    for line in order.lines
                ],
            }
            for order in orders
        ],
        indent=2,
    )

from shopkit.money import format_money
from shopkit.orders.models import Order


def render_invoice_text(order: Order) -> str:
    """Plain-text invoice sent to customers by email."""
    lines = [f"Invoice for order {order.id}", ""]
    for line in order.lines:
        lines.append(f"{line.qty} x {line.sku}: {format_money(line.total_cents, order.currency)}")
    lines.append("")
    if order.discount_percent:
        lines.append(f"Discount: -{format_money(order.discount_cents, order.currency)}")
    lines.append(f"Total: {format_money(order.total_cents, order.currency)}")
    return "\n".join(lines)
