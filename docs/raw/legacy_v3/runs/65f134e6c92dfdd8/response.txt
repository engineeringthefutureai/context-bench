from dataclasses import dataclass, field

from shopkit.utils.text import normalize_sku


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"

    @property
    def total_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

    def add_line(self, sku: str, qty: int, unit_cents: int) -> OrderLine:
        """Add a line, merging it with an existing line for the same SKU and price."""
        target_sku = normalize_sku(sku)
        for line in self.lines:
            if normalize_sku(line.sku) == target_sku and line.unit_cents == unit_cents:
                line.qty += qty
                return line
        new_line = OrderLine(sku=sku, qty=qty, unit_cents=unit_cents)
        self.lines.append(new_line)
        return new_line
