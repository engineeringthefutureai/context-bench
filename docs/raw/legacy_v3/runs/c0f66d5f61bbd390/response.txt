import re

_DURATION = re.compile(r"\d+[hm]", re.IGNORECASE)
_UNITS = {"h": 60, "m": 1}


def parse_duration(text: str) -> int:
    """Duration in whole minutes from a string of hours and/or minutes.

    Hours come before minutes, each unit at most once, units are case-insensitive and
    surrounding whitespace is ignored, e.g. "2h", "45m", "1h30m" or "1h 30m" -> 90.
    Anything else (empty text, other units, a missing number, minutes before hours,
    negative numbers) raises ValueError.
    """
    cleaned = text.strip()
    if not cleaned:
        raise ValueError("empty duration")
    parts = _DURATION.findall(cleaned)
    if not parts or "".join(parts) != re.sub(r"\s+", "", cleaned):
        raise ValueError(f"invalid duration: {text!r}")
    minutes = 0
    seen_hours = False
    for part in parts:
        unit = part[-1].lower()
        value = int(part[:-1])
        if unit == "h":
            if seen_hours:
                raise ValueError("hours specified more than once")
            seen_hours = True
        elif unit == "m":
            if seen_hours:
                raise ValueError("minutes must come before hours")
        minutes += value * _UNITS[unit]
    return minutes
