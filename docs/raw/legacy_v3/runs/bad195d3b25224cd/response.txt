from shopkit.errors import Conflict


class Stock:
    """On-hand quantities per SKU."""

    def __init__(self):
        self._on_hand: dict[str, int] = {}

    def receive(self, sku: str, qty: int) -> None:
        self._on_hand[sku] = self._on_hand.get(sku, 0) + qty

    def available(self, sku: str) -> int:
        return self._on_hand.get(sku, 0)

    def reserve(self, sku: str, qty: int) -> None:
        avail = self._on_hand.get(sku, 0)
        if qty > avail:
            raise Conflict(f"insufficient stock for {sku}", requested=qty, available=avail)
        self._on_hand[sku] = avail - qty
