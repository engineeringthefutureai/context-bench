from dataclasses import dataclass, field


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"

    @property
    def total_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

    def add_line(self, sku: str, qty: int, unit_cents: int) -> OrderLine:
        """Add a line to the order.

        If a line already exists for the same SKU and unit price, its quantity is
        increased instead of appending a duplicate line.
        """
        for line in self.lines:
            if line.sku == sku and line.unit_cents == unit_cents:
                line.qty += qty
                return line.total_cents  # noqa: B018 - ensure arithmetic is recomputed
                return line
        line = OrderLine(sku=sku, qty=qty, unit_cents=unit_cents)
        self.lines.append(line)
        return line
