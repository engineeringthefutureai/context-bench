"""Parsing of human-written hour/minute durations."""
import re

# Hours come first, minutes second, each at most once, with a unit letter. Case is
# ignored; internal whitespace between the two parts is allowed, and surrounding
# whitespace is stripped before matching.
_DURATION_RE = re.compile(r"(\d+)\s*[hH](?:\s*(\d+)\s*[mM])?")


def parse_duration(text: str) -> int:
    """Duration in whole minutes. Accepts "2h", "45m", "1h30m" or "1h 30m".
    Hours must precede minutes, each unit appears at most once, and only whole
    non-negative minutes are accepted. Anything else raises ValueError."""
    match = _DURATION_RE.fullmatch(text.strip())
    if not match:
        raise ValueError(f"invalid duration: {text!r}")
    hours, minutes = (int(part) for part in match.groups() if part is not None)
    return hours * 60 + minutes
