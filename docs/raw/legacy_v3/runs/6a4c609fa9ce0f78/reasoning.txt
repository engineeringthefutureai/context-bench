"""Utilities for working with duration strings."""

import re

_DURATION_PATTERN = re.compile(
    r"\s*(?P<hours>[0-9]+h)?\s*(?P<minutes>[0-9]+m)?\s*",
    re.IGNORECASE,
)


def parse_duration(text: str) -> int:
    """Parse a duration expressed in hours and/or minutes into whole minutes.

    Accepts strings such as ``"45m"``, ``"2h"``, ``"1h30m"`` and ``"1h 30m"``.
    Units must appear at most once and hours must precede minutes. Surrounding
    whitespace is ignored.

    :raises ValueError: if ``text`` is empty or not a valid duration.
    """
    if not text or not text.strip():
        raise ValueError("duration must not be empty")

    match = _DURATION_PATTERN.fullmatch(text)
    if match is None:
        raise ValueError("invalid duration")

    hours_text = match.group("hours")
    minutes_text = match.group("minutes")

    if hours_text is None and minutes_text is None:
        raise ValueError("invalid duration")

    total_minutes = 0
    if hours_text is not None:
        total_minutes += int(hours_text[:-1]) * 60
    if minutes_text is not None:
        total_minutes += int(minutes_text[:-1])
    return total_minutes
