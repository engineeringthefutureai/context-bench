import re

_DURATION_RE = re.compile(r"\s*(?:(\d+)h)?\s*(?:(\d+)m)?\s*", re.IGNORECASE)

def parse_duration(text: str) -> int:
    """Parse a duration in hours and minutes and return total whole minutes."""
    if not isinstance(text, str):
        raise ValueError("duration must be a string")
    match = _DURATION_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"invalid duration: {text!r}")
    hours_text, minutes_text = match.groups()
    if hours_text is None and minutes_text is None:
        raise ValueError(f"invalid duration: {text!r}")
    hours = int(hours_text) if hours_text is not None else 0
    minutes = int(minutes_text) if minutes_text is not None else 0
    return hours * 60 + minutes
```

import re

_DURATION_RE = re.compile(
    r"\s*(?:(\d+)h)?\s*(?:(\d+)m)?\s*",
    re.IGNORECASE | re.ASCII,
)

def parse_duration(text: str) -> int:
    """Return the total whole minutes represented by `text`.

    Accepted forms include "2h", "45m", "1h30m", and "1h 30m". Units are
    case-insensitive, surrounding whitespace is ignored, and each unit may
    appear at most once.
    """
    if not isinstance(text, str):
        raise ValueError("duration must be a string")

    match = _DURATION_RE.fullmatch(text)
    if match is None:
        raise ValueError("invalid duration")

    hours_text, minutes_text = match.groups()
    if hours_text is None and minutes_text is None:
        raise ValueError("invalid duration")

    hours = int(hours_text) if hours_text is not None else 0
    minutes = int(minutes_text) if minutes_text is not None else 0
    return hours * 60 + minutes
```
