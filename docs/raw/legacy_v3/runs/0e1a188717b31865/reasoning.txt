```python
"""Parse and convert weights typed by warehouse staff."""
import re
from decimal import ROUND_HALF_UP, Decimal

# Exact conversion factors to grams (international avoirdupois pound/ounce).
_GRAMS_PER_UNIT = {
    "g": Decimal(1),
    "kg": Decimal(1000),
    "lb": Decimal("453.59237"),
    "oz": Decimal("28.349523125"),
}

# A number (digits with at most one '.'-separated decimal part), optional whitespace, a unit.
_WEIGHT_RE = re.compile(r"([0-9]+(?:\.[0-9]+)?)\s*([A-Za-z]+)")


def parse_weight(text: str) -> int:
    """Weight in whole grams, e.g. parse_weight("2.5 kg") == 2500, parse_weight("1 lb") == 454.

    Units are g, kg, lb or oz, case-insensitive, with optional whitespace before the unit.
    Raises ValueError for a missing number or unit, an unknown unit, negative values,
    a "," decimal separator and any other malformed input."""
    cleaned = text.strip()
    if cleaned.startswith("-"):
        raise ValueError(f"negative weight: {text!r}")
    match = _WEIGHT_RE.fullmatch(cleaned)
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")
    number, unit = match.groups()
    factor = _GRAMS_PER_UNIT.get(unit.lower())
    if factor is None:
        raise ValueError(f"unknown weight unit: {unit!r}")
    grams = (Decimal(number) * factor).quantize(Decimal(1), rounding=ROUND_HALF_UP)
    return int(grams)
```
