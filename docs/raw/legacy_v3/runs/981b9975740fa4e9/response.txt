"""Parsing of package weights entered by warehouse staff."""

import re

__all__ = ["parse_weight"]

# Conversion factors to grams. 1 lb = 453.59237 g, 1 oz = 28.349523125 g.
GRAMS_PER_UNIT = {
    "g": 1.0,
    "kg": 1000.0,
    "lb": 453.59237,
    "oz": 28.349523125,
}

# A non-negative integer or "."-decimal number, optional whitespace, then a unit.
_WEIGHT_RE = re.compile(r"(?P<number>[0-9]+(?:\.[0-9]+)?)\s*(?P<unit>g|kg|lb|oz)\Z", re.IGNORECASE)


def parse_weight(text: str) -> int:
    """Parse a weight entered by warehouse staff into whole grams.

    The text is a number (integer or a decimal using "." as separator),
    optional whitespace, then a unit (``g``, ``kg``, ``lb`` or ``oz``,
    case-insensitive). Leading and trailing whitespace is ignored. 1 lb =
    453.59237 g and 1 oz = 28.349523125 g. The result is rounded to the
    nearest whole gram.

    >>> parse_weight("1.5 kg")
    1500
    >>> parse_weight("2 lb")
    907
    >>> parse_weight(" 8 OZ ")
    227
    >>> parse_weight("250g")
    250

    Raises:
        ValueError: if the text is not a non-negative number followed by a
            known unit (including a missing number or unit, a negative
            value, or "," as the decimal separator).
    """
    if not isinstance(text, str):
        raise ValueError("weight must be a string")
    match = _WEIGHT_RE.fullmatch(text.strip())
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")
    number = float(match["number"])
    factor = GRAMS_PER_UNIT[match["unit"].lower()]
    grams = number * factor
    return int(grams + 0.5)
