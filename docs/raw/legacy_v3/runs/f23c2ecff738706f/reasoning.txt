import re

_DURATION_RE = re.compile(r"(?P<hours>\d+h)?\s*(?P<minutes>\d+m)?", re.IGNORECASE)


def parse_duration(text: str) -> int:
    """Parse a human duration of hours and/or minutes into whole minutes.

    Accepted forms: "45m", "2h", "1h30m", "1h 30m". Units are case-insensitive.
    Raises ValueError for empty text, unknown units, missing numbers, minutes
    before hours, repeated units, or negative numbers.
    """
    if not isinstance(text, str):
        raise ValueError("duration must be a string")
    cleaned = text.strip()
    if not cleaned:
        raise ValueError("duration must not be empty")
    match = _DURATION_RE.fullmatch(cleaned)
    if match is None:
        raise ValueError(f"invalid duration: {text!r}")
    hours_text, minutes_text = match.groups()
    if hours_text is None and minutes_text is None:
        raise ValueError(f"invalid duration: {text!r}")
    total_minutes = 0
    if hours_text is not None:
        total_minutes += int(hours_text[:-1]) * 60
    if minutes_text is not None:
        total_minutes += int(minutes_text[:-1])
    return total_minutes

_DURATION_RE = re.compile(
    r"(?P<hours>[0-9]+h)?\s*(?P<minutes>[0-9]+m)?",
    re.IGNORECASE,
)


def parse_duration(text: str) -> int:
    """Parse a duration such as "45m", "2h", "1h30m" or "1h 30m" into whole minutes."""
    if not isinstance(text, str):
        raise ValueError("duration must be a string")

    cleaned = text.strip()

    match = _DURATION_RE.fullmatch(cleaned)
    if match is None:

    hours = match.group("hours")
    minutes = match.group("minutes")
    if hours is None and minutes is None:
        raise ValueError(f"invalid duration: {cleaned!r}")

    total = 0
    if hours is not None:
        total += int(hours[:-1]) * 60
    if minutes is not None:
        total += int(minutes[:-1])
    return total
