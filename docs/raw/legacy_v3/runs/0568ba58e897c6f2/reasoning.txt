from datetime import date, timedelta

from shopkit.config import SETTINGS
from shopkit.errors import Conflict, ShopError

from shopkit.utils.ids import short_code

class InvalidReturn(ShopError):

@dataclass
class ReturnRequest:
    order_id: str
    items: dict[str, int]            # sku -> qty being returned

    ...

"""Deployment settings.

Every setting has a sane default and can be overridden with an environment
variable (SHOPKIT_ prefixed).  ``load_settings`` reads from ``os.environ`` by
default so the same code runs in every deployment.
"""
import os
from dataclasses import dataclass, field

_TRUE = {"1", "true", "yes", "on"}


def _env_int(env: dict[str, str], name: str, default: int) -> int:
    return int(env[name]) if name in env else default


def _env_bool(env: dict[str, str], name: str) -> bool:
    return env.get(name, "").strip().lower() in _TRUE


@dataclass(frozen=True)
class Settings:
    default_currency: str = "USD"
    store_country: str = "US"
    low_stock_threshold: int = 10
    invoice_prefix: str = "INV-"
    page_size: int = 20
    max_page_size: int = 100
    support_email: str = "support@shopkit.example.com"
    free_shipping_over_cents: int = 5000
    supported_currencies: tuple[str, ...] = ("USD", "EUR", "GBP")
    feature_flags: dict[str, bool] = field(default_factory=dict)
    return_window_days: int = 30

    def flag(self, name: str) -> bool:
        return self.feature_flags.get(name, False)


def load_settings(env: dict[str, str] | None = None) -> Settings:
    env = env if env is not None else os.environ
    currencies = [c.strip() for c in env.get("SHOPKIT_CURRENCIES", "USD,EUR,GBP").split(",") if c.strip()]
    flags = {
        name[len("SHOPKIT_FLAG_"):]: True
        for name, value in env.items()
        if name.startswith("SHOPKIT_FLAG_") and _env_bool(env, name)
    }
    return Settings(
        default_currency=env.get("SHOPKIT_CURRENCY", "USD"),
        store_country=env.get("SHOPKIT_COUNTRY", "US"),
        low_stock_threshold=_env_int(env, "SHOPKIT_LOW_STOCK", 10),
        invoice_prefix=env.get("SHOPKIT_INVOICE_PREFIX", "INV-"),
        page_size=_env_int(env, "SHOPKIT_PAGE_SIZE", 20),
        max_page_size=_env_int(env, "SHOPKIT_MAX_PAGE_SIZE", 100),
        support_email=env.get("SHOPKIT_SUPPORT_EMAIL", "support@shopkit.example.com"),
        free_shipping_over_cents=_env_int(env, "SHOPKIT_FREE_SHIPPING_OVER_CENTS", 5000),
        supported_currencies=tuple(currencies),
        feature_flags=flags,

    pass

    http_status = 422

class ReturnRequest:
    order_id: str
    items: dict[str, int]            # sku -> qty being returned
    reason: str
    code: str = field(default_factory=lambda: short_code(8, groups=4))
    approved: bool = False

        prices = {line.sku: line.unit_cents for line in order.lines}
        return sum(prices[sku] * qty for sku, qty in self.items.items())

                today: date) -> ReturnRequest:

        raise InvalidReturn(f"unknown reason {reason!r}", reason=reason)
    if today > delivered_on + timedelta(days=RETURN_WINDOW_DAYS):
        raise ReturnNotAllowed(f"order {order.id} is outside the return window", order_id=order.id,

    for sku, qty in items.items():
        if qty < 1 or qty > ordered.get(sku, 0):
            raise InvalidReturn(f"cannot return {qty} x {sku}", sku=sku, requested=qty,
                                ordered=ordered.get(sku, 0))
    # changed-mind returns need manual approval; faults are approved automatically
    return ReturnRequest(order.id, dict(items), reason, approved=reason != "changed_mind")
