```python
"""Weight entry helpers: parse staff-typed weights like "1.5 kg" into whole grams."""
import re
from decimal import ROUND_HALF_UP, Decimal

# Exact gram equivalents; lb/oz from the avoirdupois definitions (1 lb = 16 oz, 1 oz = 28.349523125 g).
_GRAMS_PER_UNIT = {
    "g": Decimal(1),
    "kg": Decimal(1000),
    "lb": Decimal("453.59237"),
    "oz": Decimal("28.349523125"),
}

# Number ("." decimals only, no sign, no thousands separators), optional whitespace, then a unit token.
_WEIGHT_RE = re.compile(r"(\d+(?:\.\d+)?)\s*([A-Za-z]+)")


def parse_weight(text: str) -> int:
    """Weight in whole grams, e.g. parse_weight("2.5 kg") -> 2500, parse_weight("1 lb") -> 454.
    Units g, kg, lb, oz are case-insensitive; leading/trailing whitespace is ignored.
    Raises ValueError on a missing or negative number, a missing or unknown unit, or a "," decimal separator."""
    stripped = text.strip()
    match = _WEIGHT_RE.fullmatch(stripped)
    if match is None:
        if stripped.startswith("-"):
            raise ValueError(f"weight must be >= 0: {text!r}")
        raise ValueError(f"invalid weight: {text!r}")
    number, unit = match.group(1), match.group(2).lower()
    if unit not in _GRAMS_PER_UNIT:
        raise ValueError(f"unknown unit {unit!r}: {text!r}")
    grams = Decimal(number) * _GRAMS_PER_UNIT[unit]
    return int(grams.quantize(Decimal(1), rounding=ROUND_HALF_UP))
```

```python
"""Unit parsing for warehouse data entry. Weights come back as whole grams."""
import re
from decimal import ROUND_HALF_UP, Decimal

_GRAMS_PER_UNIT = {"g": 1, "kg": 1000, "lb": 453.59237, "oz": 28.349523125}
_WEIGHT_RE = re.compile(
    r"([+-]?(?:\d+(?:\.\d*)?|\.\d+))"
    r"\s*(" + "|".join(_GRAMS_PER_UNIT) + r")"
    r"\Z", re.IGNORECASE)


def parse_weight(text: str) -> int:
    """Weight in whole grams, e.g. "1.5 kg", "16 OZ", " 2 g". Unit is g, kg, lb or oz,
    case-insensitive; result rounds half-up to the nearest gram.
    Raises ValueError on a missing number, a missing or unknown unit, negative values,
    or a "," as decimal separator."""
    if not isinstance(text, str):
        raise ValueError(f"weight must be a string, got {type(text).__name__}")
    match = _WEIGHT_RE.match(text.strip())
    if not match:
        raise ValueError(f"cannot parse weight: {text!r}")
    number = Decimal(match.group(1))
    if number < 0 or not number.is_finite():
        raise ValueError(f"weight must be a plain non-negative number: {text!r}")
    return int((number * _GRAMS_PER_UNIT[match.group(2).lower()]).quantize(
        Decimal(1), rounding=ROUND_HALF_UP))
```
