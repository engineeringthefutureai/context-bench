import re

_DURATION_RE = re.compile(r"\s*(\d+h)?\s*(\d+m)?\s*")


def parse_duration(text: str) -> int:
    """Duration in whole minutes from text like "2h", "45m", "1h30m" or "1h 30m".

    Hours come before minutes, each unit at most once, units are case-insensitive and
    surrounding whitespace is ignored. Anything else (empty text, unknown units, a missing
    number, minutes before hours, negative numbers) raises ValueError.
    """
    match = _DURATION_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"invalid duration: {text!r}")
    hours_part, minutes_part = match.groups()
    if hours_part is None and minutes_part is None:
        raise ValueError(f"invalid duration: {text!r}")
    hours = int(hours_part[:-1]) if hours_part else 0
    minutes = int(minutes_part[:-1]) if minutes_part else 0
    return hours * 60 + minutes
