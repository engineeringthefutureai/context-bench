The codebase already has `shopkit/utils/dates.py::add_business_days(start, days)` which counts business days after `start` (skipping weekends and `HOLIDAYS`), used by both shipping and invoice terms. I'll reuse it for the 5-business-day bank rule, taking the issue day as `refund.at.date()` so the function doesn't depend on the current time.

"""Refunds against captured payments. Partial refunds are allowed up to the captured amount."""
from dataclasses import dataclass, field
from datetime import date, datetime

from shopkit.errors import Conflict
from shopkit.money import format_money
from shopkit.payments.models import Payment
from shopkit.utils.dates import add_business_days, utc_now

# Business days (shop's calendar: weekends and HOLIDAYS skipped) the bank needs to
# post a card refund before it shows on the customer's statement.
BANK_POSTING_BUSINESS_DAYS = 5


class RefundTooLarge(Conflict):
    pass


@dataclass(frozen=True)
class Refund:
    payment_id: str
    amount_cents: int
    reason: str
    at: datetime = field(default_factory=utc_now)


def refund(payment: Payment, amount_cents: int, reason: str) -> Refund:
    if amount_cents <= 0:
        raise ValueError("refund amount must be positive")
    if amount_cents > payment.refundable_cents:
        raise RefundTooLarge(
            f"refund {format_money(amount_cents, payment.currency)} exceeds refundable "
            f"{format_money(payment.refundable_cents, payment.currency)}",
            payment_id=payment.id, requested=amount_cents, refundable=payment.refundable_cents)
    payment.refunded_cents += amount_cents
    return Refund(payment.id, amount_cents, reason)


def expected_arrival(refund: Refund) -> date:
    """The date banks expect the money to reach the customer's card.

    Banks take BANK_POSTING_BUSINESS_DAYS business days, counted from the day the
    refund was issued (refund.at), on the shop's business-day calendar (weekends and
    the days the shop is closed don't count).
    """
    return add_business_days(refund.at.date(), BANK_POSTING_BUSINESS_DAYS)
