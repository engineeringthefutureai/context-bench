class Box:
    name: str
    length_cm: int
    width_cm: int
    height_cm: int
    tare_g: int

    @property
    def volume_cm3(self) -> int:
        return self.length_cm * self.width_cm * self.height_cm


BOXES = [Box("S", 20, 15, 10, 120), Box("M", 35, 25, 15, 250), Box("L", 50, 40, 30, 480)]
