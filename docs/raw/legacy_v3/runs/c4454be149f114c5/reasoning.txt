class Stock:
    """On-hand quantities per SKU."""

    def __init__(self):
        self._on_hand: dict[str, int] = {}

    def receive(self, sku: str, qty: int) -> None:
        self._on_hand[sku] = self._on_hand.get(sku, 0) + qty

    def available(self, sku: str) -> int:
        return self._on_hand.get(sku, 0)

    def reserve(self, sku: str, qty: int) -> None:
        """Reserve stock, lowering available quantity by qty.

        Raises Conflict if insufficient stock; available is 0 for unknown SKUs.
        """
        available = self.available(sku)
        if qty > available:
            raise Conflict(
                f"insufficient stock for {sku}",
                sku=sku,
                requested=qty,
                available=available,
            )
        self._on_hand[sku] = available - qty
