"""Duration parsing for human-readable "1h30m"-style values, e.g. SLAs and report windows."""
import re

# Hours come before minutes, each unit at most once; the match is checked with fullmatch.
_DURATION_RE = re.compile(r"(?:(\d+)h\s*)?(?:(\d+)m\s*)?", re.IGNORECASE)


def parse_duration(text: str) -> int:
    """Total whole minutes for a duration such as "2h", "45m", "1h30m" or "1h 30m".
    Units are case-insensitive and surrounding whitespace is ignored. Anything else
    (empty text, other units, a missing number, minutes before hours, negatives)
    raises ValueError."""
    match = _DURATION_RE.fullmatch(text.strip())
    if match is None or (match.group(1) is None and match.group(2) is None):
        raise ValueError(f"invalid duration: {text!r}")
    hours = int(match.group(1) or 0)
    minutes = int(match.group(2) or 0)
    return hours * 60 + minutes
