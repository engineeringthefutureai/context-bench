```python
"""Duration parsing for display and scheduling fields."""
import re

_DURATION_RE = re.compile(r"^(\d+)[hH](?:\s*(\d+)[mM])?$")


def parse_duration(text: str) -> int:
    """Total minutes in a "hours and/or minutes" string, e.g. "2h" -> 120, "1h 30m" -> 90.

    Hours come before minutes, each unit appears at most once, units are
    case-insensitive and surrounding whitespace is ignored. Anything else
    raises ValueError."""
    match = _DURATION_RE.match(text.strip())
    if not match:
        raise ValueError(f"invalid duration: {text!r}")
    hours = int(match.group(1))
    minutes = int(match.group(2) or 0)
    return hours * 60 + minutes
```

```python
"""Duration parsing: "hours and/or minutes" strings (e.g. "1h 30m") to whole minutes."""
import re

# Hours before minutes, each unit at most once, e.g. "2h", "45m", "1h30m", "1h 30m".
# Case-insensitive units; whitespace is only allowed around the text or between the parts.
_DURATION_RE = re.compile(r"(\d+)[hH](?:\s+(\d+)[mM])?")


def parse_duration(text: str) -> int:
    """Total whole minutes in `text`, e.g. parse_duration("1h 30m") -> 90.
    Raises ValueError on empty text, other units, missing numbers, minutes
    before hours or negative numbers."""
    match = _DURATION_RE.fullmatch(text.strip())
    if not match:
        raise ValueError(f"invalid duration: {text!r}")
    hours = int(match.group(1))
    minutes = int(match.group(2) or 0)
    return hours * 60 + minutes
```

```python
"""Human-readable duration helpers, e.g. shipping SLAs and report windows."""
import re

_DURATION_RE = re.compile(r"(?:(\d+)h\s*)?(?:(\d+)m\s*)?")


def parse_duration(text: str) -> int:
    """Total minutes for an "hours and/or minutes" string, e.g. "2h", "45m", "1h30m", "1h 30m".
    Hours come before minutes, each unit at most once, units case-insensitive, surrounding
    whitespace ignored. Anything else raises ValueError."""
    stripped = text.strip()
    match = _DURATION_RE.fullmatch(stripped)
    if match is None or (match.group(1) is None and match.group(2) is None):
        raise ValueError(f"invalid duration: {text!r}")
    hours = int(match.group(1) or 0)
    minutes = int(match.group(2) or 0)
```
