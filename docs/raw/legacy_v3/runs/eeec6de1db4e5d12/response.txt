```python
import re


def parse_duration(text: str) -> int:
    """Parse a duration of hours and/or minutes into whole minutes.

    Input is a combination of hours and/or minutes (e.g. "2h", "45m", "1h30m").
    Hours must come before minutes; each unit appears at most once.
    Raises ValueError on any other input.
    """
    if text is None:
        raise ValueError("text must be a string")

    cleaned = text.strip()
    if not cleaned:
        raise ValueError("duration cannot be empty")

    pattern = r"^\s*(\d+)\s*h(?:ours?)?\s*|^\s*(\d+)\s*m(?:in(?:ute)?)?\s*$"
    full_pattern = (
        r"^(?:\s*(\d+)\s*h(?:ours?)?\s*)"
        r"(?:\s*(\d+)\s*m(?:in(?:ute)?)?\s*)?$"
    )
    match = re.fullmatch(full_pattern, cleaned, re.IGNORECASE)
    if not match:
        raise ValueError(f"invalid duration: {text!r}")

    hours_str, minutes_str = match.groups()
    hours = int(hours_str) if hours_str else 0
    minutes = int(minutes_str) if minutes_str else 0

    if hours == 0 and minutes == 0:
        # Accept "0" or "0h", "0m" as 0 minutes? 
        # Actually "0h" or "0m" should be fine. But we must ensure at least one part was present.
        # In our regex, if cleaned is "0h", hours_str="0", minutes_str=None.
        # If cleaned is "0", the fullmatch fails because it expects h or m suffix.
        # So this branch is only for "0h", "0m", or "0h0m".
        pass

    if hours == 0 and minutes == 0:
        # Check if the input was essentially zero?
        # Actually, if input is "0h", it's 0 minutes. If input is "0m", it's 0 minutes.
        # Is "0h" valid? Yes, "0" is a number.
        # Let's just return hours * 60 + minutes.
        pass

    return hours * 60 + minutes
```
