import re

# A duration is a whole number of hours, minutes, or both; hours come first,
# each unit at most once, e.g. "2h", "45m", "1h30m". Units are case-insensitive.
_DURATION_RE = re.compile(r"(\d+)h(?:\s*(\d+)m)?")


def parse_duration(text: str) -> int:
    """Duration in whole minutes: "2h", "45m", "1h30m" or "1h 30m".

    Surrounding whitespace is ignored; anything else (empty text, unknown
    units, a missing number, minutes before hours, negative or fractional
    numbers, repeated units) raises ValueError.
    """
    match = _DURATION_RE.fullmatch(text.strip(), re.IGNORECASE)
    if match is None:
        raise ValueError(f"invalid duration: {text!r}")
    hours = int(match.group(1))
    minutes = int(match.group(2) or 0)
    return hours * 60 + minutes
