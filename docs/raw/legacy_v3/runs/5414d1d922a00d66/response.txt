import re

# Exact decimal factors from the binary values used for parsing (never float arithmetic).
_GRAMS_PER = {
    "g": ("1", "1"),
    "kg": ("1000", "1"),
    "lb": ("45359237", "100000"),
    "oz": ("28349523125", "1000000000"),
}

_WEIGHT_RE = re.compile(r"(?P<num>\.?\d+(?:\.\d+)?)(?:\s+)(?P<unit>[a-zA-Z]+)\Z")


def parse_weight(text: str) -> int:
    """Weight typed by warehouse staff, as whole grams, e.g. '2.5 kg' -> 2500.

    Accepts a non-negative number (integer or '.' decimal) and one of the units g,
    kg, lb or oz (case-insensitive), with optional whitespace between number and
    unit and ignored surrounding whitespace. Rounds to the nearest gram.
    Raises ValueError for anything else.
    """
    if not isinstance(text, str):
        raise ValueError(f"weight must be a string, got {type(text).__name__}")

    match = _WEIGHT_RE.fullmatch(text.strip())
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")

    num = match["num"].lstrip(".") or "0"  # ".5" parses as "5", with a decimal point
    num = num.rstrip("0") or "0"  # drop insignificant trailing zeros: 2.500 -> 2.5

    whole, dot, frac = num.partition(".")
    num_int = int(whole + frac)
    scale = 10 ** len(frac)

    factor_num, factor_den = _GRAMS_PER[match["unit"].lower()]
    numerator = int(num_int) * int(factor_num)
    denominator = scale * int(factor_den)
    return (2 * numerator + denominator) // (2 * denominator)  # half rounds up, no float</file>
