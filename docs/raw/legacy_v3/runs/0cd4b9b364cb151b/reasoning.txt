```python
import re

_G_PER_UNIT = {"g": 1, "kg": 1000, "lb": 453.59237, "oz": 28.349523125}


def parse_weight(text: str) -> int:
    """Package weight entered by staff, e.g. "2.5 kg" -> 2500 (grams).

    Accepts an integer or decimal number (decimal point "."), an optional
    space, and a unit of g/kg/lb/oz (case-insensitive). Returns grams rounded
    to the nearest whole number; raises ValueError on any other input,
    including missing/unknown unit and negative values.
    """
    if not isinstance(text, str):
        raise ValueError(f"expected a string, got {type(text).__name__}")
    match = re.fullmatch(r"\s*(-?\d+(?:\.\d+)?)\s*([A-Za-z]+)\s*", text)
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")
    number, unit = match.group(1), match.group(2).lower()
    if unit not in _G_PER_UNIT:
        raise ValueError(f"unknown unit: {unit!r}")
    value = float(number)
    if value < 0:
        raise ValueError(f"weight must not be negative: {text!r}")
```

```python
# Grams per unit, kept as exact (numerator, denominator) fractions so the
# rounding to whole grams never depends on binary floating point.
_GRAMS_PER_UNIT = {
    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (28349523125, 1000000000),
}

_WEIGHT_RE = re.compile(r"\s*(-?\d+(?:\.\d+)?)\s*([a-zA-Z]+)\s*")


def parse_weight(text: str) -> int:
    """Package weight as typed by warehouse staff, in whole grams.

    Accepts an integer or decimal (decimal point ".") number, optional
    whitespace, and a unit of g, kg, lb or oz (case-insensitive); leading
    and trailing whitespace is ignored. The result is rounded to the
    nearest gram. Raises ValueError for anything else, including a
    missing number or unit, an unknown unit, negative values and ","
    as a decimal separator.
```

```python
import re
from decimal import Decimal, ROUND_HALF_UP

# Exact gram equivalents of the units warehouse staff can type.
_GRAMS_PER = {
    "g": Decimal(1),
    "kg": Decimal(1000),
    "lb": Decimal("453.59237"),
    "oz": Decimal("28.349523125"),
}

_WEIGHT_RE = re.compile(

    re.IGNORECASE,

def parse_weight(text: str) -> int:
    """Weight typed by warehouse staff as number + unit (g, kg, lb, oz), in whole grams.

    e.g. "1.5 kg" -> 1500, "2 lb" -> 907. Rounds to the nearest gram.
    Raises ValueError for a missing number or unit, unknown units, negative
    values and anything else (e.g. "," as decimal separator).
    """
    match = _WEIGHT_RE.match(text)
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")
    value = Decimal(match["number"])
    if value < 0:
        raise ValueError(f"weight must not be negative: {text!r}")
    grams = value * _GRAMS_PER[match["unit"].lower()]
```
