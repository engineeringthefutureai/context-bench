"""Parsing for compact hour/minute duration strings."""

import re

_DURATION_RE = re.compile(r"([0-9]+h)?\s*([0-
DURATION_RE = re.compile(r"([0-9]+h)?\s*([0-9]+m)?", re.IGNORECASE)


def parse_duration(text: str) -> int:
    """Return the duration in whole minutes.

    Accepts hours and/or minutes, with hours before minutes, e.g. ``2h``,
    ``45m``, ``1h30m``, or ``1h 30m``. Units are case-insensitive and
    surrounding whitespace is ignored. Each unit may appear at most once.

    Raises:
        ValueError: If ``text`` is not a valid hours/minutes duration.
    """
    if not isinstance(text, str):
        raise ValueError("duration must be a string")

    candidate = text.strip()
    match = _DURATION_RE.fullmatch(candidate)
    if match is None:
        raise ValueError(f"invalid duration: {text!r}")

    hours_token, minutes_token = match.groups()
    if hours_token is None and minutes_token is None:
        raise ValueError(f"invalid duration: {text!r}")

    hours = int(hours_token[:-1]) if hours_token is not None else 0
    minutes = int(minutes_token[:-1]) if minutes_token is not None else 0
    return hours * 60 + minutes
