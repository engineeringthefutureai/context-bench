"""Utilities for parsing human-readable duration strings."""

import re

__all__ = ["parse_duration"]

_DURATION_RE = re.compile(
    r"\s*(?:(?P<hours>[0-9]+)h)?\s*(?:(?P<minutes>[0-9]+)m)?\s*",
    re.IGNORECASE,
)


def parse_duration(text: str) -> int:
    """Parse a duration string such as ``2h``, ``45m`` or ``1h30m``.

    Hours must come before minutes, each unit may appear at most once, and
    units are case-insensitive. Surrounding whitespace is ignored. The
    returned value is the duration in whole minutes.

    Raises:
        ValueError: If ``text`` is not a valid duration string.
    """
    if not isinstance(text, str):
        raise ValueError("duration must be a string")

    match = _DURATION_RE.fullmatch(text)
    if match is None:
        raise ValueError("invalid duration")

    hours = match.group("hours")
    minutes = match.group("minutes")
    if hours is None and minutes is None:
        raise ValueError("invalid duration")

    return int(hours or 0) * 60 + int(minutes or 0)
