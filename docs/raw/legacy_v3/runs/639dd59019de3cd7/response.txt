"""Weight parsing for warehouse staff input.

Converts typed weight strings to whole grams."""

import re

_G_PER_LB = 453.59237
_G_PER_OZ = 28.349523125

_UNIT_FACTORS = {
    "g": 1,
    "kg": 1000,
    "lb": _G_PER_LB,
    "oz": _G_PER_OZ,
}


def parse_weight(text: str) -> int:
    """Parse a weight string into whole grams.

    Accepts a number (integer or decimal with "." separator), optional whitespace,
    and a unit (g, kg, lb or oz, case-insensitive). Rounds to the nearest gram.

    Raises ValueError for missing/invalid numbers, missing/unknown units,
    negative values, or "," used as a decimal separator.
    """
    if not text or not text.strip():
        raise ValueError("Empty weight string")

    stripped = text.strip()

    # Reject commas (common mistake for decimal separator or thousands)
    if "," in stripped:
        raise ValueError(f"Invalid decimal separator ',' in weight: {text!r}")

    # Match optional sign, number (digits with optional single '.'), whitespace, unit
    m = re.match(
        r"^([+-]?)(\d+(?:\.\d+)?|\.\d+)(\s*)([a-zA-Z]+)$",
        stripped,
    )
    if not m:
        raise ValueError(f"Invalid weight format: {text!r}")

    sign_str, num_str, _ws, unit_str = m.groups()

    unit = unit_str.lower()
    if unit not in _UNIT_FACTORS:
        raise ValueError(f"Unknown unit {unit_str!r} in weight: {text!r}")

    # Check for negative values
    if sign_str == "-":
        raise ValueError(f"Negative weight not allowed: {text!r}")

    try:
        value = float(num_str)
    except ValueError:
        raise ValueError(f"Invalid number in weight: {text!r}")

    if value < 0:
        raise ValueError(f"Negative weight not allowed: {text!r}")

    factor = _UNIT_FACTORS[unit]
    grams = value * factor

    # Round to nearest whole gram
    return int(round(grams))
