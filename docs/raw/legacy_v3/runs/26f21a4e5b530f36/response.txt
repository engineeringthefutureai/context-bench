```python
    """
    Aggregate transfer quantities by ISO week.
    
    Returns a dictionary with week labels (e.g., '2023-W04') as keys and 
    total units moved as values, ordered chronologically.
    """
    if not transfers:
        return {}
    
    # Group by week label
    weeks: dict[str, int] = {}
    
    # To ensure chronological order in the resulting dict, we should process
    # the weeks in sorted order. 
    # First, extract all weeks and their totals.
    # Since we need the output dict to be chronological, we can sort the keys at the end.
    
    for t in transfers:
        iso = t.at.isocalendar()
        label = f"{iso[0]}-W{iso[1]:02d}"
        weeks[label] = weeks.get(label, 0) + t.qty
        
    # Sort keys chronologically to maintain insertion order in the final dict
    sorted_weeks = sorted(weeks.keys())
    return {week: weeks[week] for week in sorted_weeks}
```

```python
"""Stock transfers between warehouses (rebalancing before peak season)."""
from dataclasses import dataclass, field
from datetime import datetime
from typing import Iterable

from shopkit.errors import Conflict
from shopkit.inventory.warehouses import Network
from shopkit.utils.dates import utc_now


class TransferFailed(Conflict):
    pass


@dataclass(frozen=True)
class Transfer:
    sku: str
    qty: int
    source: str
    destination: str
    at: datetime = field(default_factory=utc_now)


def transfer(network: Network, sku: str, qty: int, source: str, destination: str) -> Transfer:
    if qty < 1:
        raise ValueError("qty must be >= 1")
    src = network.get(source)
    if src.stock.available(sku) < qty:
        raise TransferFailed(f"{source} has only {src.stock.available(sku)} x {sku}", sku=sku,
                             requested=qty, available=src.stock.available(sku))
    # Stock has no public removal API; transfers adjust the ledger directly.
    src.stock._on_hand[sku] -= qty
    network.get(destination).stock.receive(sku, qty)
    return Transfer(sku, qty, source, destination)

def transfers_by_week(transfers: Iterable[Transfer]) -> dict[str, int]:
    """
    Aggregate transfer quantities by ISO week.
    
    Args:
        transfers: An iterable of Transfer objects.
        
    Returns:
        A dictionary mapping week labels (e.g., '2024-W03') to the total 
        quantity of units moved in that week, ordered chronologically.
    """
    if not transfers:
        return {}
    
    # Accumulate totals per week label
    totals: dict[str, int] = {}
    for t in transfers:
        iso_year, iso_week, _ = t.at.isocalendar()
        # Standard week label format: YYYY-Www
        label = f"{iso_year}-W{iso_week:02d}"
        totals[label] = totals.get(label, 0) + t.qty
        
    # Return a new dict with keys in chronological order
    # Since labels are YYYY-Www, string sorting matches chronological sorting
    return {week: totals[week] for week in sorted(totals)}
```
