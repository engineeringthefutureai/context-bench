import re


def parse_duration(text: str) -> int:
    """
    Parse a duration string into whole minutes.

    Args:
        text: A string like "1h", "30m", "1h 30m", "2H30M".

    Returns:
        The duration in minutes.

    Raises:
        ValueError: If the string is empty, contains invalid units,
                    has minutes before hours, is negative, or is malformed.
    """
    if not isinstance(text, str):
        raise ValueError("Duration must be a string")

    stripped = text.strip()
    if not stripped:
        raise ValueError("Duration cannot be empty")

    # Allowed pattern:
    # (hours: h/H) (minutes: m/M), each optional, at most once, hours before minutes
    # Numbers must be digits only.
    pattern = r'^(\d+h)?\s*(\d+m)?$'

    match = re.fullmatch(pattern, stripped, re.IGNORECASE)
    if not match:
        raise ValueError(f"Invalid duration format: {text!r}")

    hours_part = match.group(1)
    minutes_part = match.group(2)

    hours = 0
    minutes = 0

    if hours_part:
        hours = int(hours_part[:-1])

    if minutes_part:
        minutes = int(minutes_part[:-1])

    if hours == 0 and minutes == 0:
        raise ValueError("Duration must specify at least one unit")

    return hours * 60 + minutes
