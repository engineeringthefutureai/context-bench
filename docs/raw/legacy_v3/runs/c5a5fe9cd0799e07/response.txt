"""Duration strings for schedules and shipping windows: hours and/or minutes, e.g. "2h" or "1h 30m".
Everything downstream needs a whole number of minutes, so parsing returns int minutes."""
import re

# hours come before minutes, each at most once; unit letters are case-insensitive;
# surrounding whitespace is ignored. A unit without a number ("h", "5m") does not match.
_DURATION = re.compile(r"^\s*(?:(\d+)\s*h)?\s*(?:(\d+)\s*m)?\s*$")


def parse_duration(text: str) -> int:
    """Whole minutes in `text`, written as hours and/or minutes such as "2h", "45m", "1h30m"
    or "1h 30m". Units are case-insensitive and surrounding whitespace is ignored. Raises
    ValueError for empty text, other units, a missing number, minutes before hours or negative
    numbers."""
    if not isinstance(text, str):
        raise ValueError(f"duration must be a string, not {type(text).__name__}")
    m = _DURATION.match(text)
    if not m or (m.group(1) is None and m.group(2) is None):
        raise ValueError(f"invalid duration {text!r}")
    hours = int(m.group(1)) if m.group(1) is not None else 0
    minutes = int(m.group(2)) if m.group(2) is not None else 0
    return hours * 60 + minutes
