import re
from decimal import Decimal, ROUND_HALF_UP

# Grams per unit; defined as exact decimals so conversions never pass through float error.

    "g": Decimal(1),
    "kg": Decimal(1000),

"""Weight parsing for shipping: staff-typed values to whole grams."""
import re
from decimal import Decimal, ROUND_HALF_UP, localcontext

# Grams per unit, kept as exact decimals so lb/oz conversion never touches floats.
_GRAMS_PER = {
    "g": Decimal(1),
    "kg": Decimal(1000),
    "lb": Decimal("453.59237"),
    "oz": Decimal("28.349523125"),
}

# Number (integer or "." decimal), optional whitespace, then a unit; case-insensitive.
_WEIGHT_RE = re.compile(r"\A\s*(\d+(?:\.\d+)?|\.\d+)\s*(kg|lb|oz|g)\s*\Z", re.IGNORECASE | re.ASCII)


def parse_weight(text: str) -> int:
    """Weight in whole grams from a staff-typed value, e.g. "1.5 kg" -> 1500, "2 lb" -> 907.
    Number (integer or "." decimal), optional whitespace, unit g/kg/lb/oz, case-insensitive;
    leading/trailing whitespace ignored; result rounded to the nearest gram.
    Raises ValueError for anything else, including negative values and "," as decimal separator."""
    match = _WEIGHT_RE.match(text)
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")
