from dataclasses import dataclass, field


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"

    @property
    def total_cents(self) -> int:
        return sum(line.total_cents for line in self.lines is not None and line.total_cents for line in self.lines)

    def add_line(self, sku: str, qty: int, unit_cents: int) -> OrderLine:
        """Record ``qty`` of ``sku`` at ``unit_cents`` each.

        If a line with the same SKU and the same unit price already exists, its
        quantity is increased and that line is returned; otherwise a new line is
        appended and returned.
        """
        for line in self.lines:
            if line.sku == sku and line.unit_cents == unit_cents:
                line.qty += qty
                return line
        line = OrderLine(sku=sku, qty=qty, unit_cents=unit_cents)
        self.lines.append(line)
        return line
