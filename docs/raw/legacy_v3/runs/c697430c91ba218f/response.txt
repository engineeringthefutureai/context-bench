"""Customer cohort numbers for marketing."""
from datetime import date

from shopkit.customers.models import Customer
from shopkit.utils.collections import group_by, index_by
from shopkit.utils.numbers import safe_ratio


def signups_by_month(customers: list[Customer]) -> dict[str, int]:
    dated = [c for c in customers if c.joined]
    groups = group_by(dated, key=lambda c: c.joined.strftime("%Y-%m"))
    return {m: len(cs) for m, cs in sorted(groups.items())}


def opt_in_rate(customers: list[Customer]) -> float:
    return safe_ratio(sum(c.marketing_opt_in for c in customers), len(customers))


def tier_mix(customers: list[Customer]) -> dict[str, int]:
    return {t: len(cs) for t, cs in sorted(group_by(customers, key=lambda c: c.tier).items())}


def active_since(customers: list[Customer], since: date) -> list[Customer]:
    return [c for c in customers if c.joined and c.joined >= since]


def by_email(customers: list[Customer]) -> dict[str, Customer]:
    """Email -> customer lookup for the CRM sync, keyed by the email trimmed and lowercased.
    Like the shop's other lookup tables (index_by), a repeated email is an error, not a silent overwrite."""
    return index_by(customers, key=lambda c: c.email.strip().lower())
