"""Business-rule failures. The API layer maps each one to an HTTP status via `http_status`
and puts `details` into the error body."""


class ShopError(Exception):
    http_status = 400

    def __init__(self, message: str, **details):
        super().__init__(message)
        self.details = details


class NotFound(ShopError):
    http_status = 404


class Conflict(ShopError):
    http_status = 409


class OrderNotFound(NotFound):
    pass


class DuplicateOrder(Conflict):
    pass


class InsufficientStock(Conflict):
    pass

from shopkit.errors import InsufficientStock


class Stock:
    """On-hand quantities per SKU."""

    def __init__(self):
        self._on_hand: dict[str, int] = {}

    def receive(self, sku: str, qty: int) -> None:
        self._on_hand[sku] = self._on_hand.get(sku, 0) + qty

    def available(self, sku: str) -> int:
        return self._on_hand.get(sku, 0)

    def reserve(self, sku: str, qty: int) -> None:
        """Hold `qty` units of `sku` for an order; on failure nothing changes."""
        avail = self.available(sku)
        if avail < qty:
            raise InsufficientStock(f"cannot reserve {qty} x {sku}", sku=sku,
                                    requested=qty, available=avail)
        self._on_hand[sku] = avail - qty
