"""Weight parsing and conversion helpers."""


def parse_weight(text: str) -> int:
    """Parse a weight string into whole grams.

    Accepts an integer or decimal number (using "." as the decimal separator),
    optional whitespace, and a unit of g, kg, lb, or oz (case-insensitive).
    Leading and trailing whitespace is ignored.
    1 lb = 453.59237 g and 1 oz = 28.349523125 g.
    The result is rounded to the nearest gram.

    Raises ValueError for malformed input, missing number, missing or unknown
    unit, negative values, or "," used as a decimal separator.
    """
    if not isinstance(text, str):
        raise ValueError(f"expected str, got {type(text).__name__}")

    stripped = text.strip()
    if not stripped:
        raise ValueError("empty string")

    # Reject comma as decimal separator
    if "," in stripped:
        raise ValueError("comma is not a valid decimal separator")

    # Separate number and unit: find where digits/decimal end and unit begins
    i = 0
    n = len(stripped)
    in_number = False

    while i < n:
        c = stripped[i]
        if c.isdigit() or c == "." or (c == "-" and not in_number):
            in_number = True
            i += 1
        elif c in " \t":
            i += 1
        else:
            break

    num_part = stripped[:i]
    unit_part = stripped[i:].strip()

    if not num_part:
        raise ValueError("missing number")

    # Validate the number part: at most one ".", at least one digit
    dot_count = num_part.count(".")
    if dot_count > 1:
        raise ValueError(f"malformed number: {num_part!r}")
    if not any(c.isdigit() for c in num_part):
        raise ValueError(f"missing number in {text!r}")

    # Check for negative value
    try:
        value = float(num_part)
    except ValueError:
        raise ValueError(f"invalid number: {num_part!r}")

    if value < 0:
        raise ValueError(f"negative value not allowed: {num_part!r}")

    if not unit_part:
        raise ValueError(f"missing unit in {text!r}")

    # Normalize unit
    unit = unit_part.lower()
    if len(unit) > 1 and unit[-1] == "s":
        unit = unit[:-1]

    unit_to_grams = {
        "g": 1.0,
        "kg": 1000.0,
        "lb": 453.59237,
        "oz": 28.349523125,
    }

    if unit not in unit_to_grams:
        raise ValueError(f"unknown unit: {unit_part!r}")

    grams = value * unit_to_grams[unit]
    # Round to nearest whole gram (half rounds up)
    return int(grams + 0.5)
