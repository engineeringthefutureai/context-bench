import re


def parse_duration(text: str) -> int:
    """Parse a duration string like "2h", "45m", "1h30m" or "1h 30m" into whole minutes."""
    if not isinstance(text, str):
        raise ValueError("text must be a string")
    cleaned = text.strip()
    if not cleaned:
        raise ValueError("duration string must not be empty")
    
    # Remove all whitespace for consistent matching
    normalized = re.sub(r'\s+', '', cleaned)
    
    # Define a pattern for a valid duration
    # Must have either hours or minutes, in that order (hours before minutes)
    # Numbers must be non-negative integers
    pattern = r'^(?:(\d+)h)?(?:(\d+)m)?$'
    match = re.match(pattern, normalized, re.IGNORECASE)
    
    if not match:
        raise ValueError(f"Invalid duration format: '{text}'")
    
    # Ensure at least one unit is present
    hours_str, minutes_str = match.group(1), match.group(2)
    if hours_str is None and minutes_str is None:
        raise ValueError("Duration must include hours and/or minutes")
    
    hours = int(hours_str) if hours_str is not None else 0
    minutes = int(minutes_str) if minutes_str is not None else 0
    
    return hours * 60 + minutes

    """Parse a duration string like "2h", "45m", "1h30m" or "1h 30m" into whole minutes.

    Args:
        text: A string containing hours and/or minutes, e.g. "2h", "45m", "1h30m".

    Returns:
        The duration in whole minutes.

    Raises:
        ValueError: If the input is empty, has invalid units, missing numbers,
                    wrong order (minutes before hours), or negative numbers.
    """
    if not isinstance(text, str):
        raise ValueError("text must be a string")
    
    cleaned = text.strip()
    if not cleaned:
        raise ValueError("duration string must not be empty")
    
    # Remove all internal whitespace for consistent matching
    normalized = re.sub(r'\s+', '', cleaned)
    
    # Pattern: optional hours followed by optional minutes, each with a number and unit
    # Numbers are non-negative integers only
    pattern = r'^(?:(\d+)h)?(?:(\d+)m)?$'
    match = re.match(pattern, normalized, re.IGNORECASE)
    
    if not match:
        raise ValueError(f"Invalid duration format: '{text}'")
    
    # Check that at least one unit is present
    hours_str, minutes_str = match.group(1), match.group(2)
    if hours_str is None and minutes_str is None:
        raise ValueError("Duration must include hours and/or minutes")
    
    hours = int(hours_str) if hours_str is not None else 0
    minutes = int(minutes_str) if minutes_str is not None else 0
    
    return hours * 60 + minutes
