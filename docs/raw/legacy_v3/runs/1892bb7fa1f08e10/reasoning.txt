```python
_DURATION_RE = re.compile(
    r"^\s*(?:(\d+)[hH]\s*(\d+)[mM]|(\d+)[mM]|(\d+)[hH])\s*$"

def parse_duration(text: str) -> int:
    """Parse hours and/or minutes into total whole minutes.

    Accepts forms like "2h", "45m", "1h30m" and "1h 30m". Units are
    case-insensitive and surrounding whitespace is ignored. Anything else
    raises ValueError.
    """

        hours = 0
        minutes = int(match.group(3))

    return hours * 60 + minutes
```

```
import re

_DURATION_RE = re.compile(
    r"\s*"
    r"(?:"
    r"(?P<hours>[0-9]+)[hH]\s*(?P<minutes>[0-9]+)[mM]"
    r"|(?P<minutes_only>[0-9]+)[mM]"
    r"|(?P<hours_only>[0-9]+)[hH]"
    r")"
    r"\s*"
)


def parse_duration(text: str) -> int:
    """Parse a simple hours/minutes duration string into whole minutes.

    Accepts "2h", "45m", "1h30m" and "1h 30m" (units case-insensitive,
    surrounding whitespace ignored). Anything else raises ValueError.
    """
    match = _DURATION_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"invalid duration: {text!r}")

    if match.group("hours") is not None:
        hours = int(match.group("hours"))
        minutes = int(match.group("minutes") or 0)
    elif match.group("minutes_only") is not None:
        hours = 0
        minutes = int(match.group("minutes_only"))
    else:
        hours = int(match.group("hours_only") or 0)
        minutes = 0

    return hours * 60 + minutes
```
