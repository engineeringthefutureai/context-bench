import re

_DURATION_RE = re.compile(r"\A(\d+h)?(\d+m)?\Z", re.IGNORECASE)


def parse_duration(text: str) -> int:
    """Duration in whole minutes from hours and/or minutes.

    Accepts inputs like "2h", "45m", "1h30m" or "1h 30m": hours before minutes,
    each unit at most once, units case-insensitive, surrounding whitespace ignored.
    Raises ValueError for anything else.
    """
    if not isinstance(text, str):
        raise ValueError(f"duration must be a string, got {type(text).__name__}")
    match = _DURATION_RE.match(text.strip())
    if not match or (not match.group(1) and not match.group(2)):
        raise ValueError(f"invalid duration: {text!r}")
    total = 0
    for part in match.groups():
        if part is not None:
            number = int(part[:-1])
            total += number * (60 if part.endswith("h") else 1)
    return total
