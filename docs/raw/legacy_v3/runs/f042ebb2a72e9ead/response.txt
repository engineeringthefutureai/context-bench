from __future__ import annotations

import re

UNIT_FACTORS: dict[str, float] = {
    "g": 1.0,
    "kg": 1000.0,
    "lb": 453.59237,
    "oz": 28.349523125,
}

_WEIGHT_RE = re.compile(
    r"\s*([+-]?\d+(?:\.\d+)?)\s*(g|kg|lb|oz)\s*",
    re.IGNORECASE,
)


def parse_weight(text: str) -> int:
    """Parse a package weight string and return the weight in whole grams.

    The input is a number (integer or decimal using "."), optional whitespace,
    and a unit (g, kg, lb, oz) case-insensitively. Leading/trailing whitespace
    is ignored. Raises ValueError for invalid input including negatives,
    missing/unknown units, or "," as decimal separator.

    Examples:
        >>> parse_weight("100g")
        100
        >>> parse_weight(" 1.5 kg ")
        1500
        >>> parse_weight("1 lb")
        454
    """
    if not isinstance(text, str):
        raise ValueError(f"Expected a string, got {type(text).__name__}")

    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        # More specific error messages for common mistakes
        stripped = text.strip()
        if "," in stripped:
            raise ValueError(
                f"Invalid weight {text!r}: ',' is not a valid decimal separator"
            )
        if stripped.lower().startswith("-"):
            raise ValueError(
                f"Invalid weight {text!r}: negative values are not allowed"
            )
        # Check if number is present but unit is missing/unknown
        # or unit is present but number is missing
        if not stripped:
            raise ValueError(f"Invalid weight {text!r}: missing number and unit")

        # Try to extract a number part
        parts = stripped.split()
        has_number = False
        has_unit = False
        for part in parts:
            try:
                if part[-1] in ("g", "G") and len(part) > 1:
                    float(part[:-1])
                    has_number = True
                    has_unit = True
                    break
                if part.lower() in ("kg", "lb", "oz"):
                    has_unit = True
                    if len(parts) > 1:
                        float(parts[0])
                        has_number = True
                    break
                float(part)
                has_number = True
            except (ValueError, IndexError):
                pass

        if has_number and not has_unit:
            raise ValueError(f"Invalid weight {text!r}: missing unit")
        if has_unit and not has_number:
            raise ValueError(f"Invalid weight {text!r}: missing number")
        raise ValueError(f"Invalid weight {text!r}")

    number_str = match.group(1)
    unit = match.group(2).lower()

    # Reject negative values (the regex allows a leading +, we keep that)
    if number_str.startswith("-"):
        raise ValueError(f"Invalid weight {text!r}: negative values are not allowed")

    value = float(number_str)
    factor = UNIT_FACTORS[unit]
    grams = value * factor
    return int(round(grams))
