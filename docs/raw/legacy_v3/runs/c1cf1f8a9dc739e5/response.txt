import re


def parse_duration(text: str) -> int:
    """Parse a duration string like ``"2h"``, ``"45m"`` or ``"1h30m"`` into whole minutes.

    Hours must precede minutes, each unit may appear at most once, units are
    case-insensitive and surrounding whitespace is ignored.

    Raises:
        ValueError: if the string does not represent a valid duration.
    """
    if not isinstance(text, str):
        raise ValueError("duration must be a string")

    stripped = text.strip()
    if not stripped:
        raise ValueError("empty duration string")

    match = re.fullmatch(
        r"(?P<hours>\d+)[HhMm]?\s*(?:(?P<minutes>\d+)[Mm])?\s*",
        stripped,
    )
    if not match:
        raise ValueError(f"invalid duration: {text!r}")

    hours = int(match.group("hours")) if match.group("hours") is not None else 0
    minutes = int(match.group("minutes")) if match.group("minutes") is not None else 0

    return hours * 60 + minutes
