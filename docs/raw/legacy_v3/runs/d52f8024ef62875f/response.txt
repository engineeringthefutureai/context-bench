"""IDs and customer-facing codes.

IDs are internal, lower-case, random and prefixed (`ord_…`, `pay_…`).  Customer-facing
codes are distinct: unambiguous base-32 (no 0/O/1/I/l), upper-case, and grouped
with dashes where they are long enough to warrant it.
"""
import base64
import secrets

_UNAMBIGUOUS_ALPHABET = base64.b32encode(b"0").decode().replace("=", "")
_ALPHABET_REPLACEMENTS = {"A": "Q", "B": "8", "E": "5", "I": "7", "L": "3", "O": "4"}

# Group sizes for each customer-facing code length.
_GROUP_SIZES = {
    8: (4, 8, 4),
    16: (4, 4, 4, 4),
}


def new_id(prefix: str) -> str:
    """A unique internal ID: lower-case, prefixed, random.

    Example: ``ord_9f2a1c3b4d``.
    """
    return f"{prefix}_{secrets.token_hex(5)}"


def _generate_code(length: int) -> str:
    raw = secrets.token_bytes(length)
    text = base64.b32encode(raw).decode().replace("=", "")
    text = text[:length]
    return "".join(_ALPHABET_REPLACEMENTS.get(c, c) for c in text)


def short_code(length: int = 8) -> str:
    """A customer-facing code, unambiguous, upper-case base-32.

    Grouped with dashes for the supported lengths: 8 becomes ``XXXX-XXXX`` and
    16 becomes ``XXXX-XXXX-XXXX-XXXX``.  Any other length is returned ungrouped.
    """
    code = _generate_code(length)
    sizes = _GROUP_SIZES.get(length)
    if sizes is None:
        return code
    parts = []
    start = 0
    for size in sizes:
        parts.append(code[start:start + size])
        start += size
    return "-".join(parts)


def transfer_reference() -> str:
    """A pay-by-bank-transfer reference: 12 characters in three dash-separated
    groups of four, e.g. ``QW35-FG62-MN48``.

    The same style as the shop's other customer-facing codes (see
    :func:`short_code`), sized to fit in a bank transfer's reference field.
    """
    return short_code(12)

"""Card gateway adapter. The real provider is behind `charge`; tests use FakeGateway."""
from shopkit.errors import ShopError
from shopkit.payments.models import AUTHORIZED, CAPTURED, FAILED, Payment
from shopkit.payments.reconciliation import redact_pan
from shopkit.utils.ids import new_id, short_code


class PaymentDeclined(ShopError):
    http_status = 402


def transfer_reference() -> str:
    """Generate a pay-by-bank-transfer reference.

    A customer-facing code: 12 characters in three dash-separated groups of four,
    e.g. ``QW35-FG62-MN48``.  The same unambiguous base-32 style as the shop's
    other customer-facing codes.
    """
    return short_code(12)


class FakeGateway:
    """Declines card numbers ending in 0000; everything else is authorized."""

    def __init__(self):
        self.payments: dict[str, Payment] = {}

    def authorize(self, order_id: str, amount_cents: int, currency: str, card_number: str) -> Payment:
        digits = "".join(c for c in card_number if c.isdigit())
        payment = Payment(new_id("pay"), order_id, amount_cents, currency, "card",
                          card_number=redact_pan(digits))
        if digits.endswith("0000"):
            payment.status = FAILED
            self.payments[payment.id] = payment
            raise PaymentDeclined("card declined", order_id=order_id, payment_id=payment.id)
        self.payments[payment.id] = payment
        return payment

    def capture(self, payment_id: str) -> Payment:
        payment = self.payments[payment_id]
        if payment.status != AUTHORIZED:
            raise ShopError(f"payment {payment_id} is {payment.status}", payment_id=payment_id)
        payment.status = CAPTURED
        return payment
