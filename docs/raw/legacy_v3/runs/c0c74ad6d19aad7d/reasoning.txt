```
import re

_DURATION_RE = re.compile(
    r"\s*(?:(\d+)h)?\s*(?:(\d+)m)?\s*",
    re.IGNORECASE,
)


def parse_duration(text: str) -> int:
    """Parse a duration like "2h", "45m", "1h30m" or "1h 30m" into whole minutes.

    Hours must come before minutes, each unit may appear at most once, and
    surrounding whitespace is ignored. Raises ValueError for any other input.
    """
    if not isinstance(text, str):
        raise ValueError(f"invalid duration: {text!r}")

    match = _DURATION_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"invalid duration: {text!r}")

    hours = match.group(1)
    minutes = match.group(2)
    if hours is None and minutes is None:
        raise ValueError(f"invalid duration: {text!r}")

    total_minutes = 0
    if hours is not None:
        total_minutes += int(hours) * 60
    if minutes is not None:
        total_minutes += int(minutes)
    return total_minutes
```
