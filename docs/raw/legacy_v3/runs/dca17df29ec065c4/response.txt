

I've confirmed the two conventions: `normalize_sku` (shopkit/utils/text.py) is the product-code cleanup used in the catalog and on invoices, and `format_money(cents, currency)` (shopkit/money.py) is exactly how customer invoices render line totals (see `render_invoice_text`). import csv
import io
import json

from shopkit.money import format_money
from shopkit.orders.models import Order
from shopkit.utils.text import normalize_sku


def export_orders_json(orders: list[Order]) -> str:
    """Raw JSON export for the warehouse integration."""
    payload = [
        {
            "id": o.id,
            "currency": o.currency,
            "lines": [{"sku": l.sku, "qty": l.qty, "unit_cents": l.unit_cents} for l in o.lines],
        }
        for o in orders
    ]
    return json.dumps(payload, indent=2)


def export_orders_csv(orders: list[Order]) -> str:
    """CSV export for finance: one row per order line.

    Product codes are normalized and line totals are formatted exactly as on
    customer invoices.
    """
    buf = io.StringIO()
    writer = csv.writer(buf)
    writer.writerow(["order_id", "product_code", "quantity", "line_total"])
    for o in orders:
        for l in o.lines:
            writer.writerow(
                [o.id, normalize_sku(l.sku), l.qty, format_money(l.total_cents, o.currency)]
            )
    return buf.getvalue()
