import csv
import io
import json

from shopkit.orders.models import Order


def export_orders_json(orders: list[Order]) -> str:
    """Raw JSON export for the warehouse integration."""
    payload = [
        {
            "id": o.id,
            "currency": o.currency,
            "lines": [{"sku": l.sku, "qty": l.qty, "unit_cents": l.unit_cents} for l in o.lines],
        }
        for o in orders
    ]
    return json.dumps(payload, indent=2)


def _line_total_cents(line) -> int:
    if hasattr(line, "line_total"):
        return line.line_total
    if hasattr(line, "price_cents"):
        return line.price_cents * line.qty
    return line.unit_cents * line.qty


def _clean_product_code(sku: str) -> str:
    if hasattr(sku, "clean"):
        return sku.clean()
    if hasattr(sku, "upper"):
        return sku.upper()
    if hasattr(sku, "strip"):
        return sku.strip()
    return str(sku).strip().upper()


def export_orders_csv(orders: list[Order]) -> str:
    """Spreadsheet export for finance: one row per order line."""
    buffer = io.StringIO()
    writer = csv.writer(buffer, lineterminator="\n")
    writer.writerow(["order_id", "product_code", "quantity", "line_total"])
    for order in orders:
        for line in order.lines:
            line_total_cents = _line_total_cents(line)
            writer.writerow(
                [
                    order.id,
                    _clean_product_code(line.sku),
                    line.qty,
                    f"{line_total_cents // 100}.{abs(line_total_cents) % 100:02d}",
                ]
            )
    return buffer.getvalue()
