from dataclasses import dataclass


@dataclass
class Product:
    sku: str
    title: str
    price_cents: int
    active: bool = True
    weight_g: int = 0  # grams per unit

    def is_purchasable(self) -> bool:
        return self.active and self.price_cents > 0

from dataclasses import dataclass, field

from shopkit.catalog.models import Product
from shopkit.orders.models import Order
from shopkit.shipping.packaging import format_weight
from shopkit.utils.collections import chunked, group_by


@dataclass
class Parcel:
    order_id: str
    number: int
    items: list[tuple[str, int]] = field(default_factory=list)

    @property
    def item_count(self) -> int:
        return sum(q for _, q in self.items)


def packing_slip(parcel: Parcel, weight_g: int) -> str:
    """Slip packed inside each parcel: order, parcel number, contents and weight."""
    lines = [f"Order {parcel.order_id} - parcel {parcel.number}", f"Weight: {format_weight(weight_g)}"]
    lines += [f"  {qty} x {sku}" for sku, qty in parcel.items]
    return "\n".join(lines)


def parcel_weight_g(parcel: Parcel, products: dict[str, Product]) -> int:
    """Total weight in grams of the products in a parcel."""
    return sum(qty * products[sku].weight_g for sku, qty in parcel.items)
