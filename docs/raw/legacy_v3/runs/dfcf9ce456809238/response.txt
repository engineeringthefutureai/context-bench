"""Parse duration strings like "2h" or "1h 30m" into whole minutes."""
import re


def parse_duration(text: str) -> int:
    """
    Parse a duration expressed in hours and/or minutes into a total number of
    minutes.

    Accepts strings such as "2h", "45m", "1h30m", or "1h 30m". Whitespace
    around the string is ignored. At least one of hours or minutes must be
    present, and each unit (h, m) may appear at most once.
    """
    if not isinstance(text, str):
        raise ValueError("Input must be a string")

    stripped = text.strip()
    if not stripped:
        raise ValueError("Empty string is not a valid duration")

    normalized = stripped.replace(" ", "")

    # Pattern: optional hours followed by optional minutes
    # Each unit must be preceded by a positive integer.
    pattern = re.compile(
        r'^(?:(\d+)h)?(?:(\d+)m)?$'
    )
    
    match = pattern.match(normalized)
    if not match:
        raise ValueError(f"Invalid duration format: {text}")
    
    hours_str, minutes_str = match.groups()
    
    # Ensure at least one unit is present
    if hours_str is None and minutes_str is None:
        raise ValueError("Duration must include at least one of hours or minutes")
    
    hours = int(hours_str) if hours_str else 0
    minutes = int(minutes_str) if minutes_str else 0
    
    return hours * 60 + minutes
