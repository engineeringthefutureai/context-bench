"""Weight parsing for package dimensions and shipping chargeable weight."""
import re
from decimal import ROUND_HALF_UP, Decimal

# Exact conversion factors to grams.
_GRAMS_PER = {
    "g": Decimal(1),
    "kg": Decimal(1000),
    "lb": Decimal("453.59237"),
    "oz": Decimal("28.349523125"),
}

_WEIGHT_RE = re.compile(r"^(\d+(?:\.\d+)?)([a-zA-Z]+)$]+)$")


def parse_weight(text: str) -> int:
    """Parse a staff-entered package weight into whole grams.

    Accepts an optional sign is *not* allowed; a non-negative number (integer or
    decimal using "."), optional whitespace, then a unit: g, kg, lb or oz
    (case-insensitive). Leading and trailing whitespace is ignored. The result
    is rounded to the nearest whole gram.

    parse_weight("1.5 kg") -> 1500; parse_weight("2 lb") -> 907.
    Raises ValueError for missing/unknown units, a missing number, negative
    values, or "," used as the decimal separator.
    """
    if not isinstance(text, str):
        raise ValueError(f"weight must be a string, got {type(text).__name__}")

    stripped = text.strip()
    match = _WEIGHT_RE.match(stripped)
    if not match:
        # Covers missing number, missing/unknown unit, negative values, and
        # "," (or any other character) used as the decimal separator.
        raise ValueError(f"invalid weight: {text!r}")

    unit = match.group(2).lower()
    if unit not in _GRAMS_PER:
        raise ValueError(f"unknown unit in weight: {text!r}")

    grams = Decimal(match.group(1)) * _GRAMS_PER[unit]
    return int(grams.quantize(Decimal(1), rounding=ROUND_HALF_UP))
