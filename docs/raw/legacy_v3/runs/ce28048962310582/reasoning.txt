import re

_DURATION_RE = re.compile(
    r'\s*(?:(\d+)h)?\s*(?:(\d+)m)?\s*',
    re.IGNORECASE,
)

def parse_duration(text: str) -> int:
    """Parse hours/minutes string into whole minutes.

    Accepts e.g. '2h', '45m', '1h30m', '1h 30m'. Raises ValueError for
    empty text, other units, missing numbers, minutes before hours, or
    negative numbers.
    """
    if not isinstance(text, str):
        raise ValueError("duration must be a string")
    match = _DURATION_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"Invalid duration: {text!r}")
    hours, minutes = match.groups()
    if hours is None and minutes is None:
        raise ValueError(f"Invalid duration: {text!r}")
    return int(hours or 0) * 60 + int(minutes or 0)

import re

_DURATION_RE = re.compile(
    r"\s*(?:([0-9]+)h)?\s*(?:([0-9]+)m)?\s*",
    re.IGNORECASE,
)


def parse_duration(text: str) -> int:
    """Parse a duration string into whole minutes.

    Accepts hours and/or minutes, with hours before minutes, e.g. "2h",
    "45m", "1h30m", or "1h 30m". Units are case-insensitive and
    surrounding whitespace is ignored.

    Raises:
        ValueError: If the text is not a valid duration.
    """
    if not isinstance(text, str):
        raise ValueError("duration must be a string")

    match = _DURATION_RE.fullmatch(text)
    if match is None:
        raise ValueError("invalid duration")

    hours, minutes = match.groups()
    if hours is None and minutes is None:
        raise ValueError("invalid duration")

    return int(hours or 0) * 60 + int(minutes or 0)
