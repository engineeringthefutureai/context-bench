```python
import re


def parse_duration(text: str) -> int:
    """Parse a duration string like '2h', '45m', '1h30m', or '1h 30m' into whole minutes."""
    if not isinstance(text, str):
        raise ValueError("Duration must be a string")
    
    stripped = text.strip()
    if not stripped:
        raise ValueError("Duration string cannot be empty")
    
    # Remove all whitespace to normalize the format
    normalized = stripped.replace(" ", "")
    
    # Match one or more duration units in order: h before m, each at most once
    pattern = r'^(?:(\d+)h)?(?:(\d+)m)?$'
    match = re.match(pattern, normalized)
    
    if not match:
        raise ValueError(f"Invalid duration format: {text!r}")
    
    # At least one unit must be present
    hours_part, minutes_part = match.groups()
    
    if hours_part is None and minutes_part is None:
        raise ValueError(f"Invalid duration format: {text!r}")
    
    hours = int(hours_part) if hours_part is not None else 0
    minutes = int(minutes_part) if minutes_part is not None else 0
    
    return hours * 60 + minutes
```
