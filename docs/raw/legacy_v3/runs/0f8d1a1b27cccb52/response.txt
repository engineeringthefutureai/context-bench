The request is clear: orders gain an optional percentage discount that reduces the total (rounded down to whole cents), appears on customer invoices for discounted orders only, and is exported in the warehouse JSON as `discount_percent`.

I traced every consumer of `Order.total_cents` (checkout, tax, invoices, reports, fulfillment) — they all operate on the post-discount total, which is the correct reading. The `Order` dataclass gains the field; the renderer adds a conditional discount line; the export includes the field.

from dataclasses import dataclass, field


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"
    discount_percent: int = 0

    @property
    def gross_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

    @property
    def discount_cents(self) -> int:
        """Discount off the goods, rounded down to whole cents (0 with no discount)."""
        return self.gross_cents * self.discount_percent // 100

    @property
    def total_cents(self) -> int:
        return self.gross_cents - self.discount_cents

from shopkit.money import format_money
from shopkit.orders.models import Order
from shopkit.utils.text import normalize_sku


def render_invoice_text(order: Order) -> str:
    """Plain-text invoice sent to customers by email."""
    rows = [f"Invoice {order.id}"]
    for line in order.lines:
        rows.append(f"{normalize_sku(line.sku):<16}{line.qty:>4} x {format_money(line.unit_cents, order.currency):>12}"
                    f"  = {format_money(line.total_cents, order.currency):>12}")
    if order.discount_percent:
        rows.append(f"{'Discount':<36}{format_money(-order.discount_cents, order.currency):>12}")
    rows.append(f"{'TOTAL':<36}{format_money(order.total_cents, order.currency):>12}")
    return "\n".join(rows)

import json

from shopkit.orders.models import Order


def export_orders_json(orders: list[Order]) -> str:
    """Raw JSON export for the warehouse integration."""
    payload = [
        {
            "id": o.id,
            "currency": o.currency,
            "discount_percent": o.discount_percent,
            "lines": [{"sku": l.sku, "qty": l.qty, "unit_cents": l.unit_cents} for l in o.lines],
        }
        for o in orders
    ]
    return json.dumps(payload, indent=2)
