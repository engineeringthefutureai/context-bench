import re

_DURATION_RE = re.compile(
    r"\s*(?:(?P<hours>[0-9]+)h\s*)?(?:(?P<minutes>[0-9]+)m)?\s*",
    re.IGNORECASE,
)


def parse_duration(text: str) -> int:
    """Parse an hours/minutes duration into whole minutes.

    Accepted forms include "2h", "45m", "1h30m" and "1h 30m". Units are
    case-insensitive and surrounding whitespace is ignored. Anything else
    raises ValueError.
    """
    match = _DURATION_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"invalid duration: {text!r}")

    hours = match.group("hours")
    minutes = match.group("minutes")
    if hours is None and minutes is None:
        raise ValueError(f"invalid duration: {text!r}")

    total_minutes = int(hours) * 60 if hours is not None else 0
    if minutes is not None:
        total_minutes += int(minutes)
    return total_minutes
