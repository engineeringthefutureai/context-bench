"""Unit parsing utilities for the shopkit package."""


def parse_weight(text: str) -> int:
    """Parse a weight string into whole grams.

    Accepted format (case-insensitive):
        "<number>[<whitespace>]<unit>"

    where <number> is an integer or a decimal number using "." as the
    decimal separator, and <unit> is one of: g, kg, lb, oz.

    Leading and trailing whitespace around the whole input is ignored.
    The weight is converted to grams and rounded to the nearest whole
    gram.

    Raises:
        ValueError: If the input does not match the expected format,
            the value is negative, or an unknown unit is supplied.
    """
    if text is None:
        raise ValueError("weight must be a string, got None")

    stripped = text.strip()
    if not stripped:
        raise ValueError("empty weight string")

    # Find where the numeric part ends and the unit begins.
    # We scan from the left to find the first character that is not
    # part of a valid number (digit, '.', or optional leading '-').
    # But negative values are not allowed, so we reject '-' explicitly.
    if stripped.startswith('-'):
        raise ValueError(f"negative weight is not allowed: {text!r}")

    # Separate the numeric part from the unit part.
    # The numeric part consists of digits and at most one '.',
    # possibly with whitespace between number and unit.
    i = 0
    n = len(stripped)
    has_dot = False
    while i < n:
        ch = stripped[i]
        if ch.isdigit():
            i += 1
        elif ch == '.' and not has_dot:
            has_dot = True
            i += 1
        elif ch.isspace():
            # Whitespace may appear between the number and the unit.
            break
        else:
            # Any other character means the numeric part ended
            # before we expected, or the input is malformed.
            break

    if i == 0:
        raise ValueError(f"missing number in weight string: {text!r}")

    num_part = stripped[:i]

    # Reject "," as decimal separator.
    if ',' in num_part:
        raise ValueError(f"invalid decimal separator in weight string: {text!r}")

    # Reject malformed number formats like "..", ".5" is okay? Let's think:
    # "a number (an integer or a decimal using '.' as separator)"
    # Typically ".5" should be valid (e.g. ".5 kg" = 500 g).
    # But "5." might be ambiguous; let's check what makes sense.
    # The spec says "an integer or a decimal using '.' as separator".
    # I'll accept ".5" and "5." as valid decimal forms since they parse to floats.
    # But let's be careful: empty num_part is already rejected (i == 0).
    # num_part like "." -> i would be 1, num_part = "." -> float(".") raises.
    # num_part like "1.2.3" -> we stop at the second '.' because has_dot is True,
    # so i stops at index of second '.', num_part = "1." which is fine,
    # but then unit part starts with ".3" which is not a valid unit -> ValueError.
    # Actually wait: if input is "1.2.3 g", i stops at index 2 (the second '.'),
    # num_part = "1.", unit_part = ".3 g".strip() = ".3 g" -> not a valid unit -> error. Good.

    # Reject num_part that is just "." or empty string (empty already handled).
    if num_part == '.':
        raise ValueError(f"invalid number in weight string: {text!r}")

    try:
        value = float(num_part)
    except ValueError:
        raise ValueError(f"invalid number in weight string: {text!r}") from None

    # The rest is the unit part (may have leading whitespace).
    unit_part = stripped[i:].strip().lower()

    if not unit_part:
        raise ValueError(f"missing unit in weight string: {text!r}")

    # Map unit to grams conversion factor.
    unit_to_grams = {
        'g': 1.0,
        'kg': 1000.0,
        'lb': 453.59237,
        'oz': 28.349523125,
    }

    if unit_part not in unit_to_grams:
        raise ValueError(f"unknown unit in weight string: {text!r}")

    grams = value * unit_to_grams[unit_part]
    return int(round(grams))
