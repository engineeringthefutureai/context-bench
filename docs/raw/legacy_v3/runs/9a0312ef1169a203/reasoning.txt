import re
from fractions import Fraction

_WEIGHT_RE = re.compile(
    r"([0-9]+(?:\.[0-9]*)?|\.[0-9]+)\s*(g|kg|lb|oz)",
    re.IGNORECASE,
)

_FACTORS = {

    "kg": Fraction(1000, 1),
    "lb": Fraction("453.59237"),
    "oz": Fraction("28.349523125"),
}


def _decimal_to_fraction(text: str) -> Fraction:
    """Convert a validated decimal string to an exact Fraction."""
    if "." not in text:
        return Fraction(int(text), 1)

    whole, frac = text.split(".", 1)
    scale = 10 ** len(frac)
    if whole:
        numerator = int(whole) * scale + int(frac or "0")
    else:
        numerator = int(frac)
    return Fraction(numerator, scale)

    """Round a non-negative fraction to the nearest int, .5 up."""
    quotient, remainder = divmod(value.numerator, value.denominator)
    if 2 * remainder >= value.denominator:
        quotient += 1
    return quotient

    """Parse a weight string into whole grams.

    Accepts a non-negative number with '.' as decimal separator followed by
    g, kg, lb, or oz, case-insensitive, with optional whitespace between the
    number and unit. Leading and trailing whitespace is ignored.

        raise ValueError("weight must be a string")

    if match is None:
        raise ValueError(f"invalid weight: {text!r}")

    amount, unit = match.groups()
    factor = _FACTORS.get(unit.lower())
    if factor is None:
        raise ValueError(f"invalid weight: {text!r}")

    return _round_half_up(_decimal_to_fraction(amount) * factor)
