import re
from fractions import Fraction

_WEIGHT_RE = re.compile(
    r"([0-9]+(?:\.[0-9]*)?|\.[0-9]+)\s*(g|kg|lb|oz)",
    re.IGNORECASE,
)

_FACTORS = {
    "g": Fraction(1, 1),
    "kg": Fraction(1000, 1),
    "lb": Fraction(45359237, 100000),
    "oz": Fraction(28349523125, 1000000000),
}


def _decimal_to_fraction(text: str) -> Fraction:
    """Convert a validated decimal string to an exact Fraction."""
    if "." not in text:
        digits = text.lstrip("0") or "0"
        if digits == "0":
            return Fraction(0, 1)
        return Fraction(int(digits), 1)

    whole, frac = text.split(".", 1)
    digits = (whole + frac).lstrip("0") or "0"
    if digits == "0":
        return Fraction(0, 1)
    return Fraction(int(digits), 10 ** len(frac))


def _round_half_up(value: Fraction) -> int:
    """Round a non-negative fraction to the nearest integer, .5 up."""
    quotient, remainder = divmod(value.numerator, value.denominator)
    if 2 * remainder >= value.denominator:
        quotient += 1
    return quotient


def parse_weight(text: str) -> int:
    """Parse a weight string into whole grams.

    Accepts a non-negative number with "." as decimal separator followed by
    g, kg, lb, or oz, case-insensitive, with optional whitespace between the
    number and unit. Leading and trailing whitespace is ignored.

    Raises ValueError for malformed input, missing number or unit, unknown
    unit, negative values, and "," as a decimal separator.
    """
    if not isinstance(text, str):
        raise ValueError(f"invalid weight: {text!r}")

    match = _WEIGHT_RE.fullmatch(text.strip())
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")

    amount, unit = match.groups()
    factor = _FACTORS.get(unit.lower())
    if factor is None:
        raise ValueError(f"invalid weight: {text!r}")

    return _round_half_up(_decimal_to_fraction(amount) * factor)
