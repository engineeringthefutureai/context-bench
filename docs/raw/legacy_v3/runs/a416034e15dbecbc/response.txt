"""Parse and convert physical quantities that come in as typed text.

Warehouse staff enter package weights in a handful of units ("1.5 kg",
"12 oz", ...); the rest of the system works in whole grams, so the
conversion is centralised here and validated strictly.
"""

from __future__ import annotations

# Exact gram equivalents used by the task; integer form avoids float drift.
_G_PER_UNIT: dict[str, int] = {
    "g": 1000,
    "kg": 1_000_000,
    "lb": 45_359_237,
    "oz": 2_834_952_3125 // 100,
}


def parse_weight(text: str) -> int:
    """Return the weight in whole grams.

    Accepts an integer or decimal (with "." separator) followed by a unit
    of "g", "kg", "lb" or "oz", case-insensitive, with surrounding
    whitespace ignored. 1 lb = 453.59237 g, 1 oz = 28.349523125 g.
    The result is rounded to the nearest gram.

    Raises ValueError on anything else: missing/unknown unit, a negative
    value, or "," as the decimal separator.
    """
    if not isinstance(text, str):
        raise ValueError(f"expected a string, got {type(text).__name__}")

    parts = text.strip().split()
    if len(parts) != 2:
        raise ValueError(f"expected '<number> <unit>', got {text!r}")

    num_text, unit = parts[0], parts[1].lower()
    if unit not in _G_PER_UNIT:
        raise ValueError(f"unknown unit {parts[1]!r}")

    value = _parse_non_negative_decimal(num_text)
    return _round_half_up(value * _G_PER_UNIT[unit])


def _parse_non_negative_decimal(text: str) -> float:
    """Parse a non-negative decimal using "." as the separator.

    Returns a float close enough for rounding to a gram. Raises ValueError
    for negatives, an absent or duplicated separator, or non-digit junk.
    """
    if "," in text:
        raise ValueError("use '.' as the decimal separator")
    if text.startswith("-"):
        raise ValueError("weight must be non-negative")

    if "." in text:
        whole, _, frac = text.partition(".")
        if not frac or "." in frac:
            raise ValueError(f"invalid number {text!r}")
        digits = whole + frac
        scale = 10 ** len(frac)
    else:
        digits, scale = text, 1

    if not digits or not digits.isdigit():
        raise ValueError(f"invalid number {text!r}")
    return int(digits) / scale


def _round_half_up(numerator: int, denominator: int = 1) -> int:
    """Round the (positive) rational numerator/denominator to the nearest
    integer, with halves rounding up (away from zero)."""
    return (2 * numerator + denominator) // (2 * denominator)
