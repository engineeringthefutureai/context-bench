```python
from dataclasses import dataclass, field


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"

    @property
    def total_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

    def add_line(self, sku: str, qty: int, unit_cents: int) -> OrderLine:
        """Add `qty` of `sku` at `unit_cents`; a line at the same price is increased, not duplicated."""
        for line in self.lines:
            if line.sku == sku and line.unit_cents == unit_cents:
                line.qty += qty
                return line
        line = OrderLine(sku, qty, unit_cents)
        self.lines.append(line)
        return line
```
