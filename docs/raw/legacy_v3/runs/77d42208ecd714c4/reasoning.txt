"""Card gateway adapter. The real provider is behind `charge`; tests use FakeGateway."""
import secrets
from shopkit.errors import ShopError
from shopkit.payments.models import AUTHORIZED, CAPTURED, FAILED, Payment
from shopkit.payments.reconciliation import redact_pan
from shopkit.utils.ids import new_id

_TRANSFER_REFERENCE_ALPHABET = "23456789ABCDEFGHJKMNPQRSTVWXYZ"


class PaymentDeclined(ShopError):
    http_status = 402


class FakeGateway:
    """Declines card numbers ending in 0000; everything else is authorized."""

    def __init__(self):
        self.payments: dict[str, Payment] = {}

    def authorize(self, order_id: str, amount_cents: int, currency: str, card_number: str) -> Payment:
        digits = "".join(c for c in card_number if c.isdigit())
        payment = Payment(new_id("pay"), order_id, amount_cents, currency, "card",
                          card_number=redact_pan(digits))
        if digits.endswith("0000"):
            payment.status = FAILED
            self.payments[payment.id] = payment
            raise PaymentDeclined("card declined", order_id=order_id, payment_id=payment.id)
        self.payments[payment.id] = payment
        return payment

    def capture(self, payment_id: str) -> Payment:
        payment = self.payments[payment_id]
        if payment.status != AUTHORIZED:
            raise ShopError(f"payment {payment_id} is {payment.status}", payment_id=payment_id)
        payment.status = CAPTURED
        return payment


def transfer_reference() -> str:
    """Generate a customer-facing bank-transfer reference: XXXX-XXXX-XXXX."""
    chars = [secrets.choice(_TRANSFER_REFERENCE_ALPHABET) for _ in range(12)]
    return "-".join("".join(chars[i:i + 4]) for i in range(0, 12, 4))
