import re

_DURATION_RE = re.compile(r"(?:(\d+)h)?\s*(?:(\d+)m)?", re.IGNORECASE)


def parse_duration(text: str) -> int:
    """Parse a duration like "2h", "45m", "1h30m", or "1h 30m" into minutes."""
    if not isinstance(text, str):
        raise ValueError("invalid duration")

    cleaned = text.strip()
    if not cleaned:
        raise ValueError("invalid duration")

    match = _DURATION_RE.fullmatch(cleaned)
    if match is None:
        raise ValueError("invalid duration")

    hours_text, minutes_text = match.groups()
    if hours_text is None and minutes_text is None:
        raise ValueError("invalid duration")

    hours = int(hours_text) if hours_text is not None else 0
    minutes = int(minutes_text) if minutes_text is not None else 0
    return hours * 60 + minutes
