from dataclasses import dataclass, field


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    def __init__(self, id: str, lines: list[OrderLine] = None, currency: str = "USD"):
        self.id = id
        self.lines = lines if lines is not None else []
        self.currency = currency

    def add_line(self, sku: str, qty: int, unit_cents: int) -> OrderLine:
        """Add qty of sku at unit_cents; merge into an existing line with the same SKU and price."""
        for line in self.lines:
            if line.sku == sku and line.unit_cents == unit_cents:
                line.qty += qty
                return line
        line = OrderLine(sku=sku, qty=qty, unit_cents=unit_cents)
        self.lines.append(line)
        return line

    @property
    def total_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)
