"""Duration parsing for user-supplied text like "2h" or "1h 30m-style" estimates and deadlines."""
import re

# Hours before minutes, each unit at most once, letters case-insensitive. Whitespace around the
# parts (including between them) is tolerated; surrounding whitespace is stripped first.
_HOURS_MINUTES_RE = re.compile(
    r"(?:([0-9]+)\s*h)?(?:\s*([0-9]+)\s*m)?",
    re.IGNORECASE,
)


def parse_duration(text: str) -> int:
    """Whole minutes in hours/minutes text: "2h" -> 120, "45m" -> 45, "1h30m" or "1h 30m" -> 90.
    Hours come before minutes, each unit at most once, units case-insensitive; anything else
    (empty text, other units, a missing number, negatives) raises ValueError."""
    match = _HOURS_MINUTES_RE.fullmatch(text.strip())
    if match is None or (match.group(1) is None and match.group(2) is None):
        raise ValueError(f"invalid duration: {text!r}")
    hours = int(match.group(1) or 0)
    minutes = int(match.group(2) or 0)
    return hours * 60 + minutes
