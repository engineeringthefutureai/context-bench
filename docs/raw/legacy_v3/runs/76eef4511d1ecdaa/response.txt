```python
from dataclasses import dataclass, field

import json


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"
    discount_percent: int = 0

    def __post_init__(self):
        if not (0 <= self.discount_percent <= 100):
            raise ValueError("discount_percent must be between 0 and 100")

    @property
    def total_cents(self) -> int:
        gross = sum(line.total_cents for line in self.lines)
        if self.discount_percent == 0:
            return gross
        discount_cents = (gross * self.discount_percent) // 100
        return gross - discount_cents

    @property
    def discount_cents(self) -> int:
        if self.discount_percent == 0:
            return 0
        gross = sum(line.total_cents for line in self.lines)
        return (gross * self.discount_percent) // 100

    def to_invoice_lines(self) -> list[str]:
        lines = []
        for line in self.lines:
            lines.append(f"{line.sku} x{line.qty} = {line.total_cents} cents")
        if self.discount_percent > 0:
            lines.append(f"Discount ({self.discount_percent}%) = -{self.discount_cents} cents")
        lines.append(f"Total = {self.total_cents} cents")
        return lines

    def to_warehouse_json(self) -> dict:
        return {
            "id": self.id,
            "currency": self.currency,
            "discount_percent": self.discount_percent,
            "lines": [
                {
                    "sku": line.sku,
                    "qty": line.qty,
                    "unit_cents": line.unit_cents,
                    "total_cents": line.total_cents,
                }
                for line in self.lines
            ],
            "total_cents": self.total_cents,
        }

    def to_warehouse_json_string(self) -> str:
        return json.dumps(self.to_warehouse_json(), indent=2)
```
