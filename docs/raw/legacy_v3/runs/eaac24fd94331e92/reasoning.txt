```python
import re
import unicodedata
from dataclasses import dataclass


def _slugify(title: str) -> str:
    """Reduce a product title to a URL-safe slug.

    Lowercases the title, drops characters that are not letters or digits,
    and replaces every run of other characters with a single hyphen.
    """
    value = unicodedata.normalize("NFKD", title).encode("ascii", "ignore").decode("ascii")
    value = re.sub(r"[^a-z0-9]+", "-", value.lower())
    return value.strip("-")


@dataclass
class Product:
    sku: str
    title: str
    price_cents: int
    active: bool = True

    def is_purchasable(self) -> bool:
        return self.active and self.price_cents > 0

    @property
    def url_path(self) -> str:
        return f"/products/{_slugify(self.title)}"
```
