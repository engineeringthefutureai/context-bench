import re

# Hours before minutes, each unit at most once, optional whitespace between parts.
_DURATION = re.compile(r"([0-9]+h)?\s*([0-9]+m)?", re.IGNORECASE)


def parse_duration(text: str) -> int:
    """Total duration in whole minutes, from "2h", "45m", "1h30m" or "1h 30m".

    Units are case-insensitive and surrounding whitespace is ignored; empty
    text, other units, a missing number or minutes before hours raise ValueError.
    """
    match = _DURATION.fullmatch(text.strip())
    if match is None or (match.group(1) is None and match.group(2) is None):
        raise ValueError(f"invalid duration: {text!r}")
    total = 0
    if match.group(1) is not None:
        total += int(match.group(1)[:-1]) * 60
    if match.group(2) is not None:
        total += int(match.group(2)[:-1])
    return total
