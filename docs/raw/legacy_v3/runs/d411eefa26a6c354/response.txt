from dataclasses import dataclass, field


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"
    gift_message: str = ""

    @property
    def total_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

from dataclasses import dataclass, field

from shopkit.orders.models import Order
from shopkit.shipping.packaging import format_weight

MAX_ITEMS_PER_PARCEL = 5


@dataclass
class Parcel:
    order_id: str
    number: int
    items: list[tuple[str, int]] = field(default_factory=list)
    gift_message: str = ""

    @property
    def item_count(self) -> int:
        return sum(qty for _, qty in self.items)


def pick_list(orders: list[Order]) -> list[tuple[str, int]]:
    """Total the units to pick per SKU across *orders*, sorted by SKU."""
    totals: dict[str, int] = {}
    for order in orders:
        for line in order.lines:
            totals[line.sku] = totals.get(line.sku, 0) + line.qty
    return sorted(totals.items())


def split_into_parcels(order: Order, max_items: int = MAX_ITEMS_PER_PARCEL) -> list[Parcel]:
    """Split *order* into parcels holding at most *max_items* units each.

    The order's gift message rides on the first parcel only, so it shows up
    on exactly one packing slip.
    """
    if max_items < 1:
        raise ValueError("max_items must be at least 1")

    parcels: list[Parcel] = []
    current: list[tuple[str, int]] = []
    count = 0
    for line in order.lines:
        remaining = line.qty
        while remaining > 0:
            if count >= max_items:
                parcels.append(Parcel(order_id=order.id, number=len(parcels) + 1, items=current))
                current = []
                count = 0
            take = min(max_items - count, remaining)
            current.append((line.sku, take))
            remaining -= take
            count += take
    if current:
        parcels.append(Parcel(order_id=order.id, number=len(parcels) + 1, items=current))

    if parcels:
        parcels[0].gift_message = order.gift_message
    return parcels


def packing_slip(parcel: Parcel, weight_g: int) -> str:
    lines = [
        "PACKING SLIP",
        f"Order: {parcel.order_id}",
        f"Parcel: {parcel.number} ({parcel.item_count} items)",
        "Items:",
    ]
    for sku, qty in parcel.items:
        lines.append(f"  {qty} x {sku}")
    lines.append(f"Weight: {format_weight(weight_g)}")
    if parcel.gift_message:
        lines.extend(["", f"Gift message: {parcel.gift_message}"])
    return "\n".join(lines)

from shopkit.catalog.models import Product
from shopkit.customers.models import Customer
from shopkit.orders.models import Order


def money(cents: int, currency: str = "USD") -> dict:
    return {"cents": cents, "currency": currency}


def product(p: Product) -> dict:
    return {
        "sku": p.sku,
        "title": p.title,
        "price": money(p.price_cents),
        "active": p.active,
    }


def order(o: Order, status: str | None = None) -> dict:
    data = {
        "id": o.id,
        "currency": o.currency,
        "gift_message": o.gift_message,
        "lines": [
            {
                "sku": line.sku,
                "qty": line.qty,
                "unit": money(line.unit_cents, o.currency),
                "total": money(line.total_cents, o.currency),
            }
            for line in o.lines
        ],
        "total": money(o.total_cents, o.currency),
    }
    if status is not None:
        data["status"] = status
    return data


def customer(c: Customer) -> dict:
    return {
        "id": c.id,
        "email": c.email,
        "name": c.name,
        "tier": c.tier,
        "marketing_opt_in": c.marketing_opt_in,
        "addresses": [
            {
                "name": a.name,
                "line1": a.line1,
                "line2": a.line2,
                "city": a.city,
                "region": a.region,
                "postcode": a.postcode,
                "country": a.country,
            }
            for a in c.addresses
        ],
    }
