```
import re
from fractions import Fraction

_GRAMS_PER_UNIT = {
    "g": Fraction(1, 1),
    "kg": Fraction(1000, 1),
    "lb": Fraction(45359237, 100000),
    "oz": Fraction(28349523125, 1000000000),
}

_WEIGHT_RE = re.compile(
    r"\s*((?:\d+(?:\.\d+)?|\.\d+))\s*(g|kg|lb|oz)\s*",
    re.IGNORECASE,
)


def _decimal_to_fraction(value: str) -> Fraction:
    """Convert a validated non-negative decimal string to an exact Fraction."""
    if "." in value:
        whole, fraction_part = value.split(".")
        digits = (whole or "0") + fraction_part
        return Fraction(int(digits), 10 ** len(fraction_part))
    return Fraction(int(value))


def parse_weight(text: str) -> int:
    """Parse a package weight such as "1.2 kg" and return whole grams.

    The number may be an integer or a decimal using "." as the separator,
    followed by optional whitespace and a case-insensitive unit (g, kg, lb
    or oz). Leading and trailing whitespace is ignored. The result is
    rounded to the nearest gram, with halves rounded up.
    """
    if not isinstance(text, str):
        raise ValueError("weight must be a string")

    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")

    amount = _decimal_to_fraction(match.group(1))
    unit = match.group(2).lower()
    grams = amount * _GRAMS_PER_UNIT[unit]

    whole, remainder = divmod(grams.numerator, grams.denominator)
    if remainder * 2 >= grams.denominator:
        whole += 1
    return int(whole)
```
