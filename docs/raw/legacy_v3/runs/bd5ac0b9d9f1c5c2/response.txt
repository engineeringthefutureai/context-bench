from shopkit.inventory.lots import NotEnoughStock


class Stock:
    """On-hand quantities per SKU."""

    def __init__(self):
        self._on_hand: dict[str, int] = {}

    def receive(self, sku: str, qty: int) -> None:
        self._on_hand[sku] = self._on_hand.get(sku, 0) + qty

    def available(self, sku: str) -> int:
        return self._on_hand.get(sku, 0)

    def reserve(self, sku: str, qty: int) -> None:
        """Reserve ``qty`` units of ``sku``, lowering the available quantity.

        Raises NotEnoughStock (HTTP 409) without changing stock when fewer
        than ``qty`` units are available.
        """
        if qty < 0:
            raise ValueError("qty must be non-negative")
        available = self.available(sku)
        if available < qty:
            raise NotEnoughStock(
                f"Insufficient stock for {sku}: requested {qty}, "
                f"available {available}",
                requested=qty,
                available=available,
            )
        self._on_hand[sku] = available - qty
