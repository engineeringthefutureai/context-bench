"""Unit helpers for warehouse inputs."""
import re
from fractions import Fraction

_WEIGHT_RE = re.compile(r"\s*(\d+(?:\.\d+)?)\s*(g|kg|lb|oz)\s*", re.IGNORECASE)

_UNIT_TO_GRAMS = {
    "g": Fraction(1, 1),
    "kg": Fraction(1000, 1),
    "lb": Fraction(45359237, 100000),
    "oz": Fraction(28349523125, 1000000000),
}

def parse_weight(text: str) -> int:
    """Parse a typed package weight into whole grams.

    Accepts an integer or decimal number followed by g, kg, lb, or oz
    (case-insensitive), with optional whitespace. Raises ValueError for
    anything else, including negatives and comma decimal separators.
    """
    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"Invalid weight format: {text!r}")
    number, unit = match.group(1), match.group(2).lower()
    value = _parse_decimal(number) * _UNIT_TO_GRAMS[unit]
    return _round_half_up(value)

def _parse_decimal(number: str) -> Fraction:
    if "." in number:
        whole, fraction = number.split(".", 1)
        scale = 10 ** len(fraction)
        return Fraction(int(whole) * scale + int(fraction), scale)
    return Fraction(int(number), 1)

def _round_half_up(value: Fraction) -> int:
    numerator, denominator = value.numerator, value.denominator
    return (2 * numerator + denominator) // (2 * denominator)

"""Helpers for parsing warehouse-entered physical units."""

import re
from fractions import Fraction

_WEIGHT_RE = re.compile(
    r"\s*(\d+(?:\.\d+)?)\s*(g|kg|lb|oz)\s*",
    re.IGNORECASE | re.ASCII,
)

_UNIT_TO_GRAMS = {
    "g": Fraction(1, 1),
    "kg": Fraction(1000, 1),
    "lb": Fraction(45359237, 100000),
    "oz": Fraction(28349523125, 1000000000),
}


def parse_weight(text: str) -> int:
    """Parse a typed package weight into whole grams.

    Accepts an integer or decimal number followed by g, kg, lb, or oz
    (case-insensitive), with optional surrounding or separating whitespace.
    Ties to the next gram. Raises ValueError for malformed input, unknown
    units, negative values, and comma decimal separators.
    """
    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError("Invalid weight format")

    number_text, unit_text = match.group(1), match.group(2).lower()
    grams = _parse_decimal(number_text) * _UNIT_TO_GRAMS[unit_text]
    return _round_half_up(grams)


def _parse_decimal(number_text: str) -> Fraction:
    """Convert a strict decimal string to an exact Fraction."""
    if "." in number_text:
        whole, fraction = number_text.split(".", 1)
        scale = 10 ** len(fraction)
        return Fraction(int(whole) * scale + int(fraction), scale)
    return Fraction(int(number_text), 1)


def _round_half_up(value: Fraction) -> int:
    """Round a non-negative Fraction to the nearest integer, ties up."""
    numerator, denominator = value.numerator, value.denominator
    return (2 * numerator + denominator) // (2 * denominator)
