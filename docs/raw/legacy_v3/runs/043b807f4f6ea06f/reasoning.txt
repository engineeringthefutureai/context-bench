"""Unit conversion helpers for warehouse input.

Currently provides parsing of human-typed weights into whole grams.
"""
import re
from decimal import Decimal, ROUND_HALF_UP, localcontext

_WEIGHT_RE = re.compile(
    r"\s*([0-9]+(?:\.[0-9]+)?|\.[0-9]+)\s*(g|kg|lb|oz)\s*",
    re.IGNORECASE,
)

_G_PER_UNIT = {
    "g": Decimal(1),
    "kg": Decimal(1000),
    "lb": Decimal("453.59237"),
    "oz": Decimal("28.349523125"),
}


def parse_weight(text: str) -> int:
    """Parse a weight typed by warehouse staff and return it in whole grams.

    Accepts an integer or decimal number, optional whitespace, then one of g, kg, lb or oz.
    Units are case-insensitive. The result is rounded to the nearest gram, with halves
    rounded up. Raises ValueError for anything else, including negative values, a missing
    number or unit, and "," as a decimal separator.
    """
    if not isinstance(text, str):
        raise ValueError("weight must be a string")
    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError("invalid weight")

    number_text, unit = match.group(1), match.group(2).lower()
    # Give the decimal context enough precision to multiply a long input exactly before
    # rounding, instead of relying on the default 28-digit precision.
    with localcontext() as ctx:
        ctx.prec = max(ctx.prec, len(number_text) + 20)
        grams = Decimal(number_text) * _G_PER_UNIT[unit]
        return int(grams.quantize(Decimal(1), rounding=ROUND_HALF_UP))

"""Parse human-typed weights.
"""
import re
from decimal import Decimal, ROUND_HALF_UP, localcontext

_WEIGHT_RE = re.compile(
    r"\s*([0-9]+(?:\.[0-9]+)?|\.[0-9]+)\s*(g|kg|lb|oz)\s*",
    re.IGNORECASE,
)

_G_PER_UNIT = {
    "g": Decimal(1),
    "kg": Decimal(1000),
    "lb": Decimal("453.59237"),
    "oz": Decimal("28.349523125"),
}


def parse_weight(text: str) -> int:
    """..."""
    if not isinstance(text, str):
        raise ValueError("weight must be a string")
    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError("invalid weight")
    number_text, unit = match.group(1), match.group(2).lower()
    with localcontext() as ctx:
        ctx.prec = max(ctx.prec, len(number_text) + 20)
        grams = Decimal(number_text) * _G_PER_UNIT[unit]
        return int(grams.quantize(Decimal(1), rounding=ROUND_HALF_UP))
