"""Duration text parsing shared by schedules and estimates."""
import re

# Hours then minutes, each unit optional but at most once: "45m", "2h", "1h30m" (an inner space
# such as "1h 30m" is allowed, extra surrounding whitespace is stripped first).
_DURATION_RE = re.compile(r"(\d+h)?\s*(\d+m)?")


def parse_duration(text: str) -> int:
    """Whole minutes from text like "2h", "45m", "1h30m" or "1h 30m" (units case-insensitive).
    Anything else (empty text, other units, a missing number, minutes before hours, negatives)
    raises ValueError."""
    if not isinstance(text, str):
        raise ValueError(f"invalid duration: {text!r}")
    parts = _DURATION_RE.fullmatch(text.strip().lower())
    if not parts or not (parts.group(1) or parts.group(2)):
        raise ValueError(f"invalid duration: {text!r}")
    hours = int(parts.group(1)[:-1]) if parts.group(1) else 0
    minutes = int(parts.group(2)[:-1]) if parts.group(2) else 0
    return hours * 60 + minutes
