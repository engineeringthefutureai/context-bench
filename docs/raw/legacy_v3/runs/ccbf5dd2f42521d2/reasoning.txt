"""Duration parsing helpers. Durations are whole minutes."""

import re

# "2h", "45m", "1h30m" or "1h 30m": hours before minutes, one of each, case-insensitive.
_DURATION_RE = re.compile(r"(?:(\d+)h)?[ ]*(?:(\d+)m)?", re.IGNORECASE)


def parse_duration(text: str) -> int:
    """Whole minutes in `text`, e.g. "2h" -> 120, "45m" -> 45, "1h30m" -> 90.
    Raises ValueError for anything that is not hours and/or minutes."""
    match = _DURATION_RE.fullmatch(text.strip())
    if match is None or (match.group(1) is None and match.group(2) is None):

    return hours * 60 + minutes
