```python
from dataclasses import dataclass, field

import json


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"
    discount_percent: int = 0

    def __post_init__(self):
        if not (0 <= self.discount_percent <= 100):
            raise ValueError("discount_percent must be between 0 and 100")

    @property
    def gross_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

    @property
    def discount_cents(self) -> int:
        if self.discount_percent == 0:
            return 0
        gross = self.gross_cents
        # Round down the discount amount to whole cents
        return (gross * self.discount_percent) // 100

    @property
    def total_cents(self) -> int:
        return self.gross_cents - self.discount_cents

    def to_invoice_lines(self) -> list[str]:
        """Return the lines for the customer invoice."""
        lines = []
        for line in self.lines:
            lines.append(f"{line.sku} x {line.qty} @ ${line.unit_cents / 100:.2f} = ${line.total_cents / 100:.2f}")
        if self.discount_percent > 0:
            lines.append(f"Discount ({self.discount_percent}%) = -${self.discount_cents / 100:.2f}")
        lines.append(f"Total = ${self.total_cents / 100:.2f}")
        return lines

    def to_warehouse_json(self) -> dict:
        """Return the warehouse JSON export dict for this order."""
        return {
            "id": self.id,
            "currency": self.currency,
            "discount_percent": self.discount_percent,
            "lines": [
                {
                    "sku": line.sku,
                    "qty": line.qty,
                    "unit_cents": line.unit_cents,
                }
                for line in self.lines
            ],
            "total_cents": self.total_cents,
        }
```
