"""Duration parsing: "2h", "45m", "1h30m", "1h 30m" -> whole minutes."""
import re

# Hours before minutes, each unit at most once, unit attached to its number; optional
# whitespace between the parts; case-insensitive.
_DURATION = re.compile(r"^(?:(?P<hours>[0-9]+)h)?\s*(?:(?P<minutes>[0-9]+)m)?$", re.IGNORECASE)


def parse_duration(text: str) -> int:
    """Whole minutes for a duration like "2h", "45m", "1h 30m" or "1H30M": hours come before
    minutes, each unit appears at most once, units are case-insensitive and surrounding
    whitespace is ignored. Raises ValueError for anything else: empty text, other units, a
    missing number, minutes before hours, negative numbers."""
    if not isinstance(text, str):
        raise ValueError(f"invalid duration {text!r}")
    m = _DURATION.match(text.strip())
    if m is None or (m["hours"] is None and m["minutes"] is None):
        raise ValueError(f"invalid duration {text!r}")
    hours = int(m["hours"] or 0)
    minutes = int(m["minutes"] or 0)
    return hours * 60 + minutes
