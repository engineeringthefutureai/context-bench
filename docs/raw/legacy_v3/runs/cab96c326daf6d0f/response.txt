"""Helpers for parsing human-readable durations into whole minutes."""

import re

_DURATION_RE = re.compile(r"^\s*(?:(\d+)h)?\s*(?:(\d+)m)?\s*$", re.IGNORECASE)


def parse_duration(text: str) -> int:
    """Parse a duration string like ``"1h30m"`` into whole minutes.

    Args:
        text: A duration expressed as hours and/or minutes, e.g. ``"2h"``,
            ``"45m"`` or ``"1h 30m"``. Units are case-insensitive and
            surrounding whitespace is ignored.

    Returns:
        The total number of whole minutes.

    Raises:
        ValueError: If ``text`` does not match the expected format, is empty,
            uses other units, has a missing number, places minutes before
            hours, or contains a negative number.
    """
    if not isinstance(text, str):
        raise ValueError(f"duration must be a string, got {type(text).__name__}")

    match = _DURATION_RE.match(text)
    if match is None:
        raise ValueError(f"invalid duration: {text!r}")

    hours_str, minutes_str = match.groups()

    # Ensure at least one unit is present
    if hours_str is None and minutes_str is None:
        raise ValueError(f"invalid duration: {text!r}")

    hours = int(hours_str) if hours_str is not None else 0
    minutes = int(minutes_str) if minutes_str is not None else 0

    return hours * 60 + minutes
