from dataclasses import dataclass, field
from typing import Any


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Parcel:
    lines: list[OrderLine] = field(default_factory=list)
    gift_message: str = ""

    @property
    def weight_g(self) -> int:
        return sum(line.qty * 100 for line in self.lines)

    @property
    def total_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"
    gift_message: str = ""

    @property
    def total_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

    def split_into_parcels(self, max_weight_g: int = 2000) -> list[Parcel]:
        """Split lines into parcels by weight. The first parcel carries the gift message."""
        parcels: list[Parcel] = []
        current_lines: list[OrderLine] = []
        current_weight = 0

        for line in self.lines:
            line_weight = line.qty * 100
            if current_lines and current_weight + line_weight > max_weight_g:
                message = self.gift_message if not parcels else ""
                parcels.append(Parcel(lines=current_lines, gift_message=message))
                current_lines = [line]
                current_weight = line_weight
            else:
                current_lines.append(line)
                current_weight += line_weight

        if current_lines:
            message = self.gift_message if not parcels else ""
            parcels.append(Parcel(lines=current_lines, gift_message=message))
        return parcels

    def to_api_dict(self) -> dict[str, Any]:
        return {
            "id": self.id,
            "currency": self.currency,
            "gift_message": self.gift_message,
            "total_cents": self.total_cents,
            "lines": [
                {
                    "sku": line.sku,
                    "qty": line.qty,
                    "unit_cents": line.unit_cents,
                    "total_cents": line.total_cents,
                }
                for line in self.lines
            ],
        }


def packing_slip(parcel: Parcel, weight_g: int) -> str:
    """Render the packing slip for a parcel. Includes the gift message when present."""
    lines = [
        f"Packing slip — parcel weight {weight_g} g",
    ]
    if parcel.gift_message:
        lines.append(f"Gift message: {parcel.gift_message}")
    for line in parcel.lines:
        lines.append(f"{line.qty} x {line.sku} @ {line.unit_cents}c")
    return "\n".join(lines)


def customer_invoice(order: Order) -> str:
    """Render the customer invoice. The gift message is intentionally omitted."""
    lines = [
        f"Invoice for order {order.id} ({order.currency})",
    ]
    for line in order.lines:
        lines.append(f"{line.qty} x {line.sku} @ {line.unit_cents}c")
    lines.append(f"Total: {order.total_cents}c")
    return "\n".join(lines)
