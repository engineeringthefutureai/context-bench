```
_WEIGHT_PATTERN = re.compile(

    "kg": (1000, 0),
    "lb": (45359237, 5),
    "oz": (28349523125, 9),
}


def parse_weight(text: str) -> int:
    """Parse a weight typed by warehouse staff and return whole grams.

    Accepts an unsigned integer or decimal number using "." followed by a
    unit of g, kg, lb, or oz, ignoring surrounding whitespace.
    """
    if not isinstance(text, str):
        raise ValueError("weight must be a string")

    match = _WEIGHT_PATTERN.fullmatch(text)
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")

    number = match.group("number")
    unit = match.group("unit").lower()

    if "." in number:
        whole, _, frac = number.partition(".")
    else:
        whole, frac = number, ""

    digits = whole + frac
    if not digits:
        raise ValueError(f"invalid weight: {text!r}")

    value_num = int(digits)
    denom_exp = len(frac) + _UNIT_GRAMS[unit][1]
    unit_num = _UNIT_GRAMS[unit][0]
    denom = 10 ** denom_exp

    return (2 * value_num * unit_num + denom) // (2 * denom)
```

```
import re

_WEIGHT_PATTERN = re.compile(
    r"\s*(?P<number>[0-9]+(?:\.[0-9]*)?|\.[0-9]+)\s*(?P<unit>kg|g|lb|oz)\s*",
    re.IGNORECASE | re.ASCII,
)

# Each unit is expressed as (numerator, denominator) in grams, so the factor
# is numerator / denominator.
_UNIT_GRAMS = {
    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (28349523125, 1000000000),
}


def parse_weight(text: str) -> int:
    """Parse a weight such as "1.5 kg" and return the whole-gram weight.

    The input must be an unsigned integer or decimal number using "." as the
    decimal separator, optional whitespace, and one of the units g, kg, lb, or
    oz (case-insensitive). The result is rounded to the nearest gram.
    """
    if not isinstance(text, str):
        raise ValueError("weight must be a string")

    match = _WEIGHT_PATTERN.fullmatch(text)
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")

    number = match.group("number")
    unit = match.group("unit").lower()

    whole, _, frac = number.partition(".")
    digits = (whole + frac).lstrip("0")
    if not digits:
        return 0

    value_num = int(digits)
    unit_num, unit_denom = _UNIT_GRAMS[unit]
    denom = 10 ** len(frac) * unit_denom

    # floor(value_num * unit_num / denom + 1/2), computed with integers.
    return (2 * value_num * unit_num + denom) // (2 * denom)
```

```
import re

_WEIGHT_RE = re.compile(
    r"\s*(?P<number>[0-9]+(?:\.[0-9]+)?|\.[0-9]+)\s*(?P<unit>kg|g|lb|oz)\s*",
    re.IGNORECASE | re.ASCII,
)

# Each unit is expressed as (numerator, denominator) in grams, so the factor
# is numerator / denominator.
_UNIT_GRAMS = {
    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (28349523125, 1000000000),
}


def parse_weight(text: str) -> int:
    """Parse a weight such as "1.5 kg" and return the whole-gram weight.

    The input must be an unsigned integer or decimal number using "." as the
    decimal separator, optional whitespace, and one of the units g, kg, lb, or
    oz (case-insensitive). The result is rounded to the nearest gram.
    """
    if not isinstance(text, str):
        raise ValueError("weight must be a string")

    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")

    number = match.group("number")
    unit = match.group("unit").lower()

    whole, _, frac = number.partition(".")
    digits = (whole + frac
    if not digits.strip("0"):
        return 0

    value_num = int(digits.strip("0"))
    unit_num, unit_denom = _UNIT_GRAMS[unit]
    denom = 10 ** len(frac) * unit_denom

    # Round half-up: floor((value_num / denom_fraction) + 0.5) using integers.
    return (2 * value_num * unit_num + denom) // (2 * denom)
```
