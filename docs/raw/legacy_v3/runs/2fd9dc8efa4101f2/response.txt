"""Refunds against captured payments. Partial refunds are allowed up to the captured amount."""
from dataclasses import dataclass, field
from datetime import: date, datetime

from shopkit.errors import Conflict
from shopkit.money import format_money
from shopkit.payments.models import Payment
from shopkit.utils.dates import add_business_days, utc_now


class RefundTooLarge(Conflict):
    pass


@dataclass(frozen=True)
class Refund:
    payment_id: str
    amount_cents: int
    reason: str
    at: datetime = field(default_factory=utc_now)


#: Business days a card network takes to post a refund to the customer's card.
BANK_PROCESSING_DAYS = 5


def expected_arrival(refund: Refund) -> date:
    """The day the refunded money is expected to reach the customer's card.

    Banks need five business days, counted from the day the refund was issued
    (the shop's usual business-day calendar: weekends and HOLIDAYS don't count).
    """
    return add_business_days(refund.at.date(), BANK_PROCESSING_DAYS)
