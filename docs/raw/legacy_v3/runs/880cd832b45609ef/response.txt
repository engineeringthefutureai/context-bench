import re

_WEIGHT_RE = re.compile(
    r"\s*(?P<number>\d+(?:\.\d+)?)\s*(?P<unit>kg|lb|oz|g)\s*",
    re.IGNORECASE | re.ASCII,
)

_UNIT_FACTORS = {
    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (28349523125, 1000000000),
}


def parse_weight(text: str) -> int:
    """Parse a weight string into whole grams, rounded to the nearest gram.

    Accepts a non-negative integer or decimal number followed by g, kg, lb,
    or oz, case-insensitively, with optional whitespace around the parts.
    """
    if not isinstance(text, str):
        raise ValueError("invalid weight")

    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError("invalid weight")

    number = match.group("number")
    unit = match.group("unit").lower()
    factor_numerator, factor_denominator = _UNIT_FACTORS[unit]

    if "." in number:
        whole, fraction = number.split(".")
        numerator = int(whole + fraction)
        scale = len(fraction)
    else:
        numerator = int(number)
        scale = 0

    if numerator == 0:
        return 0

    # Drop trailing zeros so 1.20 is treated as 12/10, not 120/100.
    while scale and numerator % 10 == 0:
        numerator //= 10
        scale -= 1

    denominator = (10 ** scale) * factor_denominator
    value_numerator = numerator * factor_numerator

    # Round half up for non-negative values: floor(value + 1/2).
    return (2 * value_numerator + denominator) // (2 * denominator)
