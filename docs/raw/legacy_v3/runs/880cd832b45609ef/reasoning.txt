    r"\s*(?P<number>\d+(?:\.\d+)?)\s*(?P<unit>kg|lb|oz|g)\s*",
    re.IGNORECASE,

_UNIT_FACTORS = {

    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (45359237, 1600000),
}

def parse_weight(text: str) -> int:
    """Parse a weight string into whole grams, rounded to the nearest gram."""
    if not isinstance(text, str):
        raise ValueError("invalid weight")
    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError("invalid weight")
    number = match.group("number")
    unit = match.group("unit").lower()
    factor_num, factor_den = _UNIT_FACTORS[unit]
    if "." in number:
        whole, fraction = number.split(".")
        numerator = int(whole + fraction)
        scale = len(fraction)
    else:
