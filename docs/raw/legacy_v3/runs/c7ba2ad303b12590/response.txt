class InsufficientStockError(Exception):
    """Raised when a reservation requests more stock than is available."""

    def __init__(self, sku: str, requested: int, available: int):
        self.sku = sku
        self.requested = requested
        self.available = available
        super().__init__(
            f"Insufficient stock for {sku}: requested {requested}, available {available}"
        )

class Stock:
    """On-hand quantities per SKU."""

    def __init__(self):
        self._on_hand: dict[str, int] = {}

    def receive(self, sku: str, qty: int) -> None:
        self._on_hand[sku] = self._on_hand.get(sku, 0) + qty

    def available(self, sku: str) -> int:
        return self._on_hand.get(sku, 0)

    def reserve(self, sku: str, qty: int) -> None:
        """Lower the available quantity of `sku` by `qty`.

        Raises InsufficientStockError if the available quantity is less than `qty`.
        """
        avail = self.available(sku)
        if qty > avail:
            raise InsufficientStockError(sku, qty, avail)
        self._on_hand[sku] = avail - qty
