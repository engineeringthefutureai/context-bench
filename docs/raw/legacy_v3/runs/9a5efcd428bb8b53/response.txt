"""Utilities for parsing and formatting human-readable durations."""

import re


def parse_duration(text: str) -> int:
    """Parse a duration string into whole minutes.

    Valid inputs (case-insensitive, whitespace-tolerant):
        "2h" -> 120
        "45m" -> 45
        "1h30m" -> 90
        "1h 30m" -> 90
    Hours may be combined with minutes (hours first, each unit at most once).
    A missing number, minutes before hours, negative numbers, or any other
    format raises ``ValueError``.
    """
    if text is None:
        raise ValueError("duration text must not be None")

    stripped = text.strip()
    if not stripped:
        raise ValueError("empty duration text")

    # Normalize whitespace: collapse multiple spaces to one, then drop all
    # spaces so "1h 30m" becomes "1h30m".
    normalized = re.sub(r"\s+", "", stripped)

    # Match exactly: optional hours followed by optional minutes, each with
    # a non-negative integer and a unit letter.  No leading signs allowed.
    pattern = re.compile(r"^(\d+[hH])?(\d+[mM])?$")
    match = pattern.match(normalized)
    if not match:
        raise ValueError(f"invalid duration: {text!r}")

    hours_part = match.group(1)
    minutes_part = match.group(2)

    if hours_part is None and minutes_part is None:
        raise ValueError(f"invalid duration: {text!r}")

    hours = 0
    if hours_part is not None:
        hours = int(hours_part[:-1])

    minutes = 0
    if minutes_part is not None:
        minutes = int(minutes_part[:-1])

    return hours * 60 + minutes
