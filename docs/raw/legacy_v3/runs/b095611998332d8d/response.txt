import csv
import io
import json

from shopkit.orders.models import Order
from shopkit.utils.text import normalize_sku


def export_orders_json(orders: list[Order]) -> str:
    """Raw JSON export for the warehouse integration."""
    payload = [
        {
            "id": o.id,
            "currency": o.currency,
            "lines": [{"sku": l.sku, "qty": l.qty, "unit_cents": l.unit_cents} for l in o.lines],
        }
        for o in orders
    ]
    return json.dumps(payload, indent=2)


def _money(cents: int) -> str:
    """Render a cent amount the way customer invoices show it."""
    sign = "-" if cents < 0 else ""
    c = abs(cents)
    return f"{sign}${c // 100:,}.{c % 100:02d}"


def export_orders_csv(orders: list[Order]) -> str:
    """One row per order line, for the finance spreadsheet.

    Product codes are normalised the same way they are everywhere else, and
    line totals match the customer-invoice money format.
    """
    buf = io.StringIO()
    writer = csv.writer(buf)
    writer.writerow(["order_id", "product_code", "quantity", "line_total"])
    for o in orders:
        for l in o.lines:
            writer.writerow([
                o.id,
                normalize_sku(l.sku),
                l.qty,
                _money(l.total_cents),
            ])
    return buf.getvalue()
