"""Parse human-friendly duration strings into whole minutes.

Accepts hours and/or minutes, e.g. ``"2h"``, ``"45m"``, ``"1h30m"`` or
``"1h 30m"``. Hours must come before minutes, each unit may appear at most
once, units are case-insensitive, and surrounding whitespace is ignored.

Anything that does not match those rules (empty text, unknown units, a
missing or non-numeric count, minutes before hours, negative numbers, etc.)
raises :class:`ValueError`.
"""

import re

# One or two "number + unit" pieces in hour -> minute order, with optional
# whitespace between them.
_DURATION_RE = re.compile(
    r"^\s*(?:(\d+)h)?(?:(\d+)m)?\s*$",
    re.IGNORECASE,
)


def parse_duration(text: str) -> int:
    """Return the duration described by ``text`` in whole minutes.

    ``text`` may combine hours and minutes: ``"2h"`` -> 120, ``"45m"`` -> 45,
    ``"1h30m"`` / ``"1h 30m"`` -> 90. Hours come first, each unit at most
    once, case is ignored and surrounding whitespace is stripped.

    :param text: The duration string to parse.
    :raises ValueError: If ``text`` is empty, has a missing or invalid number,
        an unknown unit, minutes before hours, or is otherwise malformed.
    """
    if not isinstance(text, str) or not text.strip():
        raise ValueError("duration must be a non-empty string of hours and/or minutes")

    match = _DURATION_RE.match(text)
    if not match or (match.group(1) is None and match.group(2) is None):
        raise ValueError(f"invalid duration: {text!r}")

    hours = int(match.group(1)) if match.group(1) is not None else 0
    minutes = int(match.group(2)) if match.group(2) is not None else 0
    return hours * 60 + minutes
