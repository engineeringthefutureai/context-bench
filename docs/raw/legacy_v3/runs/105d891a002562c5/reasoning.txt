"""Parsing of human-friendly duration strings."""

import re

_DURATION_RE = re.compile(
    r"\s*(?:(?P<hours>[0-9]+)h)?\s*(?:(?P<minutes>[0-9]+)m)?\s*",
    re.IGNORECASE,
)


def parse_duration(text: str) -> int:
    """Return the number of minutes in ``text``.

    Valid inputs use an optional hours part before an optional minutes part,
    for example ``"2h"``, ``"45m"``, ``"1h30m"`` or ``"1h 30m"`` with spaces.
    """
    if not isinstance(text, str):
        raise ValueError("duration must be a string")

    match = _DURATION_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"Invalid duration: {text!r}")

    hours = match.group("hours")
    minutes = match.group("minutes")
    if hours is None and minutes is None:
        raise ValueError(f"Invalid duration: {text!r}")

    hours_value = int(hours) if hours is not None else 0
    minutes_value = int(minutes) if minutes is not None else 0
    return hours_value * 60 + minutes_value
