"""Helpers for parsing and converting package weight units."""

import re

# Exact conversion factors to grams.
_LB_TO_G = 453.59237
_OZ_TO_G = 28.349523125


def _parse_number(number_str: str) -> float:
    """Parse the numeric portion of the weight string.

    Raises ValueError if the number is missing or malformed (e.g. uses ','
    as a decimal separator).
    """
    if not number_str:
        raise ValueError("Missing number in weight string")
    # Reject commas as decimal separators
    if ',' in number_str:
        raise ValueError("Invalid number: comma is not allowed as decimal separator")
    try:
        value = float(number_str)
    except (ValueError, OverflowError):
        raise ValueError(f"Invalid number: {number_str!r}")
    # Reject inf/nan
    if value != value or value == float('inf') or value == float('-inf'):
        raise ValueError(f"Invalid number: {number_str!r}")
    if value < 0:
        raise ValueError(f"Negative weight not allowed: {number_str!r}")
    return value


def parse_weight(text: str) -> int:
    """Parse a weight string like "1.5 kg" and return the weight in whole grams.

    Accepts an integer or decimal (using "." as separator), optional whitespace,
    followed by a unit: g, kg, lb, or oz (case-insensitive).
    Leading and trailing whitespace is ignored.
    Returns the weight rounded to the nearest gram.

    Raises:
        ValueError: For invalid input, including missing/unknown unit, missing
            number, negative values, or "," as decimal separator.
    """
    if not isinstance(text, str):
        raise ValueError("Input must be a string")

    # Strip leading/trailing whitespace
    stripped = text.strip()
    if not stripped:
        raise ValueError("Empty weight string")

    # Find the last token as the unit
    # Split from the right to handle things like "1.5kg" (no space)
    # First, try to find where the number ends and the unit begins.
    # Strategy: find the last non-alphanumeric/non-dot character or use a regex.

    # Use regex to match: optional whitespace, number (int or decimal), optional whitespace, unit
    # Number: \d+\.?\d* (integer or decimal)
    # Unit: (kg|lb|oz|g)
    # Case-insensitive for unit
    pattern = r'^\s*(\d+(?:\.\d*)?)\s*(kg|lb|oz|g)\s*$'
    match = re.match(pattern, stripped, re.IGNORECASE)

    if not match:
        raise ValueError(f"Invalid weight format: {text!r}")

    number_str = match.group(1)
    unit = match.group(2).lower()

    value = _parse_number(number_str)

    if unit == 'g':
        grams = value
    elif unit == 'kg':
        grams = value * 1000.0
    elif unit == 'lb':
        grams = value * _LB_TO_G
    elif unit == 'oz':
        grams = value * _OZ_TO_G
    else:
        raise ValueError(f"Unknown unit: {unit!r}")

    # Round to the nearest gram
    return int(round(grams))
