class Stock:
    """On-hand quantities per SKU."""

    def __init__(self) -> None:
        self._on_hand: dict[str, int] = {}

    def receive(self, sku: str, qty: int) -> None:
        self._on_hand[sku] = self._on_hand.get(sku, 0) + qty

    def available(self, sku: str) -> int:
        return self._on_hand.get(sku, 0)

    def reserve(self, sku: str, qty: int) -> None:
        """Hold `qty` units of `sku`, lowering available stock.

        Raises InsufficientStockError if `qty` exceeds available stock.
        """
        if qty < 0:
            raise ValueError("qty must be non-negative")
        available_qty = self.available(sku)
        if qty > available_qty:
            raise InsufficientStockError(sku, qty, available_qty)
        self._on_hand[sku] = available_qty - qty

class Stock:
    """On-hand quantities per SKU."""

    def __init__(self) -> None:
        self._on_hand: dict[str, int] = {}

    def receive(self, sku: str, qty: int) -> None:
        self._on_hand[sku] = self._on_hand.get(sku, 0) + qty

    def available(self, sku: str) -> int:
        return self._on_hand.get(sku, 0)

    def reserve(self, sku: str, qty: int) -> None:
        """Hold `qty` units of `sku`, lowering the available quantity.

        If `qty` exceeds the available quantity, no stock is changed and
        `InsufficientStockError` is raised with the requested and available
        quantities.
        """
        if qty < 0:
            raise ValueError("qty must be non-negative")
        if qty == 0:
            return
        available_qty = self.available(sku)
        if qty > available_qty:
            raise InsufficientStockError(sku, qty, available_qty)
        self._on_hand[sku] = available_qty - qty
