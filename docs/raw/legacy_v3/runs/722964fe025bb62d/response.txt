"""Utilities for parsing and working with human-readable durations.

This module provides functions to parse textual duration representations
(such as "2h", "45m", "1h30m", "1h 30m") into whole minutes.
"""

import re

# Pattern to match optional hours followed by optional minutes.
# Hours must appear before minutes.
# Each unit appears at most once.
# Units are case-insensitive.
_DURATION_RE = re.compile(r"""
    ^\s*
    (?:(?P<hours>\d+)\s*h)?
    (?:(?P<minutes>\d+)\s*m)?
    \s*$
""", re.VERBOSE | re.IGNORECASE)


def parse_duration(text: str) -> int:
    """
    Parse a duration string into whole minutes.

    Accepted formats: "2h", "45m", "1h30m", "1h 30m", etc.
    Hours come before minutes, each unit at most once, case-insensitive,
    surrounding whitespace is ignored.

    Args:
        text: The duration string to parse.

    Returns:
        The duration in whole minutes.

    Raises:
        ValueError: If the input is invalid (empty, wrong units, missing
                    number, minutes before hours, negative numbers, etc.).
    """
    if not isinstance(text, str):
        raise ValueError("duration must be a string")

    stripped = text.strip()
    if not stripped:
        raise ValueError("duration string is empty")

    match = _DURATION_RE.match(stripped)
    if not match:
        raise ValueError(f"invalid duration format: {text!r}")

    hours = int(match.group('hours')) if match.group('hours') is not None else 0
    minutes = int(match.group('minutes')) if match.group('minutes') is not None else 0

    # Ensure at least one unit is present (regex already ensures this,
    # but be explicit)
    if match.group('hours') is None and match.group('minutes') is None:
        raise ValueError(f"invalid duration format: {text!r}")

    return hours * 60 + minutes
