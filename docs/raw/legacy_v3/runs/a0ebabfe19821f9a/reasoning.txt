from dataclasses import dataclass, field

from shopkit.catalog.models import Product
from shopkit.orders.models import Order

MAX_ITEMS_PER_PARCEL = 10


@dataclass
class Parcel:
    order_id: str
    number: int
    items: list[tuple[str, int]] = field(default_factory=list)

    @property
    def item_count(self) -> int:
        return sum(qty for _, qty in self.items)


def pick_list(orders: list[Order]) -> list[tuple[str, int]]:
    picked: dict[str, int] = {}
    for order in orders:
        for line in order.lines:
            picked[line.sku] = picked.get(line.sku, 0) + line.qty
    return list(picked.items())


def split_into_parcels(order: Order, max_items: int = MAX_ITEMS_PER_PARCEL) -> list[Parcel]:
    units: list[str] = []
    for line in order.lines:
        units.extend([line.sku] * line.qty)
    parcels: list[Parcel] = []
    for start in range(0, len(units), max_items):
        chunk = units[start:start + max_items]
        items: dict[str, int] = {}
        for sku in chunk:
            items[sku] = items.get(sku, 0) + 1
        parcels.append(Parcel(order_id=order.id, number=len(parcels) + 1, items=list(items.items())))
    return parcels

MAX_ITEMS_PER_PARCEL = 10


@dataclass
class Parcel:
    order_id: str
    number: int
    items: list[tuple[str, int]] = field(default_factory=list)

    @property
    def item_count(self) -> int:
        return sum(qty for _, qty in self.items)


def pick_list(orders: list[Order]) -> list[tuple[str, int]]:
    """Aggregate the quantities to pick across a batch of orders."""
    picked: dict[str, int] = {}
    for order in orders:
        for line in order.lines:
            picked[line.sku] = picked.get(line.sku, 0) + line.qty
    return list(picked.items())


def split_into_parcels(order: Order, max_items: int = MAX_ITEMS_PER_PARCEL) -> list[Parcel]:
    parcels: list[Parcel] = []
    pending: list[str] = []
    for line in order.lines:
        for _ in range(line.qty):
            pending.append(line.sku)
            if len(pending) >= max_items:
                parcels.append(_parcel(order.id, len(parcels) + 1, pending))
                pending = []
    if pending:
        parcels.append(_parcel(order.id, len(parcels) + 1, pending))
    return parcels

    items: dict[str, int] = {}
    for sku in skus:
        items[sku] = items.get(sku, 0) + 1
    return Parcel(order_id=order_id, number=number, items=list(items.items()))
