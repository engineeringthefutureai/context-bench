from dataclasses import dataclass

from shopkit.utils.strings import slugify


@dataclass
class Product:
    sku: str
    title: str
    price_cents: int
    active: bool = True

    def is_purchasable(self) -> bool:
        return self.active and self.price_cents > 0

    @property
    def url_path(self) -> str:
        # Stable storefront address for this product, e.g. "/products/insulated-bottle".
        return f"/products/{slugify(self.title)}"
