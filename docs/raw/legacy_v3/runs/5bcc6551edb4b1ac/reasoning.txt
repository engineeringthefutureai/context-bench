```python
import hmac
import json
from dataclasses import dataclass, field

from shopkit.events import Event, EventBus
from shopkit.payments.reconciliation import redact_pan

MAX_ATTEMPTS = 5


@dataclass
class Endpoint:
    url: str
    secret: str
    events: set[str]                          # event names, or "order.*"-style prefixes


@dataclass
class Delivery:
    endpoint: Endpoint
    body: str
    signature: str
    attempts: int = 0
    delivered: bool = False
    errors: list[str] = field(default_factory=list)


def sign(body: str, secret: str) -> str:
    return "sha256=" + hmac.new(secret.encode(), body.encode(), hashlib.sha256).hexdigest()


def verify(body: str, secret: str, signature: str) -> bool:
    return hmac.compare_digest(sign(body, secret), signature)


def _subscribed(endpoint: Endpoint, name: str) -> bool:
    return name in endpoint.events or f"{name.split('.', 1)[0]}.*" in endpoint.events


def _mask_card_numbers(value: Any) -> Any:
    """Deep copy of an event payload with every card_number field masked the same way payments
    mask them when stored (redact_pan). All other fields pass through unchanged."""
    if isinstance(value, dict):
        return {
            key: redact_pan(val) if key == "card_number" and isinstance(val, str)
            else _mask_card_numbers(val)
            for key, val in value.items()
        }
    if isinstance(value, list):
        return [_mask_card_numbers(item) for item in value]
    return value


class WebhookDispatcher:
    def __init__(self, endpoints: list[Endpoint], transport=None):
        self.endpoints = endpoints
        self.queue: list[Delivery] = []
        self._transport = transport           # callable(url, body, headers) -> status code

    def attach(self, bus: EventBus) -> None:
        for prefix in {e.split(".", 1)[0] for ep in self.endpoints for e in ep.events}:
            bus.subscribe(f"{prefix}.*", self.on_event)

    def on_event(self, event: Event) -> None:
        body = json.dumps({"event": event.name, "at": event.at.isoformat(),
                           "data": _mask_card_numbers(event.payload)},
                          sort_keys=True, default=str)
        for ep in self.endpoints:
            if _subscribed(ep, event.name):
                self.queue.append(Delivery(ep, body, sign(body, ep.secret)))

    def flush(self) -> int:
        """Try every pending delivery once; returns how many succeeded. Failures stay queued until
        MAX_ATTEMPTS."""
        ok = 0
        for d in list(self.queue):
            d.attempts += 1
            try:
                status = self._transport(d.endpoint.url, d.body, {"X-Signature": d.signature})
                d.delivered = 200 <= status < 300
                if not d.delivered:
                    d.errors.append(f"HTTP {status}")
            except OSError as e:
                d.errors.append(str(e))
            if d.delivered or d.attempts >= MAX_ATTEMPTS:
                self.queue.remove(d)
                ok += d.delivered
        return ok
```

```python
"""Outgoing webhooks to partner systems (ERP, marketing). Payloads are signed with HMAC-SHA256."""
import hashlib
import hmac
import json
from dataclasses import dataclass, field
from typing import Any

from shopkit.events import Event, EventBus
from shopkit.payments.reconciliation import redact_pan

MAX_ATTEMPTS = 5


@dataclass
class Endpoint:
    url: str
    secret: str
    events: set[str]                          # event names, or "order.*"-style prefixes


@dataclass
class Delivery:
    endpoint: Endpoint
    body: str
    signature: str
    attempts: int = 0
    delivered: bool = False
    errors: list[str] = field(default_factory=list)


def sign(body: str, secret: str) -> str:
    return "sha256=" + hmac.new(secret.encode(), body.encode(), hashlib.sha256).hexdigest()


def verify(body: str, secret: str, signature: str) -> bool:
    return hmac.compare_digest(sign(body, secret), signature)


def _subscribed(endpoint: Endpoint, name: str) -> bool:
    return name in endpoint.events or f"{name.split('.', 1)[0]}.*" in endpoint.events


def _mask_card_numbers(value: Any) -> Any:
```

```python
    def on_event(self, event: Event) -> None:
        body = json.dumps({"event": event.name, "at": event.at.isoformat(),
                           "data": _mask_card_numbers(event.payload)},
                          sort_keys=True, default=str)
        for ep in self.endpoints:
            if _subscribed(ep, event.name):
                self.queue.append(Delivery(ep, body, sign(body, ep.secret)))

    def flush(self) -> int:
        """Try every pending delivery once; returns how many succeeded. Failures stay queued until
        MAX_ATTEMPTS."""
        ok = 0
        for d in list(self.queue):
            d.attempts += 1
            try:
                status = self._transport(d.endpoint.url, d.body, {"X-Signature": d.signature})
                d.delivered = 200 <= status < 300
                if not d.delivered:
                    d.errors.append(f"HTTP {status}")
            except OSError as e:
                d.errors.append(str(e))
            if d.delivered or d.attempts >= MAX_ATTEMPTS:
                self.queue.remove(d)
                ok += d.delivered
        return ok
```
