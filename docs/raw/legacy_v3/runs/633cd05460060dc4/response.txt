from dataclasses import dataclass, field
from decimal import ROUND_DOWN, Decimal


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"
    discount_percent: int = 0

    def __post_init__(self):
        if not 0 <= self.discount_percent <= 100:
            raise ValueError(
                f"discount_percent must be between 0 and 100, got {self.discount_percent!r}"
            )

    @property
    def subtotal_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

    @property
    def discount_cents(self) -> int:
        """Discount amount in whole cents, rounded down."""
        if not self.discount_percent:
            return 0
        return int(
            Decimal(self.subtotal_cents) * Decimal(self.discount_percent)
            / Decimal(100)
        )

    @property
    def total_cents(self) -> int:
        return self.subtotal_cents - self.discount_cents

from shopkit.orders.models import Order

def export_orders_json(orders: list[Order]) -> str:
    """Raw JSON export for the warehouse integration."""
    payload = [
        {
            "id": o.id,
            "currency": o.currency,
            "discount_percent": o.discount_percent,
            "lines": [{"sku": l.sku, "qty": l.qty, "unit_cents": l.unit_cents} for l in o.lines],
        }
        for o in orders
    ]
    return json.dumps(payload, indent=2)
