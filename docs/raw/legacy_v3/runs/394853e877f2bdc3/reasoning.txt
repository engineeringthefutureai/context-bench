```python
"""Weights typed in by warehouse staff, normalised to whole grams."""

import re
from decimal import Decimal, ROUND_HALF_UP

# Exact definitions of the supported units in grams.
_GRAMS_PER_UNIT = {
    "g": Decimal("1"),
    "kg": Decimal("1000"),
    "lb": Decimal("453.59237"),
    "oz": Decimal("28.349523125"),
}

# Number (integer or decimal using "."), optional whitespace, then a unit.
_WEIGHT_RE = re.compile(r"(\d+(?:\.\d+)?)\s*(g|kg|lb|oz)", re.IGNORECASE)


def parse_weight(text: str) -> int:
    """Parse a staff-typed weight such as "1.25 kg" or "16 OZ" into whole grams.

    Accepts an integer or a decimal (with "." as separator), optional
    whitespace and a unit: g, kg, lb or oz, case-insensitive. Leading and
    trailing whitespace is ignored. Raises ValueError on anything else,
    including a missing number or unit, unknown units, negative values and
    "," as decimal separator. Rounding is half-up, matching the rest of the
    codebase.
    """
    if not isinstance(text, str):
        raise ValueError("weight must be a string")
    match = _WEIGHT_RE.fullmatch(text.strip())
    if match is None:
        raise ValueError(f"unparseable weight: {text!r}")
    grams = Decimal(match.group(1)) * _GRAMS_PER_UNIT[match.group(2).lower()]
    return int(grams.quantize(Decimal("1"), rounding=ROUND_HALF_UP))
```
