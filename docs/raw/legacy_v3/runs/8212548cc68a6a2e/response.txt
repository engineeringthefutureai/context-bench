import re

_WEIGHT_PATTERN = re.compile(r"\s*([0-9]*\.?[0-9]+)\s*([a-zA-Z]+)\s*")

_WEIGHT_FACTORS: dict[str, tuple[int, int]] = {
    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (28349523125, 1000000000),
}


def parse_weight(text: str) -> int:
    """Parse a staff-entered weight into whole grams.

    Accepts a non-negative integer or decimal, optional whitespace, and one of
    g, kg, lb, or oz (case-insensitive). The result is rounded to the nearest
    gram. Raises ValueError for any other input.
    """
    match = _WEIGHT_PATTERN.fullmatch(text)
    if match is None:
        raise ValueError("invalid weight")

    number, unit = match.groups()
    unit = unit.lower()
    factor = _WEIGHT_FACTORS.get(unit)
    if factor is None:
        raise ValueError("unknown weight unit")

    if "." in number:
        whole, frac = number.split(".")
        frac = frac.rstrip("0")
        if frac:
            numerator = int((whole or "0") + frac)
            denominator = 10 ** len(frac)
        else:
            numerator = int(whole or "0")
            denominator = 1
    else:
        numerator = int(number)
        denominator = 1

    numerator *= factor[0]
    denominator *= factor[1]
    return (2 * numerator + denominator) // (2 * denominator)
