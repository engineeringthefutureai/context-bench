import re
from dataclasses import dataclass

_SLUG_RE = re.compile(r"[^a-z0-9]+")


def _slugify(title: str) -> str:
    """Return the URL slug for a product title.

    Lowercases the title and replaces every run of non-alphanumeric
    characters with a single hyphen, dropping hyphens at either end.
    """
    slug = _SLUG_RE.sub("-", title.strip().lower())
    return slug.strip("-")


@dataclass
class Product:
    sku: str
    title: str
    price_cents: int
    active: bool = True

    def is_purchasable(self) -> bool:
        return self.active and self.price_cents > 0

    @property
    def url_path(self) -> str:
        return f"/products/{_slugify(self.title)}"
