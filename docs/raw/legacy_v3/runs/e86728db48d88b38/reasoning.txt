import re

_DURATION = re.compile(r"(?P<hours>\d+h)?\s*(?P<minutes>\d+m)?", re.IGNORECASE)

def parse_duration(text: str) -> int:
    """Parse hours/minutes like "2h", "45m", "1h30m" or "1h 30m" into whole minutes.

    Units are case-insensitive and at most once each; hours precede minutes.
    Raises ValueError for anything else.
    """
    m = _DURATION.fullmatch(text.strip())
    if m is None or not (m["hours"] or m["minutes"]):
        raise ValueError(f"bad duration {text!r}")
    hours = int(m["hours"][:-1]) if m["hours"] else 0
    minutes = int(m["minutes"][:-1]) if m["minutes"] else 0
    return hours * 60 + minutes
