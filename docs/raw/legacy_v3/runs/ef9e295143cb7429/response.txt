import re


def parse_duration(text: str) -> int:
    """Parse a human duration of the form "2h", "45m" or "1h30m" into whole minutes."""
    tokens = text.strip().lower().split()
    hours = 0
    minutes = 0
    seen_hours = False
    for token in tokens:
        match = re.fullmatch(r"(\d+)([hm])", token)
        if match is None:
            raise ValueError(f"invalid duration: {text!r}")
        value = int(match.group(1))
        if match.group(2) == "h":
            if seen_hours:
                raise ValueError(f"invalid duration: {text!r}")
            seen_hours = True
            hours = value
        else:
            if minutes:
                raise ValueError(f"invalid duration: {text!r}")
            minutes = value
    if not tokens:
        raise ValueError(f"invalid duration: {text!r}")
    return hours * 60 + minutes
