from shopkit.errors import Conflict


class Stock:
    """On-hand quantities per SKU."""

    def __init__(self):
        self._on_hand: dict[str, int] = {}

    def receive(self, sku: str, qty: int) -> None:
        self._on_hand[sku] = self._on_hand.get(sku, 0) + qty

    def available(self, sku: str) -> int:
        return self._on_hand.get(sku, 0)

    def reserve(self, sku: str, qty: int) -> None:
        """Lowers the available quantity of `sku` by `qty` (holding stock for an order).
        Raises Conflict with the requested and available quantities if there is not enough."""
        on_hand = self.available(sku)
        if on_hand < qty:
            raise Conflict(
                f"not enough {sku}: requested {qty}, available {on_hand}",
                sku=sku,
                requested=qty,
                available=on_hand,
            )
        self._on_hand[sku] = on_hand - qty
