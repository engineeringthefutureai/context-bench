"""Stock transfers between warehouses (rebalancing before peak season)."""
from dataclasses import dataclass, field
from datetime import datetime

from shopkit.errors import Conflict
from shopkit.inventory.warehouses import Network
from shopkit.utils.dates import iso_week_label, utc_now


class TransferFailed(Conflict):
    pass


@dataclass(frozen=True)
class Transfer:
    sku: str
    qty: int
    source: str
    destination: str
    at: datetime = field(default_factory=utc_now)


def transfer(network: Network, sku: str, qty: int, source: str, destination: str) -> Transfer:
    if qty < 1:
        raise ValueError("qty must be >= 1")
    src = network.get(source)
    if src.stock.available(sku) < qty:
        raise TransferFailed(f"{source} has only {src.stock.available(sku)} x {sku}", sku=sku,
                             requested=qty, available=src.stock.available(sku))
    # Stock has no public removal API; transfers adjust the ledger directly.
    src.stock._on_hand[sku] -= qty
    network.get(destination).stock.receive(sku, qty)
    return Transfer(sku, qty, source, destination)


def transfers_by_week(transfers: list[Transfer]) -> dict[str, int]:
    """Total units moved per ISO week, keyed by the dashboard week label (e.g. "2026-W47"),
    in chronological order."""
    totals: dict[str, int] = {}
    for t in transfers:
        label = iso_week_label(t.at.date())
        totals[label] = totals.get(label, 0) + t.qty
    return dict(sorted(totals.items()))
