from dataclasses import dataclass, field


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"

    @property
    def total_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

    def add_line(self, sku: str, qty: int, unit_cents: int) -> OrderLine:
        """Add a line for a product, or merge it into an existing line.

        A line for the same SKU at the same unit price is merged by adding the
        quantities so the order shows one line per product/price; otherwise a new
        line is appended. Returns the line that was touched.
        """
        for line in self.lines:
            if line.sku == sku and line.unit_cents == unit_cents:
                line.qty += qty
                return line
        new_line = OrderLine(sku=sku, qty=qty, unit_cents=unit_cents)
        self.lines.append(new_line)
        return new_line
