from dataclasses import dataclass


@dataclass
class Product:
    sku: str
    title: str
    price_cents: int
    active: bool = True
    weight_g: int = 0

    def is_purchasable(self) -> bool:
        return self.active and self.price_cents > 0

    def to_dict(self) -> dict:
        """Serialise the product for the API response body."""
        return {
            "sku": self.sku,
            "title": self.title,
            "price_cents": self.price_cents,
            "active": self.active,
            "weight_g": self.weight_g,
        }

def packing_slip(parcel: Parcel, weight_g: int) -> str:
    """Slip packed inside each parcel: order, parcel number, contents and weight."""
    lines = [f"Order {parcel.order_id} - parcel {parcel.number}", f"Weight: {format_weight(weight_g)}"]
    lines += [f"  {qty} x {sku}" for sku, qty in parcel.items]
    return "\n".join(lines)


def parcel_weight_g(parcel, products: dict[str, Product]) -> int:
    """Total weight in grams of a parcel's contents.

    Each item's weight is its unit weight (grams) from the product catalogue
    (``products`` maps SKU to :class:`Product`) times the item quantity. A SKU
    with no product entry is treated as zero weight.
    """
    return sum(
        products[sku].weight_g * qty
        for sku, qty in parcel.items
        if sku in products
    )
