import re

_DURATION_RE = re.compile(
    r"\s*(?:(\d+)h)?\s*(?:(\d+)m)?\s*$",
    re.IGNORECASE,
)


def parse_duration(text: str) -> int:
    """Parse a human duration like "2h", "45m", "1h 30m" into whole minutes.

    Hours come before minutes, each unit at most once, units are
    case-insensitive and surrounding whitespace is ignored. Anything else
    (empty text, other units, a missing number, minutes before hours,
    negative numbers) raises ValueError."""
    m = _DURATION_RE.match(text)
    if not m:
        raise ValueError(f"invalid duration: {text!r}")
    hours_s, minutes_s = m.groups()
    if hours_s is None and minutes_s is None:
        raise ValueError(f"invalid duration: {text!r}")
    hours = int(hours_s) if hours_s is not None else 0
    minutes = int(minutes_s) if minutes_s is not None else 0
    return hours * 60 + minutes
