from shopkit.errors import Conflict


class ReservationConflict(Conflict):
    pass


class Stock:
    """On-hand quantities per SKU."""

    def __init__(self):
        self._on_hand: dict[str, int] = {}

    def receive(self, sku: str, qty: int) -> None:
        self._on_hand[sku] = self._on_hand.get(sku, 0) + qty

    def available(self, sku: str) -> int:
        return self._on_hand.get(sku, 0)

    def reserve(self, sku: str, qty: int) -> None:
        """Hold `qty` units of `sku` for an order.

        Raises ReservationConflict (409) with `requested` and `available` details if there is
        not enough
        on hand, leaving stock unchanged."""
        have = self.available(sku)
        if have < qty:
            raise ReservationConflict(f"not enough {sku}", sku=sku, requested=qty, available=have)
        self._on_hand[sku] = have - qty
