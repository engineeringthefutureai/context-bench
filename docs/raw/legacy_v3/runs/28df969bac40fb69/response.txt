from dataclasses import dataclass

from shopkit.utils.text import slug_for_title


@dataclass
class Product:
    sku: str
    title: str
    price_cents: int
    active: bool = True

    def is_purchasable(self) -> bool:
        return self.active and self.price_cents > 0

    @property
    def url_path(self) -> str:
        return "/products/{}".format(slug_for_title(self.title))
