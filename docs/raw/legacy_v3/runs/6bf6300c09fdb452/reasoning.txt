"""Parsing and unit conversion helpers."""
import re

_WEIGHT_PATTERN = re.compile(
    r"\A\s*(?P<number>\d+(?:\.\d+)?|\.\d+)\s*(?P<unit>g|kg|lb|oz)\s*\Z",
    re.IGNORECASE,
)

_CONVERSIONS = {
    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (28349523125, 1000000000),
}

def parse_weight(text: str) -> int:
    """Parse a package weight such as '1.5 kg' and return whole grams.

    Accepts an integer or decimal number, optional whitespace, and one of
    the units g, kg, lb, or oz (case-insensitive). Raises ValueError for
    malformed input, unknown units, negative values, or comma decimals.
    """
    if not isinstance(text, str):
        raise ValueError("weight must be a string")
    match = _WEIGHT_PATTERN.match(text)
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")
    number = match.group("number")
    unit = match.group("unit").lower()
    unit_numerator, unit_denominator = _CONVERSIONS[unit]
    if "." in number:
        whole, fraction = number.split(".")
        if not fraction:
            raise ValueError(f"invalid weight: {text!r}") # not possible
        scale = 10 ** len(fraction)
        value = (int(whole) if whole else 0) * scale + int(fraction)
    else:
        scale = 1
        value = int(number)
    numerator = value * unit_numerator
    denominator = scale * unit_denominator
    return (2 * numerator + denominator) // (2 * denominator)

"""Parsing helpers for physical units."""

import re

_WEIGHT_RE = re.compile(
    r"\A\s*(?P<number>(?:[0-9]+(?:\.[0-9]+)?|\.[0-9]+))\s*(?P<unit>g|kg|lb|oz)\s*\Z",
    re.IGNORECASE,
)

_CONVERSIONS = {
    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (28349523125, 1000000000),
}

def parse_weight(text: str) -> int:
    """Parse a package weight and return the nearest whole gram.

    Accepted forms are an integer or a decimal number using '.' as the
    separator, optional whitespace, then one of g, kg, lb, or oz
    (case-insensitive). Ties are rounded up. Any other input, including
    negative values and ',' decimal separators, raises ValueError.
    """
    if not isinstance(text, str):
        raise ValueError("weight must be a string")

    match = _WEIGHT_RE.match(text)
    if match is None:
        raise ValueError("invalid weight")

    number = match.group("number")
    unit = match.group("unit").lower()
    unit_numerator, unit_denominator = _CONVERSIONS[unit]

    if "." in number:
        whole, fraction = number.split(".")
        fraction = fraction.rstrip("0")
        if fraction:
            scale = 10 ** len(fraction)
            value = (int(whole) if whole else 0) * scale + int(fraction)
        else:
            value = int(whole) if whole else 0
            scale = 1
    else:
        value = int(number)
        scale = 1

    numerator = value * unit_numerator
    denominator = scale * unit_denominator
    return (2 * numerator + denominator) // (2 * denominator)

    r"\A\s*(?P<number>(?:[0-9]+(?:\.[0-9]+)?|\.[0-9]+))\s*(?P<unit>g|kg|lb|oz)\s*\Z",

    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (28349523125, 1000000000),

def parse_weight(text: str) -> int:
    """Parse a package weight such as '1.5 kg' and return whole grams.

    The number may be an integer or a decimal using '.' as separator. The
    unit is one of g, kg, lb, or oz, case-insensitive. Ties are rounded up.
    Raises ValueError for malformed input, unknown units, negative values,
    or ',' decimal separators.
    """
    if not isinstance(text, str):
        raise ValueError("weight must be a string")

    match = _WEIGHT_RE.match(text)
    if match is None:
        raise ValueError("invalid weight")

    number = match.group("number")
    unit = match.group("unit").lower()
    unit_numerator, unit_denominator = _CONVERSIONS[unit]

    if "." in number:
        whole, fraction = number.split(".")
        fraction = fraction.rstrip("0")
        if fraction:
            scale = 10 ** len(fraction)
            value = (int(whole) if whole else 0) * scale + int(fraction)
        else:
            value = int(whole) if whole else 0
            scale = 1
    else:
        value = int(number)
        scale = 1

    numerator = value * unit_numerator
    denominator = scale * unit_denominator
    return (2 * numerator + denominator) // (2 * denominator)

_CONVERSIONS = {
    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (28349523125, 1000000000),
}


def parse_weight(text: str) -> int:
    """Parse a package weight such as '1.5 kg' and return whole grams.

    The number may be an integer or a decimal using '.' as separator. The
    unit is one of g, kg, lb, or oz, case-insensitive. Ties are rounded up.
    Raises ValueError for malformed input, unknown units, negative values,
    or ',' decimal separators.
    """
    if not isinstance(text, str):
        raise ValueError("weight must be a string")

    s = text.strip()
    if not s:
        raise ValueError("invalid weight")

    i = 0
    n = len(s)
    number_start = i

    while i < n and "0" <= s[i] <= "9":
        i += 1

    if i < n and s[i] == ".":
        i += 1
        fraction_start = i
        while i < n and "0" <= s[i] <= "9":
            i += 1
        if i == fraction_start:
            raise ValueError("invalid weight")

    number = s[number_start:i]
    if not number:
        raise ValueError("invalid weight")

    while i < n and s[i].isspace():
        i += 1

    unit = s[i:].lower()
    if unit not in _CONVERSIONS:
        raise ValueError("invalid weight")

    unit_numerator, unit_denominator = _CONVERSIONS[unit]

    if "." in number:
        whole, fraction = number.split(".")
        fraction = fraction.rstrip("0")
        if fraction:
            scale = 10 ** len(fraction)
            value = (int(whole) if whole else 0) * scale + int(fraction)
        else:
            value = int(whole) if whole else 0
            scale = 1
    else:
        value = int(number)
        scale = 1

    numerator = value * unit_numerator
    denominator = scale * unit_denominator
    return (2 * numerator + denominator) // (2 * denominator)
