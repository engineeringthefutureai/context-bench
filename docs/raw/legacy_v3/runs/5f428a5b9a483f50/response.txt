import re

_DURATION_RE = re.compile(r"\d+\s*[hm]\s*(?:\d+\s*[hm])?\Z", re.IGNORECASE)


def parse_duration(text: str) -> int:
    """Total minutes from a text of hours and/or minutes, e.g. "2h", "45m" or "1h 30m".

    Hours come before minutes, each unit appears at most once, unit letters are
    case-insensitive and surrounding whitespace is ignored. Raises ValueError for
    anything else: empty text, unknown units, a missing number, minutes before
    hours or negative numbers.
    """
    stripped = text.strip()
    if not _DURATION_RE.fullmatch(stripped):
        raise ValueError(f"invalid duration: {text!r}")
    parts = re.findall(r"(\d+)\s*([hm])", stripped, re.IGNORECASE)
    minutes = 0
    seen_hours = False
    for value, unit in parts:
        amount = int(value)
        if unit.upper() == "H":
            if seen_hours:
                raise ValueError(f"invalid duration: {text!r}")
            seen_hours = True
            minutes += amount * 60
        else:
            if seen_hours:
                raise ValueError(f"invalid duration: {text!r}")
            if minutes:
                raise ValueError(f"invalid duration: {text!r}")
            minutes += amount
    return minutes
