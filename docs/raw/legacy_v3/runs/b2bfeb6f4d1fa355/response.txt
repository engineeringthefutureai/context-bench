from shopkit.catalog.models import Product


def parcel_weight_g(parcel: dict, products: dict[str, Product]) -> int:
    """
    Calculate the total weight in grams of a parcel's contents.
    
    Args:
        parcel: A dictionary mapping SKU to quantity.
        products: A dictionary mapping SKU to Product.
        
    Returns:
        Total weight in grams.
    """
    total_weight = 0
    for sku, quantity in parcel.items():
        product = products.get(sku)
        if product is not None:
            total_weight += product.weight_g * quantity
    return total_weight
