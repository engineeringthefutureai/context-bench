"""Shipping label content handed to the label printer service."""
from typing import Optional

from shopkit.customers.addresses import format_address
from shopkit.customers.models import Address
from shopkit.orders.fulfillment import Parcel
from shopkit.shipping.packaging import format_weight

_LB_PER_GRAM = 0.00220462262185
_US_COUNTRY_CODES = {"US", "USA", "UNITED STATES", "UNITED STATES OF AMERICA"}


def label_code(sku: str) -> str:
    """Compact code printed under the barcode: uppercase, spaces and dashes removed (scanner-friendly)."""
    return sku.replace(" ", "").replace("-", "").upper()


def _is_us_address(address: Address) -> bool:
    """Whether the label destination is a United States address."""
    country = getattr(address, "country_code", None) or getattr(address, "country", None) or ""
    return str(country).strip().upper() in _US_COUNTRY_CODES


def _weight_display(weight_g: int, address: Address) -> str:
    """Display weight for a label: pounds for US destinations, standard kilograms elsewhere."""
    if _is_us_address(address):
        return f"{weight_g * _LB_PER_GRAM:.2f} lb"
    return format_weight(weight_g)


def label_text(parcel: Parcel, to: Address, parcels_total: int, weight_g: Optional[int] = None) -> str:
    lines = [format_address(to), "", f"Order {parcel.order_id}  parcel {parcel.number}/{parcels_total}"]
    if weight_g is not None:
        lines.append(f"Weight: {_weight_display(weight_g, to)}")
    lines += [f"{label_code(sku)} x{qty}" for sku, qty in parcel.items]
    return "\n".join(lines)
