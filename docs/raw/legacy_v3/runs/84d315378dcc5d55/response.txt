def load_settings(env: dict[str, str] | None = None) -> Settings:
    """Build Settings from SHOPKIT_* variables; unknown variables are ignored."""
    env = os.environ if env is None else env
    kwargs: dict[str, object] = {}
    if "SHOPKIT_CURRENCY" in env:
        kwargs["default_currency"] = env["SHOPKIT_CURRENCY"]
    if "SHOPKIT_COUNTRY" in env:
        kwargs["store_country"] = env["SHOPKIT_COUNTRY"]
    if "SHOPKIT_LOW_STOCK" in env:
        kwargs["low_stock_threshold"] = int(env["SHOPKIT_LOW_STOCK"])
    if "SHOPKIT_PAGE_SIZE" in env:
        kwargs["page_size"] = int(env["SHOPKIT_PAGE_SIZE"])
    if "SHOPKIT_RETURN_DAYS" in env:
        kwargs["return_window_days"] = int(env["SHOPKIT_RETURN_DAYS"])
    flags = env.get("SHOPKIT_FLAGS", "")
    kwargs["feature_flags"] = {f.strip(): True for f in flags.split(",") if f.strip()}
    return Settings(**kwargs)

SETTINGS = load_settings()

"""Customer returns (RMA). A return refunds lines of a delivered order within the return window."""
from dataclasses import dataclass, field
from datetime import date, timedelta

from shopkit.config import SETTINGS
from shopkit.errors import Conflict, ShopError
from shopkit.orders.models import Order
from shopkit.utils.ids import short_code

REASONS = ("damaged", "wrong_item", "not_as_described", "changed_mind")


class ReturnNotAllowed(Conflict):
    pass


class InvalidReturn(ShopError):
    http_status = 422


@dataclass
class ReturnRequest:
    order_id: str
    items: dict[str, int]            # sku -> qty being returned
    reason: str
    code: str = field(default_factory=lambda: short_code(8, groups=4))
    approved: bool = False

    def refund_cents(self, order: Order) -> int:
        prices = {line.sku: line.unit_cents for line in order.lines}
        return sum(prices[sku] * qty for sku, qty in self.items.items())


def open_return(order: Order, delivered_on: date, items: dict[str, int], reason: str,
                today: date) -> ReturnRequest:
    if reason not in REASONS:
        raise InvalidReturn(f"unknown reason {reason!r}", reason=reason)
    window_days = SETTINGS.return_window_days
    if today > delivered_on + timedelta(days=window_days):
        raise ReturnNotAllowed(f"order {order.id} is outside the return window", order_id=order.id,
                               delivered_on=delivered_on.isoformat())
    ordered = {line.sku: line.qty for line in order.lines}
    for sku, qty in items.items():
        if qty < 1 or qty > ordered.get(sku, 0):
            raise InvalidReturn(f"cannot return {qty} x {sku}", sku=sku, requested=qty,
                                ordered=ordered.get(sku, 0))
    # changed-mind returns need manual approval; faults are approved automatically
    return ReturnRequest(order.id, dict(items), reason, approved=reason != "changed_mind")
