from dataclasses import dataclass, field
from shopkit.errors import ShopError


class _InvalidDiscountError(ShopError, ValueError):
    pass


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"
    discount_percent: int = 0

    def __post_init__(self) -> None:
        if self.discount_percent < 0 or self.discount_percent > 100:
            raise _InvalidDiscountError("discount_percent must be between 0 and 100")

    @property
    def subtotal_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

    @property
    def discount_cents(self) -> int:
        return self.subtotal_cents * self.discount_percent // 100

    @property
    def total_cents(self) -> int:
        return self.subtotal_cents - self.discount_cents

from shopkit.money import format_money
from shopkit.orders.models import Order


def render_invoice_text(order: Order) -> str:
    lines = [f"Order {order.id}"]
    for line in order.lines:
        lines.append(
            f"{line.qty} x {line.sku} @ "
            f"{format_money(line.unit_cents, order.currency)} = "
            f"{format_money(line.total_cents, order.currency)}"
        )
    if order.discount_percent > 0:
        lines.append(
            f"Discount: {format_money(order.discount_cents, order.currency)}"
        )
    lines.append(f"Total: {format_money(order.total_cents, order.currency)}")
    return "\n".join(lines)

from dataclasses import asdict

from shopkit.orders.models import Order


def export_orders_json(orders: list[Order]) -> str:
    return json.dumps([asdict(order) for order in orders])
