import re


def parse_duration(text: str) -> int:
    """Parse a duration of hours and/or minutes into whole minutes.

    Input is a string like "2h", "45m", "1h30m", or "1h 30m" where hours come before
    minutes, each unit appears at most once, units are case-insensitive, and surrounding
    whitespace is ignored. Raises ValueError for anything else (empty text, unknown
    units, missing numbers, minutes before hours, negative numbers).
    """
    s = text.strip()
    if not s:
        raise ValueError("duration is empty")

    match = re.fullmatch(r"(?P<hours>\d+)(?P<h>\s*h(?:ours?|rs?)?)\s*(?P<minutes>\d+)(?P<m>\s*m(?:in(?:ute)?s?|s?)?)", s, re.IGNORECASE)
    if match:
        return int(match.group("hours")) * 60 + int(match.group("minutes"))

    match = re.fullmatch(r"(?P<hours>\d+)\s*h(?:ours?|rs?)?", s, re.IGNORECASE)
    if match:
        return int(match.group("hours")) * 60

    match = re.fullmatch(r"(?P<minutes>\d+)\s*m(?:in(?:ute)?s?|s?)?", s, re.IGNORECASE)
    if match:
        return int(match.group("minutes"))

    raise ValueError(f"invalid duration: {text!r}")

import re


def parse_duration(text: str) -> int:
    """Parse a duration of hours and/or minutes into whole minutes.

    Input is a string like "2h", "45m", "1h30m", or "1h 30m" where hours come before
    minutes, each unit appears at most once, units are case-insensitive, and surrounding
    whitespace is ignored. Raises ValueError for anything else (empty text, unknown
    units, missing numbers, minutes before hours, negative numbers).
    """
    s = text.strip()
    if not s:
        raise ValueError("duration is empty")

    # hours and minutes
    m = re.fullmatch(
        r"(?P<hours>\d+)\s*h(?:ours?|rs?)?\s+(?P<minutes>\d+)\s*m(?:in(?:ute)?s?|s?)?",
        s,
        re.IGNORECASE,
    )
    if m:
        return int(m.group("hours")) * 60 + int(m.group("minutes"))

    # hours and minutes without whitespace between them
    m = re.fullmatch(
        r"(?P<hours>\d+)\s*h(?:ours?|rs?)?\s*(?P<minutes>\d+)\s*m(?:in(?:ute)?s?|s?)?",
        s,
        re.IGNORECASE,
    )
    if m:
        return int(m.group("hours")) * 60 + int(m.group("minutes"))

    # hours only
    m = re.fullmatch(r"(?P<hours>\d+)\s*h(?:ours?|rs?)?", s, re.IGNORECASE)
    if m:
        return int(m.group("hours")) * 60

    # minutes only
    m = re.fullmatch(r"(?P<minutes>\d+)\s*m(?:in(?:ute)?s?|s?)?", s, re.IGNORECASE)
    if m:
        return int(m.group("minutes"))

    raise ValueError(f"invalid duration: {text!r}")
