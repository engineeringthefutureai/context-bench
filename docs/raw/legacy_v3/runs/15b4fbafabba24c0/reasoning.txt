```python
"""Parsing of durations expressed in hours and/or minutes."""

import re

__all__ = ["parse_duration"]

# Hours come before minutes and each unit may appear at most once.  The
# whitespace runs accept "1h 30m" as well as "1h30m" and untrimmed input;
# re.ASCII keeps \d and \s to plain ASCII so look-alike digits are rejected.
_DURATION_RE = re.compile(
    r"\s*(?:(\d+)h)?\s*(?:(\d+)m)?\s*",
    re.IGNORECASE | re.ASCII,
)


def parse_duration(text: str) -> int:
    """Return the number of whole minutes in a duration string.

    Accepted inputs are hours and/or minutes with hours first, e.g. ``"2h"``,
    ``"45m"``, ``"1h30m"`` or ``"1h 30m"``.  Units are case-insensitive and
    surrounding whitespace is ignored.

    Raises:
        ValueError: if the text is empty, uses any unit other than hours or
            minutes, is missing a number, repeats a unit, places minutes
            before hours, or contains a negative number.
    """
    match = _DURATION_RE.fullmatch(text)
    if match is None or (match.group(1) is None and match.group(2) is None):
        raise ValueError(f"invalid duration: {text!r}")

    hours = int(match.group(1)) if match.group(1) is not None else 0
    minutes = int(match.group(2)) if match.group(2) is not None else 0
    return hours * 60 + minutes
```
