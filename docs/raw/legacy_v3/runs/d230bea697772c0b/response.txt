```python
import re


def parse_weight(text: str) -> int:
    """Parse a weight string like '1.5 kg' into whole grams.

    Accepted units: g, kg, lb, oz (case-insensitive).
    1 lb = 453.59237 g, 1 oz = 28.349523125 g.
    Raises ValueError for malformed input.
    """
    s = text.strip()
    m = re.fullmatch(r'([+-]?\d*\.?\d+)\s*([a-zA-Z]+)', s)
    if not m:
        raise ValueError(f"Invalid weight string: {text!r}")

    number_str, unit = m.group(1), m.group(2).lower()

    # Reject commas as decimal separators
    if ',' in number_str:
        raise ValueError(f"Comma not allowed as decimal separator: {text!r}")

    try:
        value = float(number_str)
    except ValueError:
        raise ValueError(f"Invalid number: {text!r}")

    if value < 0:
        raise ValueError(f"Negative weight not allowed: {text!r}")

    if unit == 'g':
        grams = value
    elif unit == 'kg':
        grams = value * 1000
    elif unit == 'lb':
        grams = value * 453.59237
    elif unit == 'oz':
        grams = value * 28.349523125
    else:
        raise ValueError(f"Unknown unit: {unit!r}")

    return round(grams)
```
