```python
def to_csv(header: Sequence[str], rows: Iterable[Sequence[object]]) -> str:
    buf = io.StringIO()
    writer = csv.writer(buf, lineterminator="\n")
    writer.writerow(header)
    writer.writerows(rows)
    return buf.getvalue()
```

This is perfect - it uses the standard `csv` module, always `\n` line endings, header first. I'll use this helper.

import json

from shopkit.money import format_money
from shopkit.orders.models import Order
from shopkit.utils.collections import dedupe
from shopkit.utils.csvio import to_csv
from shopkit.utils.text import normalize_sku


def export_orders_json(orders: list[Order]) -> str:
    """Raw JSON export for the warehouse integration."""
    payload = [
        {
            "id": o.id,
            "currency": o.currency,
            "lines": [{"sku": l.sku, "qty": l.qty, "unit_cents": l.unit_cents} for l in o.lines],
        }
        for o in orders
    ]
    return json.dumps(payload, indent=2)


def export_orders_csv(orders: list[Order]) -> str:
    """CSV export for finance: one row per order line with product code and line total."""
    header = ["order_id", "product_code", "quantity", "line_total"]
    rows = [
        [o.id, normalize_sku(l.sku), l.qty, format_money(l.total_cents, o.currency)]
        for o in orders
        for l in o.lines
    ]
    return to_csv(header, rows)
