"""Parsing of compact human-readable durations into whole minutes."""
import re

_DURATION_RE = re.compile(r"^\s*(?:(\d+)\s*h)?\s*(?:(\d+)\s*m)?\s*$", re.IGNORECASE)


def parse_duration(text: str) -> int:
    """Parse a duration of hours and/or minutes and return the total in whole minutes.

    Accepted forms: "2h", "45m", "1h30m", "1h 30m" (whitespace optional, units
    case-insensitive, hours must come before minutes, each unit at most once).
    Raises ValueError for empty text, missing numbers, unknown units, minutes
    before hours, or negative numbers.
    """
    if not isinstance(text, str):
        raise ValueError(f"expected str, got {type(text).__name__}")
    match = _DURATION_RE.match(text)
    if match is None:
        raise ValueError(f"invalid duration: {text!r}")
    hours_part, minutes_part = match.groups()
    if hours_part is None and minutes_part is None:
        raise ValueError(f"invalid duration: {text!r}")
    hours = int(hours_part) if hours_part is not None else 0
    minutes = int(minutes_part) if minutes_part is not None else 0
    return hours * 60 + minutes
