from fractions import Fraction
import re

_WEIGHT_PATTERN = re.compile(
    r"^\s*(\d+(?:\.\d+)?)\s*(g|kg|lb|oz)\s*$",
    re.IGNORECASE,
)

_UNIT_TO_GRAMS = {
    "g": Fraction(1, 1),
    "kg": Fraction(1000, 1),
    "lb": Fraction(45359237, 100000),
    "oz": Fraction(28349523125, 1000000000),
}

def _parse_decimal(value: str) -> Fraction:
    if "." in value:
        whole, frac = value.split(".", 1)
        denominator = 10 ** len(frac)
        numerator = int(whole + frac)
    else:
        numerator = int(value)
        denominator = 1
    return Fraction(numerator, denominator)

def parse_weight(text: str) -> int:
    match = _WEIGHT_PATTERN.match(text)
    if match is None:
        raise ValueError("Invalid weight")
    number_text, unit = match.groups()
    weight = _parse_decimal(number_text) * _UNIT_TO_GRAMS[unit.lower()]
    numerator, denominator = weight.numerator, weight.denominator
    quotient, remainder = divmod(numerator, denominator)
    if remainder * 2 >= denominator:
        quotient += 1
    return quotient

import re

_WEIGHT_RE = re.compile(
    r"^\s*(\d+(?:\.\d+)?)\s*(g|kg|lb|oz)\s*$",
    re.IGNORECASE,
)

_UNIT_FACTORS = {
    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (28349523125, 1000000000),
}

def _number_to_numerator_and_denominator(number: str) -> tuple[int, int]:
    if "." in number:
        whole, fraction = number.split(".", 1)
        denominator = 10 ** len(fraction)
        numerator = int(whole + fraction)
    else:
        numerator = int(number)
        denominator = 1
    return numerator, denominator

def parse_weight(text: str) -> int:
    """Parse a weight string and return its weight in whole grams."""
    match = _WEIGHT_RE.match(text)
    if match is None:
        raise ValueError("Invalid weight format")
    number_text, unit_text = match.groups()
    number_numerator, number_denominator = _number_to_numerator_and_denominator(number_text)
    unit_numerator, unit_denominator = _UNIT_FACTORS[unit_text.lower()]
    numerator = number_numerator * unit_numerator
    denominator = number_denominator * unit_denominator
    quotient, remainder = divmod(numerator, denominator)
    if remainder * 2 >= denominator:
        quotient += 1
    return quotient

    r"^\s*(\d+(?:\.\d+)?)\s*(g|kg|lb|oz)\s*$",
    re.IGNORECASE | re.ASCII,
)

    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (28349523125, 1000000000),
}


def _decimal_to_numerator_and_denominator(value: str):
    if "." in value:
        whole, fraction = value.split(".", 1)
        denominator = 10 ** len(fraction)
        numerator = int(whole + fraction)
    else:
        numerator = int(value)
        denominator = 1
    return numerator, denominator

    """Parse a weight string and return its weight in whole grams."""
    if not isinstance(text, str):
        raise ValueError("Weight must be a string")
