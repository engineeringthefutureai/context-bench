from shopkit.errors import Conflict


class InsufficientStock(Conflict):
    """Raised when a reservation cannot be made because of insufficient stock."""


class Stock:
    """On-hand quantities per SKU."""

    def __init__(self):
        self._on_hand: dict[str, int] = {}

    def receive(self, sku: str, qty: int) -> None:
        self._on_hand[sku] = self._on_hand.get(sku, 0) + qty

    def available(self, sku: str) -> int:
        return self._on_hand.get(sku, 0)

    def reserve(self, sku: str, qty: int) -> None:
        """Reserve `qty` units of `sku`, lowering available stock by `qty`."""
        if qty < 1:
            raise ValueError("qty must be >= 1")
        available = self._on_hand.get(sku, 0)
        if available < qty:
            raise InsufficientStock(
                f"only {available} x {sku} available",
                requested=qty,
                available=available,
            )
        self._on_hand[sku] = available - qty
