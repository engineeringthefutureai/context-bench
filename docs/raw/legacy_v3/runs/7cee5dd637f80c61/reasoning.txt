def transfers_by_week(transfers: Iterable[Transfer]) -> dict[str, int]:
    totals: dict[str, int] = {}
    order: dict[str, tuple[int, int]] = {}
    for t in transfers:

        label = iso_week_label(t.at.date())

    return {label: totals[label] for label, _ in sorted(order.items(), key=lambda kv: kv[1])}
