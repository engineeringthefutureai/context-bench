from dataclasses import dataclass, field


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"
    discount_percent: int = 0

    def __post_init__(self) -> None:
        if not 0 <= self.discount_percent <= 100:
            raise ValueError(
                f"discount_percent must be between 0 and 100, got {self.discount_percent}"
            )

    @property
    def subtotal_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

    @property
    def discount_cents(self) -> int:
        # Discount amount in whole cents, rounded down.
        return (self.subtotal_cents * self.discount_percent) // 100

    @property
    def total_cents(self) -> int:
        return self.subtotal_cents - self.discount_cents

import json

from shopkit.orders.models import Order


def export_orders_json(orders: list[Order]) -> str:
    """Raw JSON export for the warehouse integration."""
    payload = [
        {
            "id": o.id,
            "currency": o.currency,
            "discount_percent": o.discount_percent,
            "lines": [{"sku": l.sku, "qty": l.qty, "unit_cents": l.unit_cents} for l in o.lines],
        }
        for o in orders
    ]
    return json.dumps(payload, indent=2)

from shopkit.money import format_money

from shopkit.orders.models import Order


def render_invoice_text(order: Order) -> str:
    """Order block of the customer invoice: one line per order line, total last."""
    lines = [
        f"{line.sku} x {line.qty}: {format_money(line.total_cents, order.currency)}"
        for line in order.lines
    ]
    if order.discount_cents:
        lines.append(
            f"Discount ({order.discount_percent}%): "
            f"{format_money(-order.discount_cents, order.currency)}"
        )
    lines.append(f"Total: {format_money(order.total_cents, order.currency)}")
    return "\n".join(lines)
