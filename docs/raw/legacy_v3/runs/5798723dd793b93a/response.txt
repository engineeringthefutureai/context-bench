"""Refunds against captured payments. Partial refunds are allowed up to the captured amount."""
from dataclasses import dataclass, field
from datetime import date, datetime, timedelta

from shopkit.errors import Conflict
from shopkit.money import format_money
from shopkit.payments.models import Payment
from shopkit.utils import dates as _dates
from shopkit.utils.dates import utc_now


class RefundTooLarge(Conflict):
    pass


@dataclass(frozen=True)
class Refund:
    payment_id: str
    amount_cents: int
    reason: str
    at: datetime = field(default_factory=utc_now)


def refund(payment: Payment, amount_cents: int, reason: str) -> Refund:
    if amount_cents <= 0:
        raise ValueError("refund amount must be positive")
    if amount_cents > payment.refundable_cents:
        raise RefundTooLarge(
            f"refund {format_money(amount_cents, payment.currency)} exceeds refundable "
            f"{format_money(payment.refundable_cents, payment.currency)}",
            payment_id=payment.id, requested=amount_cents, refundable=payment.refundable_cents)
    payment.refunded_cents += amount_cents
    return Refund(payment.id, amount_cents, reason)


def _is_business_day(day: date) -> bool:
    """Return True when the shop's business-day calendar counts ``day``."""
    checker = getattr(_dates, "is_business_day", None)
    if checker is None:
        checker = getattr(_dates, "is_shop_open", None)
    if callable(checker):
        return bool(checker(day))

    if day.weekday() >= 5:
        return False

    closed_days = getattr(_dates, "SHOP_CLOSED_DAYS", None)
    if closed_days is None:
        closed_days = getattr(_dates, "CLOSED_DAYS", None)
    if closed_days is not None and day in closed_days:
        return False

    return True


def expected_arrival(refund: Refund) -> date:
    """Return the date the refund is expected to reach the customer's card.

    Banks take five business days, counted from the day the refund was issued.
    The issue day counts when it is a business day; weekends and days the shop
    is closed are not counted.
    """
    day = refund.at.date()
    business_days_left = 5

    while True:
        if _is_business_day(day):
            business_days_left -= 1
            if business_days_left == 0:
                return day
        day += timedelta(days=1)
