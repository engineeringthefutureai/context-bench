import re

_PATTERN = re.compile(r"(\d+(?:\.\d+)?)\s*(g|kg|lb|oz)", re.IGNORECASE)

def parse_weight(text: str) -> int:
    """Whole-gram weight from staff-typed text like "1.5 kg", "2 lb" or "8 oz".

    The unit may be g, kg, lb or oz (case-insensitive); whitespace around the value and
    between the number and unit is ignored. Rounds to the nearest gram.
    Raises ValueError for a missing number or unit, an unknown unit, a "," decimal
    separator or a negative value."""
    if not isinstance(text, str):
        raise ValueError(f"expected a weight string, got {type(text).__name__}")
    m = _PATTERN.fullmatch(text.strip())
    if m is None:
        raise ValueError(f"invalid weight {text!r}")
    value, unit = float(m.group(1)), m.group(2).lower()
    return round(value * _TO_GRAMS[unit])

# fullmatch over the stripped input rejects anything else, including "," decimals,
# a missing number or unit and an unknown unit.
_PATTERN = re.compile(r"(\d+(?:\.\d+)?)\s*(g|kg|lb|oz)", re.IGNORECASE)

_TO_GRAMS = {"g": 1.0, "kg": 1000.0, "lb": 453.59237, "oz": 28.349523125}

    The unit may be g, kg, lb or oz (case-insensitive); whitespace around the value and
    between the number and unit is ignored. The result is rounded to the nearest gram.
    Raises ValueError for a missing number or unit, an unknown unit, a "," as the decimal
    separator and negative values."""
    if not isinstance(text, str):
        raise ValueError(f"expected a weight string, got {type(text).__name__}")
    match = _PATTERN.fullmatch(text.strip())
