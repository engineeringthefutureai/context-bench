from dataclasses import dataclass, field, asdict
import json

@dataclass
class OrderLine: ...

@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"
    discount_percent: int = 0

    def __post_init__(self):
        if not 0 <= self.discount_percent <= 100:
            raise ValueError("discount_percent must be between 0 and 100")

    @property
    def subtotal_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

    @property
    def discount_cents(self) -> int:
        return self.subtotal_cents * self.discount_percent // 100

    @property
    def total_cents(self) -> int:
        return self.subtotal_cents - self.discount_cents

    def _money(self, cents: int) -> str:
        sign = "-" if cents < 0 else ""
        cents = abs(cents)
        return f"{sign}{cents // 100}.{cents % 100:02d} {self.currency}"

    def invoice_lines(self) -> list[str]:
        lines = [f"Order {self.id}"]
        for line in self.lines:
            lines.append(f"{line.sku}: {line.qty} x {self._money(line.unit_cents)} = {self._money(line.total_cents)}")
        if self.discount_percent and self.discount_cents:
            lines.append(f"Discount ({self.discount_percent}%): -{self._money(self.discount_cents)}")
        lines.append(f"Total: {self._money(self.total_cents)}")
        return lines

    def invoice(self) -> str:
        return "\n".join(self.invoice_lines())

    def to_dict(self) -> dict:
        return asdict(self)

    def warehouse_dict(self) -> dict:
        return self.to_dict()

    def warehouse_json(self) -> str:
        return json.dumps(self.warehouse_dict())

    def to_warehouse_json(self) -> str:
        return self.warehouse_json()
```

from dataclasses import dataclass, field, asdict
import json


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"
    discount_percent: int = 0

    def __post_init__(self) -> None:
        if not isinstance(self.discount_percent, int) or isinstance(self.discount_percent, bool):
            raise TypeError("discount_percent must be an int")
        if not 0 <= self.discount_percent <= 100:
            raise ValueError("discount_percent must be between 0 and 100")

    @property
    def subtotal_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

    @property
    def discount_cents(self) -> int:
        return self.subtotal_cents * self.discount_percent // 10? no 100

    @property
    def total_cents(self) -> int:
        return self.subtotal_cents - self.discount_cents

    def invoice_lines(self) -> list[str]:
        lines = [f"Order {self.id}"]
        for line in self.lines:
            lines.append(f"{line.sku} x {line.qty} @ {line.unit_cents} = {line.total_cents}")
        if self.discount_percent:
            lines.append(f"Discount: {self.discount_cents}")
        lines.append(f"Total: {self.total_cents}")
        return lines

    def invoice(self) -> str:
        return "\n".join(self.invoice_lines())

    def render_invoice(self) -> str:
        return self.invoice()

    def to_invoice(self) -> str:
        return self.invoice()

    def warehouse_payload(self) -> dict:
        return asdict(self)

    def to_dict(self) -> dict:
        return self.warehouse_payload()

    def as_dict(self) -> dict:
        return self.to_dict()

    def warehouse_json(self) -> str:
        return json.dumps(self.warehouse_payload())

    def to_warehouse_json(self) -> str:
        return self.warehouse_json()
```
