"""Newsletter sends. Only customers who opted in receive marketing email."""
import re

from shopkit.customers.models import Customer
from shopkit.marketing.campaigns import Campaign
from shopkit.notifications.templates import render_template
from shopkit.utils.collections import chunked

BATCH_SIZE = 500          # provider limit per API call

# Same shape the shop validates at signup and checkout before an address is stored.
_EMAIL_RE = re.compile(r"^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$")


def _valid_email(email: str) -> bool:
    """True when the address looks like a real email; the provider rejects everything else."""
    return isinstance(email, str) and bool(_EMAIL_RE.match(email))


BODY = "Hi {{ first_name }},\n\n{{ intro }}\n\nShop now: {{ link }}\n\nUnsubscribe: {{ unsubscribe }}"


def recipients(customers: list[Customer]) -> list[Customer]:
    return [c for c in customers if c.marketing_opt_in and _valid_email(c.email)]


def render_for(customer: Customer, campaign: Campaign, intro: str) -> str:
    return render_template(BODY, {"first_name": customer.first_name or "there", "intro": intro,
                                  "link": campaign.link(),
                                  "unsubscribe": f"https://shop.example/unsubscribe/{customer.id}"})


def batches(customers: list[Customer]) -> list[list[str]]:
    return [[c.email for c in batch] for batch in chunked(recipients(customers), BATCH_SIZE)]
