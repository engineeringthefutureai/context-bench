```
import re
from decimal import ROUND_HALF_UP, Decimal

_WEIGHT_RE = re.compile(r"\s*(?P<num>\d+(?:\.\d+)?)\s*(?P<unit>[a-zA-Z]+)\s*")
_UNITS_TO_GRAMS = {
    "g": Decimal(1),
    "kg": Decimal(1000),
    "lb": Decimal("453.59237"),
    "oz": Decimal("28.349523125"),
}

def parse_weight(text: str) -> int:
    """Parse a staff-typed package weight into whole grams.

    Accepts an integer or decimal number followed by one of g, kg, lb or oz,
    case-insensitively, with optional surrounding whitespace. Raises ValueError
    for anything else.
    """
    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")
    unit = match["unit"].lower()
    factor = _UNITS_TO_GRAMS.get(unit)
    if factor is None:
        raise ValueError(f"")
    value = Decimal(match["num"])
    grams = value * factor
    return int(grams.quantize(Decimal("1"), rounding=ROUND_HALF_UP))
```
