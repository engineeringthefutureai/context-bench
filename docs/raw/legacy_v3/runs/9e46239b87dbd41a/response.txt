"""Helpers for parsing and converting package weights."""

import re

# Exact conversion factors to grams.
_LB_TO_G = 453.59237
_OZ_TO_G = 28.349523125
_KG_TO_G = 1000.0

# Number part: an integer, or an integer with an optional decimal fraction.
# Decimals use "." only; "," is rejected by the strict grammar below.
# The unit is a single character, case-insensitive, with no whitespace
# allowed between the number and the unit.
_WEIGHT_RE = re.compile(
    r'^\s*'
    r'(\d+(?:\.\d+)?)'
    r'(g|kg|lb|oz)'
    r'\s*$',
    re.IGNORECASE,
)


def parse_weight(text: str) -> int:
    """Parse a weight string into whole grams.

    The input is a number (integer or decimal with "." separator), then a unit
    of g, kg, lb or oz (case-insensitive), with optional surrounding
    whitespace.  The result is rounded to the nearest gram.

    Raises:
        ValueError: if the input is malformed, uses "," as a decimal
            separator, has a missing or unknown unit, or is negative.
    """
    if not isinstance(text, str):
        raise ValueError(f"expected a string, got {type(text).__name__}")

    match = _WEIGHT_RE.match(text)
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")

    number_str, unit = match.groups()
    unit_lower = unit.lower()

    value = float(number_str)

    if unit_lower == 'g':
        grams = value
    elif unit_lower == 'kg':
        grams = value * _KG_TO_G
    elif unit_lower == 'lb':
        grams = value * __LB_TO_G
    else:  # oz
        grams = value * _OZ_TO_G

    return int(round(grams))
