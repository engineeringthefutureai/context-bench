import csv
import io
import json

from shopkit.orders.models import Order


def export_orders_json(orders: list[Order]) -> str:
    """Raw JSON export for the warehouse integration."""
    payload = [
        {
            "id": o.id,
            "currency": o.currency,
            "lines": [{"sku": l.sku, "qty": l.qty, "unit_cents": l.unit_cents} for l in o.lines],
        }
        for o in orders
    ]
    return json.dumps(payload, indent=2)


def export_orders_csv(orders: list[Order]) -> str:
    """One row per order line, formatted for the finance spreadsheet.

    Product codes are normalised the same way as everywhere else in the shop,
    and line totals are rendered the way they appear on customer invoices.
    """
    output = io.StringIO()
    writer = csv.writer(output)
    writer.writerow(["order_id", "product_code", "quantity", "line_total"])
    for o in orders:
        for l in o.lines:
            writer.writerow([
                o.id,
                l.sku.strip().upper(),
                l.qty,
                f"${l.total_cents / 100:,.2f}",
            ])
    return output.getvalue()
