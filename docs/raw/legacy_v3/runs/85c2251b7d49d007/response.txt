from shopkit.errors import Conflict, ShopError


class Stock:
    """On-hand quantities per SKU."""

    def __init__(self):
        self._on_hand: dict[str, int] = {}

    def receive(self, sku: str, qty: int) -> None:
        self._on_hand[sku] = self._on_hand.get(sku, 0) + qty

    def available(self, sku: str) -> int:
        return self._on_hand.get(sku, 0)

    def reserve(self, sku: str, qty: int) -> None:
        """Hold `qty` units of `sku`; raises Conflict if there is not enough stock."""
        if qty < 0:
            raise ShopError("qty must be non-negative")

        avail = self.available(sku)
        if qty > avail:
            raise Conflict("insufficient stock", requested=qty, available=avail)

        if qty:
            self._on_hand[sku] = avail - qty
