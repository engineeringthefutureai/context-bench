"""Utilities for parsing and working with duration values."""

import re


def parse_duration(text: str) -> int:
    """
    Parse a duration string and return its value in whole minutes.

    Accepted formats include "2h", "45m", "1h30m", and "1h 30m".
    - Hours come before minutes.
    - Each unit appears at most once.
    - Units are case-insensitive.
    - Surrounding whitespace is ignored.

    Raises ValueError for any malformed input.
    """
    if not isinstance(text, str):
        raise ValueError(f"Expected a string, got {type(text).__name__}")
    
    # Strip leading and trailing whitespace
    text = text.strip()
    
    if not text:
        raise ValueError("Cannot parse empty duration string")
    
    # Remove all internal whitespace for easier matching
    normalized = re.sub(r'\s+', '', text)
    
    # Pattern: Optional (hours_part) followed by Optional (minutes_part)
    # h_part: number followed by 'h' or 'H'
    # m_part: number followed by 'm' or 'M'
    # Both parts must be present or one of them must be present.
    pattern = r'^(\d+h)?(\d+m)?$'
    match = re.match(pattern, normalized, re.IGNORECASE)
    
    if not match:
        raise ValueError(f"Invalid duration format: '{text}'")
    
    h_part = match.group(1)
    m_part = match.group(2)
    
    # Ensure at least one part is present (the regex allows both to be None, which happens only if text was empty, but we checked for that)
    # However, the pattern `(\d+h)?(\d+m)?` could match an empty string if we didn't check for it. Since we checked, it's fine.
    # But wait, if the input was just "h", it wouldn't match. If it was "1h2h", it wouldn't match.
    
    if h_part is None and m_part is None:
        raise ValueError(f"Invalid duration format: '{text}'")
        
    total_minutes = 0
    
    if h_part:
        hours = int(h_part[:-1])
        total_minutes += hours * 60
        
    if m_part:
        minutes = int(m_part[:-1])
        total_minutes += minutes
        
    return total_minutes
