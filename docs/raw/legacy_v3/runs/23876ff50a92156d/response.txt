"""Weight parsing for warehouse staff input.

Converts a weight expressed with a unit into whole grams.
"""

import re

# Matches an optional sign (we will reject negatives), an integer part,
# an optional fractional part using '.' as the decimal separator,
# optional whitespace, and a known unit.
_WEIGHT_RE = re.compile(
    r"""^\s*
    (?P<number>\d+(?:\.\d+)?)\s*
    (?P<unit>g|kg|lb|oz)\s*$
    """,
    re.VERBOSE | re.IGNORECASE,
)

# Conversion factors to grams.
_UNIT_TO_GRAMS = {
    'g': 1.0,
    'kg': 1000.0,
    'lb': 453.59237,
    'oz': 28.349523125,
}


def parse_weight(text: str) -> int:
    """Parse a weight string into whole grams.

    The input must be a non-negative number (integer or decimal using '.'),
    optional whitespace, and a unit: g, kg, lb, or oz (case-insensitive).
    Leading and trailing whitespace is ignored.

    Raises:
        ValueError: If the input is missing a number, has a missing or
            unknown unit, is negative, or uses ',' as a decimal separator.
    """
    if not isinstance(text, str):
        raise ValueError("Weight must be a string")

    match = _WEIGHT_RE.match(text)
    if match is None:
        raise ValueError(f"Invalid weight format: {text!r}")

    number_str = match.group('number')
    unit_str = match.group('unit').lower()

    # Convert to float for multiplication, then round to nearest gram
    try:
        number = float(number_str)
    except (ValueError, OverflowError):
        raise ValueError(f"Invalid number: {number_str!r}")

    grams = number * _UNIT_TO_GRAMS[unit_str]
    return round(grams)
