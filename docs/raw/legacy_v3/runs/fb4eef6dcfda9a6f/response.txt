Both the signup path (`CustomerDirectory.register`) and checkout use `shopkit.utils.validation.is_valid_email`, so the newsletter recipient filter should apply that same check on top of the existing opt-in check. Customers who opted in but whose (possibly typo'd, imported) address fails the check are silently skipped — consistent with how `recipients()` already drops non-opted-in customers and how `batches()` consumes it.

<file path="shopkit/marketing/newsletter.py">
```python
"""Newsletter sends. Only customers who opted in receive marketing email."""
from shopkit.customers.models import Customer
from shopkit.marketing.campaigns import Campaign
from shopkit.notifications.templates import render_template
from shopkit.utils.collections import chunked
from shopkit.utils.validation import is_valid_email

BATCH_SIZE = 500          # provider limit per API call

BODY = "Hi {{ first_name }},\n\n{{ intro }}\n\nShop now: {{ link }}\n\nUnsubscribe: {{ unsubscribe }}"


def recipients(customers: list[Customer]) -> list[Customer]:
    """Opted-in customers whose email passes the same check used at signup and checkout.
    Customers with an invalid (e.g. badly typed imported) email are skipped so they are
    not sent to and do not bounce."""
    return [c for c in customers if c.marketing_opt_in and is_valid_email(c.email)]


def render_for(customer: Customer, campaign: Campaign, intro: str) -> str:
    return render_template(BODY, {"first_name": customer.first_name or "there", "intro": intro,
                                  "link": campaign.link(),
                                  "unsubscribe": f"https://shop.example/unsubscribe/{customer.id}"})


def batches(customers: list[Customer]) -> list[list[str]]:
    return [[c.email for c in batch] for batch in chunked(recipients(customers), BATCH_SIZE)]
```
