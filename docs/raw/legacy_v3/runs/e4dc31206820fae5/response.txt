from dataclasses import dataclass, field


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"
    discount_percent: int = 0

    def __post_init__(self) -> None:
        if not 0 <= self.discount_percent <= 100:
            raise ValueError(f"discount_percent must be between 0 and 100, got {self.discount_percent}")

    @property
    def subtotal_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

    @property
    def discount_cents(self) -> int:
        if not self.discount_percent:
            return 0
        return self.subtotal_cents * self.discount_percent // 100

    @property
    def total_cents(self) -> int:
        return self.subtotal_cents - self.discount_cents

from shopkit.money import format_money
from shopkit.orders.models import Order


def invoice_lines(order: Order) -> list[str]:
    """Human-readable invoice body for an order."""
    lines: list[str] = []
    for l in order.lines:
        lines.append(f"{l.qty} x {l.sku}  {format_money(l.unit_cents, order.currency)}")
    if order.discount_percent:
        lines.append(f"Discount ({order.discount_percent}%)  -{format_money(order.discount_cents, order.currency)}")
    lines.append(f"Total  {format_money(order.total_cents, order.currency)}")
    return lines
