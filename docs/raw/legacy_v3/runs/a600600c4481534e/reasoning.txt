```python
import re

# "1h", "45m", or "1h30m"/"1h 30m": hours first, each unit at most once.
_HOURS_RE = re.compile(r"\A(\d+)h\Z", re.IGNORECASE | re.ASCII)
_MINUTES_RE = re.compile(r"\A(\d+)m\Z", re.IGNORECASE | re.ASCII)
_HOURS_MINUTES_RE = re.compile(r"\A(\d+)h\s*(\d+)m\Z", re.IGNORECASE | re.ASCII)


def parse_duration(text: str) -> int:
    """Whole minutes for a duration like "2h", "45m" or "1h 30m".

    Hours come before minutes and each unit may appear at most once;
    units are case-insensitive and surrounding whitespace is ignored.
    """
    cleaned = text.strip()
    match = _HOURS_MINUTES_RE.match(cleaned)
    if match:
        return int(match.group(1)) * 60 + int(match.group(2))
    match = _HOURS_RE.match(cleaned)
    if match:
        return int(match.group(1)) * 60
    match = _MINUTES_RE.match(cleaned)
    if match:
        return int(match.group(1))
    raise ValueError(f"invalid duration: {text!r}")
```
