"""Weight parsing for weights entered by warehouse staff."""
import re

# Grams per unit. lb and oz use the exact international definitions.
_GRAMS_PER_UNIT = {
    "g": 1.0,
    "kg": 1000.0,
    "lb": 453.59237,
    "oz": 28.349523125,
}

_WEIGHT_RE = re.compile(r"^\s*(\d+(?:\.\d+)?)\s*(g|kg|lb|oz)\s*$", re.IGNORECASE)


def parse_weight(text: str) -> int:
    """Weight entered by staff as number + unit ("2.5 kg", "12 LB", "8 oz") in whole grams.
    Raises ValueError for missing/unknown units, missing numbers, negative values or
    non-gram units."""
    m = _WEIGHT_RE.match(text)
    if not m:
        raise ValueError(f"invalid weight {text!r}")
    value = float(m.group(1))
    unit = m.group(2).lower()

"""Parse weights entered by warehouse staff into whole grams for the shipping flow."""
import re
from decimal import ROUND_HALF_UP, Decimal

# Grams per unit; lb and oz use the exact international avoirdupois definitions, so Decimal keeps
# the conversion lossless.
_GRAMS_PER = {
    "g": Decimal("1"),
    "kg": Decimal("1000"),
    "lb": Decimal("453.59237"),
    "oz": Decimal("28.349523125"),
}

_WEIGHT_RE = re.compile(r"^\s*(\d+(?:\.\d+)?)\s*(g|kg|lb|oz)\s*$", re.IGNORECASE)


def parse_weight(text: str) -> int:
    """Staff-entered weight as a number plus a unit, e.g. "2.5 kg", "12 LB", " 8oz " -> grams.
    The number is an integer or a "."-separated decimal ("," is not accepted); the unit is g, kg,
    lb or oz, case-insensitive; surrounding whitespace is ignored. The result is rounded to the
    nearest whole gram (halves round up). Raises ValueError for a missing number or unit, an
    unknown unit or a negative value."""
    m = _WEIGHT_RE.match(text)
    if not m:
        raise ValueError(f"invalid weight {text!r}")
    grams = Decimal(m.group(1)) * _GRAMS_PER[m.group(2).lower()]
    return int(grams.quantize(Decimal(1), rounding=ROUND_HALF_UP))
