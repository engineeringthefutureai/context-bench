To mask card numbers in webhook deliveries, I will add a helper function `_mask_card_number` and apply it within the `WebhookDispatcher.on_event` method before the payload is serialized. This ensures that any `card_number` field at the top level of the event payload is masked consistently before being sent to partners, while leaving other fields unchanged.

Here is the plan:
1.  Add a `_mask_card_number` function that handles strings of length >= 12 by keeping the last 4 digits, masking the rest with `*`. Shorter strings are fully masked to prevent leakage of sensitive data if it's not a full PAN.
2.  In `WebhookDispatcher.on_event`, create a copy of `event.payload` (if it's a dict).
3.  If `card_number` exists in the payload and is a string, replace it with the masked version.
4.  Use this sanitized payload for the JSON body.

I need to be careful about the exact implementation of the mask to match standard "shop" behaviors. Usually, "mask the same way" implies standard PCI compliance: `**** **** **** 1234` or just `************1234`. Given no specific context on the *exact* format (spaces vs no spaces), I will stick to a simple replacement of characters with `*` preserving length, which is common and safe. However, often "last 4" implies stripping the middle. Let's look at standard patterns. `input[0:-4].replace(...)`? No, usually it's `'*' * (len(s)-4) + s[-4:]`.

Let's refine `_mask_card_number`:
```python
def _mask_card_number(number: str) -> str:
    if len(number) < 12:
        return "*" * len(number)
    return "*" * (len(number) - 4) + number[-4:]
```

In `on_event`:
```python
data = event.payload
if isinstance(data, dict) and "card_number" in data:
    data = dict(data)
    if isinstance(data["card_number"], str):
        data["card_number"] = _mask_card_number(data["card_number"])
```

This ensures we don't mutate the original event payload object, which is good practice.

"""Outgoing webhooks to partner systems (ERP, marketing). Payloads are signed with HMAC-SHA256."""
import hashlib
import hmac
import json
from dataclasses import dataclass, field

from shopkit.events import Event, EventBus

MAX_ATTEMPTS = 5


@dataclass
class Endpoint:
    url: str
    secret: str
    events: set[str]                          # event names, or "order.*"-style prefixes


@dataclass
class Delivery:
    endpoint: Endpoint
    body: str
    signature: str
    attempts: int = 0
    delivered: bool = False
    errors: list[str] = field(default_factory=list)


def sign(body: str, secret: str) -> str:
    return "sha256=" + hmac.new(secret.encode(), body.encode(), hashlib.sha256).hexdigest()


def verify(body: str, secret: str, signature: str) -> bool:
    return hmac.compare_digest(sign(body, secret), signature)


def _subscribed(endpoint: Endpoint, name: str) -> bool:
    return name in endpoint.events or f"{name.split('.', 1)[0]}.*" in endpoint.events


def _mask_card_number(number: str) -> str:
    """Mask a card number, keeping only the last 4 digits."""
    if len(number) < 12:
        return "*" * len(number)
    return "*" * (len(number) - 4) + number[-4:]


class WebhookDispatcher:
    def __init__(self, endpoints: list[Endpoint], transport=None):
        self.endpoints = endpoints
        self.queue: list[Delivery] = []
        self._transport = transport           # callable(url, body, headers) -> status code

    def attach(self, bus: EventBus) -> None:
        for prefix in {e.split(".", 1)[0] for ep in self.endpoints for e in ep.events}:
            bus.subscribe(f"{prefix}.*", self.on_event)

    def on_event(self, event: Event) -> None:
        data = event.payload
        if isinstance(data, dict) and "card_number" in data:
            data = dict(data)
            if isinstance(data["card_number"], str):
                data["card_number"] = _mask_card_number(data["card_number"])
        
        body = json.dumps({"event": event.name, "at": event.at.isoformat(), "data": data},
                          sort_keys=True, default=str)
        for ep in self.endpoints:
            if _subscribed(ep, event.name):
                self.queue.append(Delivery(ep, body, sign(body, ep.secret)))

    def flush(self) -> int:
        """Try every pending delivery once; returns how many succeeded. Failures stay queued until
        MAX_ATTEMPTS."""
        ok = 0
        for d in list(self.queue):
            d.attempts += 1
            try:
                status = self._transport(d.endpoint.url, d.body, {"X-Signature": d.signature})
                d.delivered = 200 <= status < 300
                if not d.delivered:
                    d.errors.append(f"HTTP {status}")
            except OSError as e:
                d.errors.append(str(e))
            if d.delivered or d.attempts >= MAX_ATTEMPTS:
                self.queue.remove(d)
                ok += d.delivered
        return ok
