Implementation in models.py:

```python
from dataclasses import dataclass, field


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"
    discount_percent: int = 0

    def __post_init__(self) -> None:
        if not 0 <= self.discount_percent <= 100:
            raise ValueError("discount_percent must be between 0 and 100")

    @property
    def gross_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

    @property
    def discount_cents(self) -> int:
        # Discount is rounded down to whole cents.
        return self.gross_cents * self.discount_percent // 100

    @property
    def total_cents(self) -> int:
        return self.gross_cents - self.discount_cents
```

@dataclass

    unit_cents: int

@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"
    discount_percent: int = 0

    def __post_init__(self) -> None:
        if not 0 <= self.discount_percent <= 100:
            raise ValueError(f"discount_percent must be between 0 and 100, got {self.discount_percent}")

    @property
    def gross_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

    @property
    def discount_cents(self) -> int:
        # Whole cents, rounded down.
        return self.gross_cents * self.discount_percent // 100

    @property
    def total_cents(self) -> int:
        return self.gross_cents - self.discount_cents
