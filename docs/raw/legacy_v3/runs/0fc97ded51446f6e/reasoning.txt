_FACTOR = {
 'g': (1, 1),
 'kg': (1000, 1),
 'lb': (45359237, 100000),
 'oz': (28349523125, 1000000000),
}

def parse_weight(text: str) -> int:
    if not isinstance(text, str):
        raise ValueError("weight must be a string")
    match = _WEIGHT_RE.fullmatch(text.strip())
    if match is None:
       raise ValueError(f"Invalid weight: {text!r}")
    number = match.group("number")
    unit = match.group("unit").lower()
    if number.startswith("-"):
       raise ValueError("Weight cannot be negative")
    if number.startswith("+"):
       number = number[1:]
    if "." in number:
       integer_part, fractional_part = number.split(".")
       if integer_part == "":
          integer_part = "0"
       numerator = int(integer_part + fractional_part)
       denominator = 10 ** len(fractional_part)
    else:
       numerator = int(number)
       denominator = 1
    if numerator < 0: raise... (not needed)
    factor_num, factor_den = _FACTOR[unit]
    grams_num = numerator * factor_num
    grams_den = denominator * factor_den
    # round to nearest, half up
    quotient, remainder = divmod(grams_num, grams_den)
    if remainder * 2 >= grams_den:
        quotient += 1
    return quotient

_WEIGHT_RE = re.compile(
    r"(?P<number>[+-]?(?:[0-9]+(?:\.[0-9]+)?|\.[0-9]+))\s*(?P<unit>kg|g|lb|oz)",

def parse_weight(text: str) -> int:
