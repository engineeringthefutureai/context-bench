from dataclasses import dataclass, field


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"
    discount_percent: int = 0

    @property
    def total_cents(self) -> int:
        subtotal = sum(line.total_cents for line in self.lines)
        if self.discount_percent:
            discount = (subtotal * self.discount_percent) // 100
            return subtotal - discount
        return subtotal

def render_invoice_text(order: Order) -> str:
    # Plain-text invoice sent to customers by email.
    from shopkit.money import format_money
    from shopkit.orders.models import Order, OrderLine

    lines = []
    lines.append(f"Invoice {order.id}")
    lines.append("-" * 30)

    for line in order.lines:
        lines.append(f"{line.sku}: {line.qty} x {format_money(line.unit_cents, order.currency)}")

    subtotal = sum(line.total_cents for line in order.lines)
    total = order.total_cents

    if order.discount_percent > 0:
        discount_amount = subtotal - total
        lines.append(f"Discount: -{format_money(discount_amount, order.currency)}")

    lines.append("-" * 30)
    lines.append(f"Total: {format_money(total, order.currency)}")

    return "\n".join(lines)

import json
from shopkit.orders.models import Order

def export_orders_json(orders: list[Order]) -> str:
    # Raw JSON export for the warehouse integration.
    data = []
    for order in orders:
        order_dict = {
            "id": order.id,
            "currency": order.currency,
            "discount_percent": order.discount_percent,
            "lines": [
                {
                    "sku": line.sku,
                    "qty": line.qty,
                    "unit_cents": line.unit_cents
                }
                for line in order.lines
            ],
            "total_cents": order.total_cents
        }
        data.append(order_dict)
    return json.dumps(data)
