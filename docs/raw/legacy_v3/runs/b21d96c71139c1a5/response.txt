    sku: str
    qty: int_cents: int
    unit_cents: int

    @property
    def total_cents(self) -> int:

@dataclass

    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"
    discount_percent: int = 0

    @property
    def total_cents(self) -> int:
        gross = sum(line.total_cents for line in self.lines)

from shopkit.orders.models import Order
from shopkit.money import format_money


def render_invoice_text(order: Order) -> str:
    lines = []
    lines.append(f"Order: {order.id}")
    for line in order.lines:
        lines.append(f"{line.sku} x {line.qty} = {format_money(line.total_cents, order.currency)}")
    if order.discount_percent > 0:
        gross = sum(line.total_cents for line in order.lines)
        discount_cents = (gross * order.discount_percent) // 100
        lines.append(f"Discount ({order.discount_percent}%) = {format_money(discount_cents, order.currency)}")
    lines.append(f"Total = {format_money(order.total_cents, order.currency)}")
    return "\n".join(lines)

import json
from shopkit.orders.models import Order


def export_orders_json(orders: list[Order]) -> str:
    data = []
    for order in orders:
        item = {
            "id": order.id,
            "currency": order.currency,
            "discount_percent": order.discount_percent,
            "lines": [
                {
                    "sku": line.sku,
                    "qty": line.qty,
                    "unit_cents": line.unit_cents
                }
                for line in order.lines
            ],
            "total_cents": order.total_cents
        }
        data.append(item)
    return json.dumps(data, indent=2)
