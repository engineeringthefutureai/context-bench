import re

_DURATION_RE = re.compile(r"(?:(\d+)h)?\s*(?:(\d+)m)?", re.ASCII)


def parse_duration(text: str) -> int:
    """Parse a duration made of hours and/or minutes into total minutes.

    Accepts forms like "2h", "45m", "1h30m", and "1h 30m". Units are
    case-insensitive and surrounding whitespace is ignored.

    Raises:
        ValueError: If the text is not a valid duration string.
    """
    if not isinstance(text, str):
        raise ValueError("Invalid duration")

    normalized = text.strip().lower()
    if not normalized:
        raise ValueError("Invalid duration")

    match = _DURATION_RE.fullmatch(normalized)
    if match is None or (match.group(1) is None and match.group(2) is None):
        raise ValueError("Invalid duration")

    hours = int(match.group(1)) if match.group(1) is not None else 0
    minutes = int(match.group(2)) if match.group(2) is not None else 0
    return hours * 60 + minutes
