import re

_DURATION_RE = re.compile(
    r"\s*(?:(?P<hours>[0-9]+)h\s*(?P<minutes>[0-9]+)m"
    r"|(?P<hours_only>[0-9]+)h"
    r"|(?P<minutes_only>[0-9]+)m)\s*",
    re.IGNORECASE,
)


def parse_duration(text: str) -> int:
    """Parse hours and/or minutes into whole minutes, e.g. "1h30m" -> 90."""
    if not isinstance(text, str):
        raise ValueError("duration must be a string")

    match = _DURATION_RE.fullmatch(text)
    if match is None:
        raise ValueError("invalid duration")

    hours_text = match.group("hours") or match.group("hours_only") or "0"
    minutes_text = match.group("minutes") or match.group("minutes_only") or "0"
    return int(hours_text) * 60 + int(minutes_text)
