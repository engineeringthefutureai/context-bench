from dataclasses import dataclass, field


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"
    discount_percent: int = 0

    def __post_init__(self) -> None:
        if not 0 <= self.discount_percent <= 100:
            raise ValueError("discount_percent must be between 0 and 100")

    @property
    def subtotal_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

    @property
    def discount_cents(self) -> int:
        # Discount amount is rounded down to whole cents.
        return self.subtotal_cents * self.discount_percent // 100

    @property
    def total_cents(self) -> int:
        return self.subtotal_cents - self.discount_cents

    def invoice_lines(self) -> list[str]:
        """Customer invoice lines for this order."""
        lines = [
            f"{line.sku} x {line.qty} = {line.total_cents}"
            for line in self.lines
        ]
        if self.discount_cents:
            lines.append(f"Discount: {self.discount_cents}")
        lines.append(f"Total: {self.total_cents}")
        return lines

import json

from shopkit.orders.models import Order


def export_orders_json(orders: list[Order]) -> str:
    """Raw JSON export for the warehouse integration."""
    payload = [
        {
            "id": o.id,
            "currency": o.currency,
            "discount_percent": o.discount_percent,
            "lines": [{"sku": l.sku, "qty": l.qty, "unit_cents": l.unit_cents} for l in o.lines],
        }
        for o in orders
    ]
    return json.dumps(payload, indent=2)

from shopkit.orders.models import Order


def invoice_lines(order: Order) -> list[str]:
    """Customer invoice lines for an order."""
    return order.invoice_lines()


def render_invoice(order: Order) -> str:
    """Render customer invoice lines as a single string."""
    return "\n".join(invoice_lines(order))
