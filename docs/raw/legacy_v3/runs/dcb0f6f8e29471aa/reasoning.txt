def parse_weight(text: str) -> int:
    """Parse a package weight string into whole grams.

    Accepts a non-negative decimal or integer, optional whitespace, and one

"""Utilities for parsing physical units typed by warehouse staff."""

import re
from fractions import Fraction

_WEIGHT_RE = re.compile(
    r"\s*(?P<number>\d+(?:\.\d*)?|\.\d+)\s*(?P<unit>g|kg|lb|oz)\s*",
    re.IGNORECASE,
)

_GRAMS_PER_UNIT: dict[str, Fraction] = {
    "g": Fraction(1),
    "kg": Fraction(1000),
    "lb": Fraction(45359237, 100000),
    "oz": Fraction(28349523125, 1000000000),
}

_HALF_GRAM = Fraction(1, 2)


def parse_weight(text: str) -> int:
    """Return the weight in whole grams, rounded to the nearest gram."""
    if not isinstance(text, str):
        raise ValueError("weight must be a string")

    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")

    value = _decimal_to_fraction(match.group("number"))
    return int(value * _GRAMS_PER_UNIT[match.group("unit").lower()] + _HALF_GRAM)


def _decimal_to_fraction(text: str) -> Fraction:
    """Convert a validated decimal string to an exact Fraction."""
    if "." not in text:
        return Fraction(text)

    whole, fraction = text.split(".", 1)
    denominator = 10 ** len(fraction)
    numerator = (int(whole) if whole else 0) * denominator
    if fraction:
        numerator += int(fraction)
    return Fraction(numerator, denominator)

import re
from fractions import Fraction

_WEIGHT_RE = re.compile(
    r"\s*(?P<number>[0-9]+(?:\.[0-9]*)?|\.[0-9]+)\s*(?P<unit>g|kg|lb|oz)\s*",
    re.IGNORECASE,
)

_GRAMS_PER_UNIT: dict[str, Fraction] = {
    "g": Fraction(1),
    "kg": Fraction(1000),
    "lb": Fraction("453.59237"),
    "oz": Fraction("28.349523125"),
}

_HALF_GRAM = Fraction(1, 2)


def parse_weight(text: str) -> int:
    """Parse a warehouse weight string into whole grams.

    The input must be a non-negative integer or decimal, optional
    whitespace, and one of the units g, kg, lb, or oz. Values are rounded
    to the nearest gram (ties round up).
    """
    if not isinstance(text, str):
        raise ValueError("weight must be a string")

    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")

    grams = _decimal_to_fraction(match.group("number")) * _GRAMS_PER_UNIT[
        match.group("unit").lower()
    ]
    return int(grams + _HALF_GRAM)


def _decimal_to_fraction(text: str) -> Fraction:
    """Convert a validated decimal string to an exact Fraction."""
    if "." not in text:
        return Fraction(text)

    whole, fraction = text.split(".", 1)
    denominator = 10 ** len(fraction)
    numerator = (int(whole) if whole else 0) * denominator
    if fraction:
        numerator += int(fraction)
    return Fraction(numerator, denominator)
