from typing import Dict, List

from shopkit.catalog.models import Product


def parcel_weight_g(parcel: Dict[str, int], products: Dict[str, Product]) -> int:
    """
    Calculate the total weight of a parcel's contents.

    Args:
        parcel: A dictionary mapping SKU to quantity.
        products: A dictionary mapping SKU to Product.

    Returns:
        The total weight in grams.
    """
    total_weight = 0
    for sku, quantity in parcel.items():
        if sku in products:
            total_weight += products[sku].weight_g * quantity
    return total_weight

from shopkit.catalog.models import Product

    Args:
        product: The Product instance.

    Returns:
        A dictionary representing the product.

    return {
