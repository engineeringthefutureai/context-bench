import re

_WEIGHT_RE = re.compile(
    r"\s*\+?([0-9]+(?:\.[0-9]+)?|\.[0-9]+)\s*(g|kg|lb|oz)\s*",
    re.IGNORECASE,
)

# Grams per unit, written as exact (numerator, denominator) pairs.
_GRAMS_PER_UNIT = {
    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (28349523125, 1000000000),
}


def _number_to_fraction(number: str) -> tuple[int, int]:
    """Return an exact numerator and denominator for a decimal string."""
    if "." in number:
        whole, _, fraction = number.partition(".")
        digits = whole + fraction
        numerator = int(digits) if digits else 0
        denominator = 10 ** len(fraction)
    else:
        numerator = int(number)
        denominator = 1
    return numerator, denominator


def parse_weight(text: str) -> int:
    """Parse a staff-typed weight into whole grams.

    Accepts a non-negative integer or decimal number, optional whitespace, and
    one of g, kg, lb, or oz (case-insensitive). The result is rounded to the
    nearest gram.
    """
    if not isinstance(text, str):
        raise ValueError("weight must be a string")

    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError("invalid weight")

    number, unit = match.group(1), match.group(2).lower()
    numerator, denominator = _number_to_fraction(number)
    unit_num, unit_den = _GRAMS_PER_UNIT[unit]

    grams_num = numerator * unit_num
    grams_den = denominator * unit_den
    return (grams_num + grams_den // 2) // grams_den
