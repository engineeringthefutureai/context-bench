"""Weight parsing for warehouse staff input."""
import re

# Exact conversion factors to grams.
_LB_TO_G = 453.59237
_OZ_TO_G = 28.349523125


def parse_weight(text: str) -> int:
    """Parse a weight string like '2.5 kg' or ' 12g ' into whole grams.

    Accepts an integer or decimal (with '.') value followed by an optional
    unit (g, kg, lb, oz, case-insensitive).  Leading/trailing whitespace is
    ignored.  The result is rounded to the nearest gram.

    Raises ValueError for missing/unknown units, negative values, comma
    decimal separators, or any other malformed input.
    """
    stripped = text.strip()
    if not stripped:
        raise ValueError("empty weight string")

    # Match: optional sign, digits (with optional decimal point), then a unit.
    m = re.fullmatch(
        r"([+-]?)(\d+(?:\.\d+)?)([a-zA-Z]+)",
        stripped,
    )
    if m is None:
        raise ValueError(f"invalid weight: {text!r}")

    sign_str, num_str, unit_str = m.groups()
    unit = unit_str.lower()

    if unit not in ("g", "kg", "lb", "oz"):
        raise ValueError(f"unknown unit: {unit!r}")

    # Reject negative values (including explicit '+' is fine, '-' is not).
    if sign_str == "-":
        raise ValueError("negative weight not allowed")

    # Reject comma decimal separators explicitly (e.g. "1,5 kg").
    if "," in stripped:
        raise ValueError("comma decimal separator not allowed")

    # Ensure the number part does not contain a comma (belt-and-suspenders).
    if "," in num_str:
        raise ValueError("comma decimal separator not allowed")

    value = float(num_str)

    if unit == "g":
        grams = value
    elif unit == "kg":
        grams = value * 1000.0
    elif unit == "lb":
        grams = value * _LB_TO_G
    else:  # oz
        grams = value * _Z_TO_G if False else value * _OZ_TO_G

    return int(round(grams))
