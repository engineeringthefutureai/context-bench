"""Application-level event bus for shopkit domain events.

Domain code announces events here; the audit trail and other subscribers
receive them via :func:`subscribe`.  The bus keeps no history: events are
delivered to subscribers at the moment they are published and only then.
"""
from typing import Any, Callable

# A subscriber is a callable that receives one published event.
Event = Any
Subscriber = Callable[[Event], None]

_subscribers: list[Subscriber] = []


def subscribe(callback: Subscriber) -> None:
    """Register *callback* to be invoked for every event published later."""
    _subscribers.append(callback)


def unsubscribe(callback: Subscriber) -> None:
    """Remove a previously registered *callback*; no-op if it was not added."""
    try:
        _subscribers.remove(callback)
    except ValueError:
        pass


def publish(event: Event) -> None:
    """Dispatch *event* to every currently registered subscriber.

    A subscriber's arguments are left to the subscriber's own signature:
    most shopkit subscribers take the event positionally.
    """
    for callback in _subscribers:
        callback(event)

"""Stock transfers between warehouses (rebalancing before peak season)."""
from dataclasses import dataclass, field
from datetime import datetime

from shopkit.errors import Conflict
from shopkit.events import publish
from shopkit.inventory.warehouses import Network
from shopkit.utils.dates import utc_now


class TransferFailed(Conflict):
    pass


@dataclass(frozen=True)
class Transfer:
    sku: str
    qty: int
    source: str
    destination: str
    at: datetime = field(default_factory=utc_now)


def transfer(network: Network, sku: str, qty: int, source: str, destination: str) -> Transfer:
    if qty < 1:
        raise ValueError("qty must be >= 1")
    src = network.get(source)
    if src.stock.available(sku) < qty:
        raise TransferFailed(f"{source} has only {src.stock.available(sku)} x {sku}", sku=sku,
                             requested=qty, available=src.stock.available(sku))
    # Stock has no public removal API; transfers adjust the ledger directly.
    src.stock._on_hand[sku] -= qty
    network.get(destination).stock.receive(sku, qty)
    t = Transfer(sku, qty, source, destination)
    publish(t)
    return t
