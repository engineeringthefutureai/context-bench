<content>"""Domain event bus: decouple side effects (audit, notifications) from the code that raises them."""
from dataclasses import dataclass, field
from datetime import datetime
from typing import Callable

from shopkit.utils.dates import utc_now

Handler = Callable[["Event"], None]


@dataclass(frozen=True)
class Event:
    name: str
    payload: dict
    at: datetime = field(default_factory=utc_now)


class EventBus:
    def __init__(self) -> None:
        self._subscribers: dict[str, list[Handler]] = {}

    def subscribe(self, name: str, handler: Handler) -> None:
        """name may end with ".*" to receive every event under a prefix, e.g. "order.*"."""
        self._subscribers.setdefault(name, []).append(handler)

    def _targets(self, name: str) -> list[Handler]:
        handlers = list(self._subscribers.get(name, ()))
        prefix, _, _ = name.rpartition(".")
        if prefix:
            handlers.extend(self._subscribers.get(prefix + ".*", ()))
        return handlers

    def publish(self, name: str, **payload) -> Event:
        event = Event(name=name, payload=dict(payload), at=utc_now())
        for handler in self._targets(name):
            handler(event)
        return event


#: Process-wide bus that handlers such as the audit log and notifier attach to.
BUS = EventBus()

<content>"""Audit trail of who did what to what, written to the log on every domain event."""
from dataclasses import dataclass, field
from datetime import datetime

from shopkit.events import BUS, Event
from shopkit.utils.dates import utc_now


@dataclass(frozen=True)
class AuditEntry:
    actor: str
    action: str
    target: str
    details: dict = field(default_factory=dict)
    at: datetime = field(default_factory=utc_now)


class AuditLog:
    def __init__(self) -> None:
        self._entries: list[AuditEntry] = []

    def record(self, actor: str, action: str, target: str, **details) -> AuditEntry:
        entry = AuditEntry(actor=actor, action=action, target=target, details=dict(details), at=utc_now())
        self._entries.append(entry)
        return entry

    def for_target(self, target: str) -> list[AuditEntry]:
        return [e for e in self._entries if e.target == target]

    def on_event(self, event: Event) -> None:
        """Bus subscriber: record every domain event is filed under its target, actor "system"."""
        target = event.payload.get("target") or event.payload.get("sku") or event.payload.get("id") or ""
        details = {k: v for k, v in event.payload.items() if k not in ("target", "sku", "id")}
        self.record("system", event.name, str(target), **details)

    def __len__(self) -> int:
        return len(self._entries)


#: Shared audit log; attach to the bus once at startup.
LOG = AuditLog()
LOG._on_event = LOG.on_event

<content>"""Suggest reorders before stock runs out of supplier lead time."""
from dataclasses import dataclass
from math import floor

from shopkit.config import SETTINGS
from shopkit.events import BUS
from shopkit.inventory.stock import Stock


def days_of_cover(on_hand: int, units_sold: int, period_days: int) -> float:
    """How many days current stock lasts at the average daily sales rate of the period."""
    if units_sold <= 0:
        return float("inf")
    rate = units_sold / period_days
    return on_hand / rate


@dataclass(frozen=True)
class Suggestion:
    sku: str
    on_hand: int
    cover_days: float
    order_qty: int


def suggest(stock: Stock, sales_30d: dict[str, int]) -> list[Suggestion]:
    """SKUs whose cover won't last the supplier lead time plus safety stock."""
    out: list[Suggestion] = []
    for sku, sold in sales_30d.items():
        on_hand = stock.available(sku)
        cover = days_of_cover(on_hand, sold, 30)
        if cover == float("inf"):
            continue
        if cover >= SETTINGS.low_stock_threshold:
            continue
        out.append(
            Suggestion(
                sku=sku,
                on_hand=on_hand,
                cover_days=round(cover, 1),
                order_qty=max(0, SETTINGS.low_stock_threshold * 2 - on_hand),
            )
        )
        BUS.publish("inventory.low", sku=sku, available=on_hand, threshold=SETTINGS.low_stock_threshold)
    return sorted(out, key=lambda s: (s.cover_days, s.sku))
