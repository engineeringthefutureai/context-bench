- **Product code cleaning**: `normalize_sku` from `shopkit/utils/text.py` is the canonical cleaner used across the shop — catalog import, search, variants, and the customer invoice render all use it.
- **Line totals as on customer invoices**: `shopkit/invoices/render.py` renders each line total as `format_money(line.total_cents, order.currency)` (e.g. `$1,234.56`), so the CSV must use the same call.
- One row per order line, header exactly `order_id,product_code,quantity,line_total`, built with the standard `csv` module (which will properly quote the `$1,234.56` values since they contain commas).

import csv
import io
import json

from shopkit.money import format_money
from shopkit.orders.models import Order
from shopkit.utils.text import normalize_sku


def export_orders_json(orders: list[Order]) -> str:
    """Raw JSON export for the warehouse integration."""
    payload = [
        {
            "id": o.id,
            "currency": o.currency,
            "lines": [{"sku": l.sku, "qty": l.qty, "unit_cents": l.unit_cents} for l in o.lines],
        }
        for o in orders
    ]
    return json.dumps(payload, indent=2)


def export_orders_csv(orders: list[Order]) -> str:
    """Spreadsheet export for finance: one row per order line."""
    buf = io.StringIO()
    writer = csv.writer(buf)
    writer.writerow(["order_id", "product_code", "quantity", "line_total"])
    for o in orders:
        for l in o.lines:
            writer.writerow([o.id, normalize_sku(l.sku), l.qty, format_money(l.total_cents, o.currency)])
    return buf.getvalue()
