Implementation:

```python
import csv
import io
import json

from shopkit.money import format_money
from shopkit.orders.models import Order
from shopkit.utils.text import normalize_sku


def export_orders_json(orders: list[Order]) -> str:
    """Raw JSON export for the warehouse integration."""
    payload = [
        {
            "id": o.id,
            "currency": o.currency,
            "lines": [{"sku": l.sku, "qty": l.qty, "unit_cents": l.unit_cents} for l in o.lines],
        }
        for o in orders
    ]
    return json.dumps(payload, indent=2)


def export_orders_csv(orders: list[Order]) -> str:
    """Spreadsheet export for finance: one row per order line."""
    buf = io.StringIO()
    writer = csv.writer(buf)
    writer.writerow(["order_id", "product_code", "quantity", "line_total"])
    for o in orders:
        for l in o.lines:
            writer.writerow([o.id, normalize_sku(l.sku), l.qty, format_money(l.total_cents, o.currency)])
    return buf.getvalue()
```

Edge cases: empty order list → header only. This is correct. Orders with no lines → no rows.
