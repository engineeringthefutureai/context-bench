"""Utility to parse human-readable duration strings into minutes."""

import re


def parse_duration(text: str) -> int:
    """
    Parse a duration string (e.g. "1h30m", "45m", "1h") into an integer number of minutes.

    Args:
        text: A string representing a duration in hours and/or minutes.

    Returns:
        The duration in whole minutes.

    Raises:
        ValueError: If the input does not match the expected format.
    """
    if not isinstance(text, str):
        raise ValueError("Input must be a string")

    # Remove leading/trailing whitespace
    text = text.strip()

    if not text:
        raise ValueError("Duration string is empty")

    # Pattern to match valid durations
    # Allow optional hours, then optional minutes.
    # Allow optional whitespace between hours and minutes.
    # Units are case-insensitive.
    pattern = r'^(?:(\d+)h)?\s*(?:(\d+)m)?$'
    match = re.match(pattern, text, re.IGNORECASE)

    if not match:
        raise ValueError(f"Invalid duration format: {text!r}")

    hours_str, minutes_str = match.groups()

    # Ensure at least one unit is present
    if hours_str is None and minutes_str is None:
        raise ValueError(f"Invalid duration format: {text!r}")

    hours = int(hours_str) if hours_str is not None else 0
    minutes = int(minutes_str) if minutes_str is not None else 0

    return hours * 60 + minutes
