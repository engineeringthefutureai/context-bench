import re

__all__ = ["parse_duration"]

_DURATION_PATTERN = re.compile(
    r"(?:(?P<hours>[0-9]+)h)?\s*(?:(?P<minutes>[0-9]+)m)?",
    re.IGNORECASE,
)


def parse_duration(text: str) -> int:
    """Return the number of whole minutes represented by a duration string.

    Valid input consists of an optional hours part and/or an optional minutes
    part, e.g. "2h", "45m", "1h30m" or "1h 30m". Hours must come before
    minutes, each unit appears at most once, units are case-insensitive, and
    surrounding whitespace is ignored.
    """
    if not isinstance(text, str):
        raise ValueError("duration must be a string")

    stripped = text.strip()
    if not stripped:
        raise ValueError("duration must not be empty")

    match = _DURATION_PATTERN.fullmatch(stripped)
    if match is None:
        raise ValueError("invalid duration")

    hours_raw = match.group("hours")
    minutes_raw = match.group("minutes")
    if hours_raw is None and minutes_raw is None:
        raise ValueError("invalid duration")

    hours = int(hours_raw) if hours_raw is not None else 0
    minutes = int(minutes_raw) if minutes_raw is not None else 0
    return hours * 60 + minutes
