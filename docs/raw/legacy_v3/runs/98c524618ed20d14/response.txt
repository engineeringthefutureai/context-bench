"""Parsing of weights typed in by warehouse staff."""

import re

# Exact conversion factors, in grams per unit.
_LB_TO_G = 453.59237
_OZ_TO_G = 28.349523125

# A number (integer or decimal using "." as the separator), optional whitespace,
# then a unit: g, kg, lb or oz (case-insensitive).
_WEIGHT_RE = re.compile(
    r"^(\d+(?:\.\d+)?)\s*(g|kg|lb|oz)$",
    re.IGNORECASE,
)


def parse_weight(text: str) -> int:
    """Parse a weight string typed by warehouse staff and return whole grams.

    The input is a number (an integer or a decimal using "." as separator),
    optional whitespace, then a unit: g, kg, lb or oz, case-insensitive.
    Leading and trailing whitespace is ignored. 1 lb = 453.59237 g and
    1 oz = 28.349523125 g; the result is rounded to the nearest gram.

    Raises ValueError for anything else: a missing number, a missing or
    unknown unit, negative values, or "," as the decimal separator.
    """
    match = _WEIGHT_RE.match(text.strip())
    if match is None:
        raise ValueError(f"Invalid weight: {text!r}")
    number = float(match.group(1))
    unit = match.group(2).lower()
    if unit == "g":
        grams = number
    elif unit == "kg":
        grams = number * 1000
    elif unit == "lb":
        grams = number * _LB_TO_G
    else:  # oz
        grams = number * _OZ_TO_G
    return int(round(grams))
