from dataclasses import dataclass, field


@dataclass
class OrderLine:
    sku: str
    qty: int
    unit_cents: int

    @property
    def total_cents(self) -> int:
        return self.qty * self.unit_cents


@dataclass
class DuplicateOrderLine(ShopError):
    ...


@dataclass
class Order:
    id: str
    lines: list[OrderLine] = field(default_factory=list)
    currency: str = "USD"

    @property
    def total_cents(self) -> int:
        return sum(line.total_cents for line in self.lines)

    def add_line(self, sku: str, qty: int, unit_cents: int) -> OrderLine:
        """Add a line to the order. If a line for the same SKU at the same unit price already
        exists, its quantity is increased instead of adding a duplicate line; the line touched
        is returned. A line for the same SKU at a different price raises, since that would need
        a human decision (and the SKU is expected to appear at most once in a single order)."""
        for line in self.lines:
            if line.sku == sku:
                if line.unit_cents == unit_cents:
                    line.qty += qty
                    return line
                raise DuplicateOrderLine(f"line for {sku!r} already exists at {line.unit_cents} cents",
                                         sku=sku)
        line = OrderLine(sku, qty, unit_cents)
        self.lines.append(line)
        return line
