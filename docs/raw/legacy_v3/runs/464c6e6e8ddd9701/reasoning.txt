"""Parse compact duration text like "1h30m" into whole minutes."""

import re

# Hours and/or minutes, hours first, each unit at most once. Unit letters are
# case-insensitive; whitespace around and between the parts is ignored.
_DURATIONS_RE = re.compile(
    r"^\s*(?:(\d+)h\s*)?(?:(\d+)m)?\s*$",
    re.IGNORECASE,
)


def parse_duration(text: str) -> int:
    """Return the duration in `text` as whole minutes.

    Accepts hours and/or minutes such as "2h", "45m", "1h30m" or "1h 30m":
    hours come before minutes, each unit appears at most once, and the unit
    letters are case-insensitive. Surrounding whitespace is ignored.
    Anything else raises ValueError.
    """
    match = _DURATIONS_RE.match(text)
    if match is None:
        raise ValueError(f"invalid duration: {text!r}")
    hours = match.group(1)
    minutes = match.group(2)
    if hours is None and minutes is None:
        raise ValueError(f"invalid duration: {text!r}")
    return int(hours or 0) * 60 + int(minutes or 0)

import re

# Hours and/or minutes, hours before minutes, each unit at most once. The unit
# letters are case-insensitive; whitespace around the parts is ignored.
_DURATION_RE = re.compile(
    r"^\s*(?:([0-9]+)h\s*)?(?:([0-9]+)m)?\s*$",
    re.IGNORECASE,
)


def parse_duration(text: str) -> int:
    """Return the duration in `text` as whole minutes.

    `text` is hours and/or minutes such as "2h", "45m", "1h30m" or "1h 30m".
    Hours come before minutes, each unit appears at most once, units are
    case-insensitive, and surrounding whitespace is ignored. Anything else
    (empty text, other units, a missing number, minutes before hours,
    negative numbers) raises ValueError.
    """
    match = _DURATION_RE.match(text)
    if match is None:
        raise ValueError(f"invalid duration: {text!r}")
    hours = match.group(1)
    minutes = match.group(2)
    if hours is None and minutes is None:
        raise ValueError(f"invalid duration: {text!r}")
    return (int(hours) if hours is not None else 0) * 60 + (int(minutes) if minutes is not None else 0)
