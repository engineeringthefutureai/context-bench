```python
"""Helpers for the units warehouse staff type in by hand."""

import re
from decimal import Decimal, ROUND_HALF_UP

# Exact conversion to grams (1 lb = 453.59237 g, 1 oz = 28.3495231 oz = 28.349523125 g), kept as
# decimals so rounding is exact.
_GRAMS_PER_UNIT = {
    "g": Decimal(1),
    "kg": Decimal(1000),
    "lb": Decimal("453.59237"),
    "oz": Decimal("28.349523125"),
}

# A number (whole or decimal with "." separator), optional whitespace, then a unit.
_WEIGHT_RE = re.compile(r"(\d+(?:\.\d+)?)\s*(g|kg|lb|oz)\Z", re.IGNORECASE)


def parse_weight(text: str) -> int:
    """Whole grams for a weight typed in by warehouse staff, e.g. parse_weight("1.5 kg") -> 1500,
    parse_weight("1 lb") -> 454, parse_weight("1 oz") -> 28. Units are g, kg, lb or oz,
    case-insensitive, with optional whitespace between number and unit. Raises ValueError
    for missing/unknown units, negative values and "," as decimal separator."""
    match = _WEIGHT_RE.fullmatch(text.strip())
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")
```

```python
import re
from decimal import Decimal, ROUND_HALF_UP

# Exact factors to grams, as decimals so rounding to a whole gram is exact
# (1 lb = 453.59237 g, 1 oz = 28.349523125 g).
_GRAMS_PER_UNIT = {
    "g": Decimal(1),
    "kg": Decimal(1000),
    "lb": Decimal("453.59237"),
    "oz": Decimal("28.349523125"),
}

# Number (whole or with a "." decimal separator), optional whitespace, unit;
# anything else (negatives, ",", missing part) fails the match.
_WEIGHT_RE = re.compile(r"\s*(\d+(?:\.\d+)?)\s*(g|kg|lb|oz)\s*", re.IGNORECASE)


def parse_weight(text: str) -> int:
    """Weight typed in by warehouse staff, in whole grams. Accepts a number (whole or
    decimal with "." separator), optional whitespace, then g, kg, lb or oz, case-insensitive:
    parse_weight("1.5 kg") -> 1500, parse_weight("1 lb") -> 454, parse_weight("8 oz") -> 227.
    Raises ValueError for a missing or unknown unit, a negative value or "," as separator."""
    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")
    number, unit = match.groups()
    grams = Decimal(number) * _GRAMS_PER_UNIT[unit.lower()]
    return int(grams.to_integral_value(rounding=ROUND_HALF_UP))
```
