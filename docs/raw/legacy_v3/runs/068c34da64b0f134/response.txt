from shopkit.errors import Conflict


class NotEnoughStock(Conflict):
    """Raised when a reservation or removal exceeds the on-hand quantity."""

    def __init__(self, message: str, *, sku: str, requested: int, available: int) -> None:
        super().__init__(message)
        self.details = {"sku": sku, "requested": requested, "available": available}


class Stock:
    """On-hand quantities per SKU."""

    def __init__(self):
        self._on_hand: dict[str, int] = {}

    def receive(self, sku: str, qty: int) -> None:
        self._on_hand[sku] = self._on_hand.get(sku, 0) + qty

    def available(self, sku: str) -> int:
        return self._on_hand.get(sku, 0)

    def reserve(self, sku: str, qty: int) -> None:
        have = self._on_hand.get(sku, 0)
        if have < qty:
            raise NotEnoughStock(f"not enough stock of {sku}", sku=sku, requested=qty, available=have)
        self._on_hand[sku] = have - qty
