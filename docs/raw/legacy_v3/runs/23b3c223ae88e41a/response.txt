def split_into_parcels(order: Order) -> List[Parcel]:
    """Split an order into parcels.

    The gift message is carried on the first parcel only, so that it
    appears on exactly one packing slip.
    """
    if not order.lines:
        return [Parcel(order_id=order.id, gift_message=order.gift_message)]

    parcels: List[Parcel] = []
    for idx, line in enumerate(order.lines):
        parcel = Parcel(
            order_id=order.id,
            lines=[line],
            weight_g=line.qty * 100,
        )
        if idx == 0:
            parcel.gift_message = order.gift_message
        parcels.append(parcel)

    return parcels
