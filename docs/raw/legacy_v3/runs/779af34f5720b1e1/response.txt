"""Parsing helpers for human-typed measurements.

Weights are typed by warehouse staff in a number followed by a unit
(e.g. ``"2.5 kg"``) and are stored as whole grams.
"""

import re

__all__ = ["parse_weight"]

# Conversion factors from one unit of the given kind to grams.
# Kept as exact rationals (numerator/denominator) so that the
# "nearest whole gram" rounding is deterministic for any input size.
_GRAMS_PER_UNIT: dict[str, tuple[int, int]] = {
    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100),
    "oz": (28349523125, 1000),
}

# A leading minus (already rejected), an optional integer part, a
# single "." decimal separator and a required fraction, then optional
# whitespace and the unit.  The unit must be one of the known letters.
_WEIGHT_RE = re.compile(
    r"\A\s*(?P<number>\d+(?:\.\d+)?)\s*(?P<unit>[gGkKlLoOzZ])\s*\Z"
)


def _unit_denominator(unit: str) -> int:
    """Denominator of the exact grams-per-unit factor for *unit*."""
    return _GRAMS_PER_UNIT[unit.lower()][1]


def _grams_from_decimal(numerator: int, denominator: int) -> int:
    """Nearest whole gram for the non-negative value *numerator*/*denominator*."""
    # Round-to-nearest, ties going to the nearest even gram.
    tw = 2 * numerator
    q, r = divmod(tw, denominator)
    if 2 * r >= denominator:
        q += 1
    return q // 2

def parse_weight(text: str) -> int:  # noqa: F811
    """Parse a typed package weight and return it as whole grams.

    The input is a number (integer, or decimal using "." as the decimal
    separator), optional whitespace, then a unit of ``g``, ``kg``, ``lb``
    or ``oz`` (case-insensitive).  Leading and trailing whitespace is
    ignored.  The result is rounded to the nearest gram.

    Raises:
        ValueError: if the text is not a well-formed non-negative weight,
            including a missing number or unit, an unknown unit, a
            negative value, or "," used as a decimal separator.
    """
    if not isinstance(text, str):
        raise ValueError(f"weight must be a string, got {type(text).__name__}")

    # Reject the decimal separator we do not accept.
    if "," in text:
        raise ValueError(f"invalid weight: {text!r}")

    match = _WEIGHT_RE.match(text)
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")

    number = match.group("number")
    unit = match.group("unit").lower()

    if unit == "k":
        unit = "kg"

    if unit not in _GRAMS_PER_UNIT:
        raise ValueError(f"unknown unit: {match.group('unit')!r}")

    whole, dot, frac = number.partition(".")
    digits = int(whole + frac)
    number_denominator = 10 ** len(frac)

    factor_numerator, factor_denominator = _GRAMS_PER_UNIT[unit]

    # (digits / number_denominator) * (factor_numerator / factor_denominator)
    numerator = digits * factor_numerator
    denominator = number_denominator * factor_denominator

    return _grams_from_decimal(numerator, denominator)
