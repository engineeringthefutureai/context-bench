from dataclasses import dataclass, field

from shopkit.catalog.models import Product
from shopkit.orders.models import Order
from shopkit.shipping.packaging import format_weight
from shopkit.utils.collections import chunked, group_by


@dataclass
class Parcel:
    order_id: str
    number: int
    items: list[tuple[str, int]] = field(default_factory=list)    # (sku, qty)

    @property
    def item_count(self) -> int:
        return sum(q for _, q in self.items)


def parcel_weight_g(parcel: Parcel, products: dict[str, Product]) -> int:
    """Total weight in grams of a parcel's contents: each item line's product
    weight_g (per unit) times its quantity."""
    return sum(products[sku].weight_g * qty for sku, qty in parcel.items)


def packing_slip(parcel: Parcel, weight_g: int) -> str:
    """Packing slip text for a parcel: order/parcel header, weight, then one
    "  qty x SKU" line per item."""
    lines = [f"Order {parcel.order_id} - parcel {parcel.number}",
             f"Weight: {format_weight(weight_g)}"]
    lines.extend(f"  {qty} x {sku}" for sku, qty in parcel.items)
    return "\n".join(lines)
