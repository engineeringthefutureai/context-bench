from dataclasses import dataclass, field
from shopkit.orders.models import Order, OrderLine

MAX_ITEMS_PER_PARCEL = 10

@dataclass
class Parcel:
    order_id: str
    number: int
    items: list[tuple[str, int]] = field(default_factory=list)
    gift_message: str = ""

    @property
    def item_count(self) -> int:
        return sum(qty for _, qty in self.items)

def pick_list(orders: list[Order]) -> list[tuple[str, int]]:
    counts: dict[str, int] = {}
    for order in orders:
        for line in order.lines:
            counts[line.sku] = counts.get(line.sku, 0) + line.qty
    return sorted(counts.items())

def split_into_parcels(order: Order, max_items: int = MAX_ITEMS_PER_PARCEL) -> list[Parcel]:
    parcels = []
    current_items = []
    current_count = 0
    parcel_number = 1
    
    def flush():
        nonlocal current_items, current_count, parcel_number
        if current_items:
            gm = order.gift_message if parcel_number == 1 else ""
            parcels.append(Parcel(order_id=order.id, number=parcel_number, items=current_items, gift_message=gm))
            current_items = []
            current_count = 0
            parcel_number += 1

    for line in order.lines:
        for _ in range(line.qty):
            if current_count >= max_items:
                flush()
            current_items.append((line.sku, 1))
            current_count += 1
    flush()
    return parcels

def packing_slip(parcel: Parcel, weight_g: int) -> str:
    lines = [
        f"Order: {parcel.order_id}",
        f"Parcel: {parcel.number}",
        "",
        "Items:",
    ]
    # Sort items by sku for readability
    for sku, qty in sorted(parcel.items):
        lines.append(f"  {sku} x {qty}")
    
    lines.append("")
    lines.append(f"Weight: {weight_g}g")
    
    if parcel.gift_message:
        lines.append("")
        lines.append("Gift Message:")
        lines.append(f"  {parcel.gift_message}")
        
    return "\n".join(lines)

```python
def split_into_parcels(order: Order, max_items: int = MAX_ITEMS_PER_PARCEL) -> list[Parcel]:
    parcels = []
    current_items = []
    current_count = 0
    parcel_number = 1
    
    def flush():
        nonlocal current_items, current_count, parcel_number
        if current_items:
            gm = order.gift_message if parcel_number == 1 else ""
            parcels.append(Parcel(order_id=order.id, number=parcel_number, items=current_items, gift_message=gm))
            current_items = []
            current_count = 0
            parcel_number += 1

    for line in order.lines:
        # If the line alone is bigger than max_items, we must split it
        if line.qty > max_items:
            # Flush current if any
            if current_items:
                flush()
            # Split the line
            remaining = line.qty
            while remaining > 0:
                take = min(remaining, max_items)
                current_items.append((line.sku, take))
                current_count += take
                remaining -= take
                if remaining > 0:
                    flush()
        else:
            # Line fits in a parcel
            if current_count + line.qty > max_items:
                flush()
            current_items.append((line.sku, line.qty))
            current_count += line.qty
            
    flush()
    return parcels
```
