import re

# Case-insensitive: one or more digits followed by "h" and/or "m", hours before minutes,
# each unit at most once.  Surrounding whitespace is stripped by the caller.
_DURATION_RE = re.compile(r"^\d+h(?:\d+m)?$|^\d+m$")


def parse_duration(text: str) -> int:
    """Parse a duration of hours and/or minutes into whole minutes.

    Accepts forms such as "2h", "45m", "1h30m" and "1h 30m"; units are
    case-insensitive and surrounding whitespace is ignored.  Hours must come
    before minutes and each unit may appear at most once.

    Raises ValueError for empty text, unknown units, missing or non-integer
    numbers, minutes before hours, or negative numbers.
    """
    if text is None:
        raise ValueError("invalid duration: None")
    cleaned = text.strip()
    if not cleaned:
        raise ValueError("invalid duration: empty text")
    if not _DURATION_RE.match(cleaned):
        raise ValueError(f"invalid duration: {text!r}")
    hours = int(re.search(r"(\d+)h", cleaned, re.IGNORECASE).group(1)) if "h" in cleaned.lower() else 0
    minutes = int(re.search(r"(\d+)m", cleaned, re.IGNORECASE).group(1)) if "m" in cleaned.lower() else 0
    return hours * 60 + minutes
