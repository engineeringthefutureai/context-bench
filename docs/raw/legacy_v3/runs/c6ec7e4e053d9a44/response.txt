"""Parse compact duration strings into whole minutes.

Accepted shapes are hours and/or minutes, with the hour part coming before the
minute part and each unit appearing at most once, e.g. "2h", "45m", "1h30m"
or "1h 30m". Units are case-insensitive and surrounding whitespace is ignored.
Any other text (including other units, a missing number, minutes before hours
or negative numbers) raises ``ValueError``.
"""

import re

_UNIT = r"(?P<num>\d+)(?P<unit>[hm])\b"
_PATTERN = re.compile(rf"^\s*(?:{_UNIT}\s*)?$", re.IGNORECASE)


def parse_duration(text: str) -> int:
    """Return the number of whole minutes in a compact duration string.

    Examples::

        parse_duration("2h")     -> 120
        parse_duration("45m")    -> 45
        parse_duration("1h30m")  -> 90
        parse_duration("1h 30m") -> 90

    Raises ``ValueError`` for anything that is not a valid "hours and minutes"
    duration.
    """
    if not isinstance(text, str):
        raise ValueError(f"duration must be a string, not {type(text).__name__}")

    matches = _PATTERN.findall(text)
    if not matches:
        raise ValueError(f"invalid duration: {text!r}")

    total = 0
    seen = set()
    for number, unit in matches:
        unit = unit.lower()
        if unit in seen:
            raise ValueError(f"unit '{unit}' repeated in duration: {text!r}")
        seen.add(unit)
        if unit == "h":
            total += int(number) * 60
        else:  # "m"
            total += int(number)

    return total


if __name__ == "__main__":
    for sample in ("2h", "45m", "1h30m", "1h 30m", "  1H 30M "):
        print(f"{sample!r} -> {parse_duration(sample)}")
