```python
"""Weight parsing helpers for warehouse-entered package weights."""

import re

_UNIT_GRAMS = {
    "g": (1, 0),
    "kg": (1000, 0),
    "lb": (45359237, 5),
    "oz": (28349523125, 9),
}

_WEIGHT_RE = re.compile(
    r"(?P<number>\d+(?:\.\d*)?|\.\d+)\s*(?P<unit>g|kg|lb|oz)",
    re.IGNORECASE,
)


def parse_weight(text: str) -> int:
    """Parse a weight string into whole grams.

    Accepts an integer or decimal number (using "." as the decimal separator)
    followed by optional whitespace and one of the units g, kg, lb, or oz.
    The unit is case-insensitive. The result is rounded to the nearest gram.

    Raises ValueError for malformed input, unknown units, negative values,
    or comma decimal separators.
    """
    if not isinstance(text, str):
        raise ValueError("weight must be a string")

    match = _WEIGHT_RE.fullmatch(text.strip())
    if match is None:
        raise ValueError("invalid weight")

    number = match.group("number")
    unit = match.group("unit").lower()

    if "." in number:
        integer_part, fraction_part = number.split(".")
        digits = integer_part + fraction_part
        scale = len(fraction_part)
    else:
        digits = number
        scale = 0

    if not digits:
        raise ValueError("invalid weight")

    grams_numerator = int(digits)
    factor_numerator, factor_scale = _UNIT_GRAMS[unit]

    denominator = 10 ** (scale + factor_scale)
    return (grams_numerator * factor_numerator + denominator // 2) // denominator
```
