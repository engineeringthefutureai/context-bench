import re

# Grams per unit as exact (numerator, denominator) pairs so conversion stays
# in integer arithmetic: 1 lb = 453.59237 g, 1 oz = 28.349523125 g is 28349523125/10^9.
_GRAMS_PER_UNIT = {
    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (28349523125, 1000000000),
}

_WEIGHT_RE = re.compile(r"(\d+(?:\.\d+)?)\s*(g|kg|lb|oz)$", re.IGNORECASE)


def parse_weight(text: str) -> int:
    """Weight typed by warehouse staff as whole grams, e.g. "1.2 kg" -> 1200.

    Expects a non-negative number (integer or "."-decimal) followed by an
    optional whitespace and the unit g, kg, lb or oz; anything else raises ValueError.
    """
    match = _WEIGHT_RE.match(text.strip())
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")

    amount, unit = match.groups()
    whole, _, frac = amount.partition(".")
    numerator, scale = int(whole + frac), len(frac)

    num, den = _GRAMS_PER_UNIT[unit.lower()]
    denom = 10**scale * den
    grams, remainder = divmod(numerator * num, denom)
    if remainder * 2 >= denom:
        grams += 1
    return grams
