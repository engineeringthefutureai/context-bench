```python
import re

# grams as (numerator, denominator) so conversions stay exact
_GRAMS_PER_UNIT = {
    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (28349523125, 1000000000),
}

_WEIGHT_RE = re.compile(r"(\d+(?:\.\d+)?)\s*(g|kg|lb|oz)$", re.IGNORECASE)


def parse_weight(text: str) -> int:
    """Weight typed by warehouse staff as whole grams, e.g. "1.2 kg" -> 1200.

    The number must be non-negative and use "." as decimal separator; the unit
    is g, kg, lb or oz. Anything else raises ValueError.
    """
    match = _WEIGHT_RE.match(text.strip())
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")

    amount, unit = match.groups()
    if "." in amount:
        whole, _, frac = amount.split(".")
        numerator, scale = int(whole + frac), len(frac)
    else:
        numerator, scale = int(amount), 0

    num, den = _GRAMS_PER_UNIT[unit.lower()]
    denom = 10**scale * den
    grams, remainder = divmod(numerator * num, denom)
    if remainder * 2 >= denom:
        grams += 1
    return grams
```
