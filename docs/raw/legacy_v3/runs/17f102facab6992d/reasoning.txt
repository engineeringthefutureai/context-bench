class InsufficientStock(Conflict):

class Stock:
    """On-hand quantities per SKU."""

        self._on_hand[sku] = self._on_hand.get(sku, 0) + qty

from shopkit.errors import Conflict, ShopError


class InsufficientStock(Conflict):
    """Raised when a reservation exceeds the available stock."""


class Stock:
    """On-hand quantities per SKU."""

    def __init__(self):
        self._on_hand: dict[str, int] = {}

    def receive(self, sku: str, qty: int) -> None:
        self._on_hand[sku] = self._on_hand.get(sku, 0) + qty

    def available(self, sku: str) -> int:
        return self._on_hand.get(sku, 0)

    def reserve(self, sku: str, qty: int) -> None:
        """Hold `qty` units of `sku` for an order.

        Raises:
            ShopError: if qty is negative.
            InsufficientStock: if qty exceeds the available quantity.
        """
        if qty < 0:
            raise ShopError("qty must be non-negative")

        available_qty = self.available(sku)
        if qty > available_qty:
            raise InsufficientStock(
                "Insufficient stock",
                requested=qty,
                available=available_qty,
            )

        if qty:
            self._on_hand[sku] = available_qty - qty
