from shopkit.inventory.lots import NotEnoughStock


class Stock:
    """On-hand quantities per SKU."""

    def __init__(self):
        self._on_hand: dict[str, int] = {}

    def receive(self, sku: str, qty: int) -> None:
        self._on_hand[sku] = self._on_hand.get(sku, 0) + qty

    def available(self, sku: str) -> int:
        return self._on_hand.get(sku, 0)

    def reserve(self, sku: str, qty: int) -> None:
        """Hold `qty` units of `sku` for an order, lowering the available quantity.

        Raises NotEnoughStock (409) with the `requested` and `available`
        quantities if on hand is short; nothing changes in that case.
        """
        if qty < 0:
            raise ValueError(f"reserve qty must not be negative, got {qty}")
        have = self.available(sku)
        if qty > have:
            raise NotEnoughStock(
                f"cannot reserve {qty} of {sku}: only {have} available",
                requested=qty,
                available=have,
            )
        if qty:
            self._on_hand[sku] = have - qty
