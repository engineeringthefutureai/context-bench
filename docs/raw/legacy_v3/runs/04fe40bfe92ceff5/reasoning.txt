# Grams per unit, using the exact legal conversions where defined.

    "oz": 28.349523125,

    """Parse a package weight typed by warehouse staff, e.g. "2.5 kg" -> 2500.

    Accepts an integer or a "."-separated decimal number, optional whitespace,

```python
"""Unit conversions for warehouse input.

Weights arrive as free text typed by staff (e.g. "2.5 kg"); `parse_weight`
normalises them to whole grams.
"""

import re

_GRAMS_PER = {
    "g": 1.0,
    "kg": 1000.0,
    "lb": 453.59237,
    "oz": 28.349523125,
}

# A number (no sign, "." decimal only), optional whitespace, then a known unit.
_WEIGHT_RE = re.compile(
    r"^\s*(\d+(?:\.\d+)?)\s*(g|kg|lb|oz)\s*$", re.IGNORECASE
)


def parse_weight(text: str) -> int:
    """Parse a weight like "2.5 kg" and return whole grams, rounded.

    The number may be an integer or a "."-separated decimal; the unit is
    g, kg, lb or oz (case-insensitive) and surrounding whitespace is
    ignored. Raises ValueError for anything else, including negative
    values and "," decimal separators.
    """
    match = _WEIGHT_RE.match(text)
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")
    number, unit = match.group(1), match.group(2)
    return round(float(number) * _GRAMS_PER[unit.lower()])
```
