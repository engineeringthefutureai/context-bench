"""Unit conversions for weights typed in by warehouse staff."""

import re

_GRAMS_PER = {
    "g": 1,
    "kg": 1000,
    "lb": 453.59237,
    "oz": 28.349523125,
}

# A non-negative integer or "."-separated decimal, optional whitespace,
# then a known unit. Anything else (signs, ",", letters in the number,
# missing pieces) does not match and is rejected by parse_weight.
_WEIGHT_RE = re.compile(
    r"^\s*(\d+(?:\.\d+)?)\s*(g|kg|lb|oz)\s*$", re.IGNORECASE
)


def parse_weight(text: str) -> int:
    """Parse a weight like "2.5 kg" and return the weight in whole grams.

    The number may be an integer or a "."-separated decimal, the unit is
    g, kg, lb or oz (case-insensitive), and leading/trailing whitespace is
    ignored. The result is rounded to the nearest gram. Raises ValueError
    for anything else, including negative values and "," decimal
    separators.
    """
    match = _WEIGHT_RE.match(text)
    if match is None:
        raise ValueError(f"invalid weight: {text!r}")
    number, unit = match.group(1), match.group(2)
    return round(float(number) * _GRAMS_PER[unit.lower()])
