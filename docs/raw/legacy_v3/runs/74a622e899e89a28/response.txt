"""Duration parsing: human-facing "2h30m" strings to whole minutes."""
import re

# Hours come before minutes; each unit appears at most once and needs a number.
# The unit letter is case-insensitive; whitespace between the parts is tolerated.
_DURATION_RE = re.compile(r"(\d+[hH]\s*)?(\d+[mM])?")


def parse_duration(text: str) -> int:
    """Whole minutes from a "2h", "45m", "1h30m" or "1h 30m" string; empty text,
    other units, missing numbers, minutes before hours and negative numbers raise ValueError."""
    match = _DURATION_RE.fullmatch(text.strip())
    if match is None:
        raise ValueError(f"invalid duration: {text!r}")
    hours = int(match.group(1)[:-1]) if match.group(1) else 0
    minutes = int(match.group(2)[:-1]) if match.group(2) else 0
    return hours * 60 + minutes
