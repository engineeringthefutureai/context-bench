"""Duration parsing. Human-facing durations like "2h", "1h30m" become whole minutes."""
import re

# Hours before minutes, each unit at most once, units case-insensitive, optional
# whitespace between the two parts, digits only (so negatives can't appear).
_DURATION_RE = re.compile(r"(?:(\d+)h)?(?:\s*(\d+)m)?", re.IGNORECASE)


def parse_duration(text: str) -> int:
    """Whole minutes for a duration of hours and/or minutes, e.g. "2h" -> 120,
    "1h 30m" -> 90. Units h/m are case-insensitive and hours must come before
    minutes. Anything else raises ValueError."""
    match = _DURATION_RE.fullmatch(text.strip())
    if match is None or (match.group(1) is None and match.group(2) is None):
        raise ValueError(f"invalid duration: {text!r}")
    hours, minutes = match.group(1), match.group(2)
    return (int(hours) if hours else 0) * 60 + (int(minutes) if minutes else 0)
