```python
"""Parsing and conversion of free-typed package weights.

Warehouse staff type weights in as text (e.g. ``"1.2 kg"``). Everything is
normalised to whole grams, which is the unit the rest of the shipping code
works in.
"""

from __future__ import annotations

import re
from fractions import Fraction

# Exact grams per unit.  lb/oz factors are the internationally agreed
# definitions (1 lb = 453.59237 g, 1 oz = 28.349523125 g).
GRAMS_PER_UNIT: dict[str, Fraction] = {
    "g": Fraction(1),
    "kg": Fraction(1000),
    "lb": Fraction("453.59237"),
    "oz": Fraction("28.349523125"),
}

_WEIGHT_RE = re.compile(
    r"^\s*([0-9]+(?:\.[0-9]+)?)\s*([a-z]+)\s*$",
    re.IGNORECASE,
)


def parse_weight(text: str) -> int:
    """Parse a typed-in weight such as ``"500 g"`` or ``"1.2kg"`` into whole grams.

    The value may be an integer or a decimal using ``"."`` as separator,
    followed by an optional space and a case-insensitive unit of ``g``,
    ``kg``, ``lb`` or ``oz``. Surrounding whitespace is ignored. The result
    is rounded to the nearest gram.

    Raises:
        ValueError: If the text has no number or unit, an unknown unit, a
            negative value or uses a different (e.g. ``","``) decimal
            separator.
    """
    match = _WEIGHT_RE.match(text)
    if match is None:
        raise ValueError(f"Invalid weight: {text!r}")
    quantity, unit = match.groups()
    factor = GRAMS_PER_UNIT.get(unit.lower())
    if factor is None:
        raise ValueError(f"Unknown weight unit: {unit!r}")
    grams = Fraction(quantity) * factor
    return (2 * grams + 1) // 2
```

```python
"""Parsing of typed-in package weights.

Warehouse staff enter package weights as free text (e.g. ``"2 lb"``); the
shipping code works in whole grams, so every value is converted and rounded
to the nearest gram here.
"""

import re
from fractions import Fraction

# Exact grams per unit. The lb and oz factors are the international
# definitions; keeping them as Fractions avoids float rounding error.
_GRAMS_PER_UNIT: dict[str, Fraction] = {
    "g": Fraction(1),
    "kg": Fraction(1000),
    "lb": Fraction("453.59237"),
    "oz": Fraction("28.349523125"),
}

# A number (integer or decimal using "."), optional whitespace, then a unit.
# Digits are matched explicitly so Unicode lookalikes are rejected.
_WEIGHT_RE = re.compile(r"^\s*([0-9]+(?:\.[0-9]+)?)\s*([a-z]+)\s*$", re.IGNORECASE)


def parse_weight(text: str) -> int:
    """Parse a typed-in weight like ``"500 g"``, ``"1.5kg"`` or ``"2 LB"``.

    The value must be an integer or a decimal using ``.`` as the separator,
    followed by an optional space and one of the case-insensitive units
    ``g``, ``kg``, ``lb`` or ``oz``. Leading and trailing whitespace is
    ignored. The result is the weight in whole grams, rounded half up.

    Raises:
        ValueError: for missing numbers or units, unknown units, negative
            values and any other malformed input (including ``","`` as the
            decimal separator).
    """
    match = _WEIGHT_RE.match(text)
    if match is None:
        raise ValueError(
            f"invalid weight {text!r}: expected a number followed by g, kg, lb or oz"
        )
    quantity, unit = match.groups()
    factor = _GRAMS_PER_UNIT.get(unit.lower())
    if factor is None:
        raise ValueError(
            f"invalid weight {text!r}: unknown unit {unit!r} "
            f"(expected g, kg, lb or oz)"
        )
    grams = Fraction(quantity) * factor
    # Round half up to the nearest whole gram.
    return (2 * grams + 1) // 2
```
