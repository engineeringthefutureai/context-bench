from shopkit.errors import Conflict


class InsufficientStock(Conflict):
    """Raised when a 409 when a reservation cannot be fulfilled."""
    pass


class Stock:
    """On-hand quantities per SKU."""

    def __init__(self):
        self._on_hand: dict[str, int] = {}

    def receive(self, sku: str, qty: int) -> None:
        self._on_hand[sku] = self._on_hand.get(sku, 0) + qty

    def available(self, sku: str) -> int:
        return self._on_hand.get(sku, 0)

    def reserve(self, sku: str, qty: int) -> None:
        """Lower the available quantity of ``sku`` by ``qty``.

        If there is not enough stock, nothing changes and :class:`InsufficientStock`
        (a 409 Conflict) is raised, carrying ``requested`` and ``available`` in
        the error details.
        """
        if qty < 1:
            raise ValueError("qty must be >= 1")
        avail = self._on_hand.get(sku, 0)
        if avail < qty:
            raise InsufficientStock(f"{sku}: requested {qty} but only {avail} available",
                                    sku=sku, requested=qty, available=avail)
        self._on_hand[sku] = avail - qty
