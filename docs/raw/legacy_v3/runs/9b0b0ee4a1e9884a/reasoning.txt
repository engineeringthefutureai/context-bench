```
from __future__ import annotations
import re

_WEIGHT_RE = re.compile(r"\s*(\d+(?:\.\d*)?|\.\d+)\s*(kg|g|lb|oz)\s*", re.IGNORECASE)
_UNIT_G_PER = {
    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (45359237, 1600000),
}

def parse_weight(text: str) -> int:
    """..."""
    if not isinstance(text, str):
        raise ValueError("Invalid weight")
    m = _WEIGHT_RE.fullmatch(text)
    if m is None:
        raise ValueError("Invalid weight")
    number = m.group(1)
    unit = m.group(2).lower()
    int_part, dot, frac_part = number.partition(".")
    frac = frac_part.rstrip("0") if dot else ""
    denom = 10 ** len(frac)
    num = int(int_part or "0") * denom + (int(frac) if frac else 0)
    unit_num, unit_den = _UNIT_G_PER[unit]
    numerator = num * unit_num
    denominator = denom * unit_den
    return (2 * numerator + denominator) // (2 * denominator)
```

```
"""Helpers for human-typed units of measure."""
import re

# Exact grams per unit as (numerator, denominator). lb/oz use the international
# avoirdupois pound/ounce definitions so parsing avoids float drift.
_GRAMS_PER_UNIT = {
    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (45359237, 1600000),
}

_WEIGHT_RE = re.compile(r"\s*(\d+(?:\.\d+)?|\.\d+)\s*(kg|g|lb|oz)\s*", re.IGNORECASE)

def parse_weight(text: str) -> int:
    """Parse a staff-entered weight string into whole grams.

    Accepts a non-negative integer or decimal ("." only), optional whitespace,
    and a unit of g, kg, lb or oz (case-insensitive). Raises ValueError for any
    other input, including negatives and commas.
    """
    if not isinstance(text, str):
        raise ValueError(f"Invalid weight: {text!r}")
    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"Invalid weight: {text!r}")

    number, unit = match.group(1), match.group(2).lower()

    whole, dot, fractional = number.partition(".")
    fractional = fractional.rstrip("0") if dot else ""
    denominator = 10 ** len(fractional)
    numerator = int(whole or "0") * denominator + (int(fractional) if fractional else 0)

    unit_num, unit_den = _GRAMS_PER_UNIT[unit]
    grams = numerator * unit_num
    scale = denominator * unit_den

    # Round half up (all accepted values are non-negative).
    return (2 * grams + scale) // (2 * scale)
```

```
import re

# Exact grams per unit as (numerator, denominator), using the definitions
# 1 lb = 453.59237 g and 1 oz = 28.349523125 g.
_GRAMS_PER_UNIT = {
    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (45359237, 1600000),
}

_WEIGHT_RE = re.compile(r"\s*(\d+(?:\.\d+)?|\.\d+)\s*(kg|g|lb|oz)\s*", re.IGNORECASE)

def parse_weight(text: str) -> int:
    """Parse a typed weight such as "1.5 kg" or ".5 lb" into whole grams.

    The unit is g, kg, lb or oz, case-insensitive, with optional whitespace
    around it. Raises ValueError for missing or unknown units, negative values,
    comma decimal separators, and any other malformed input.
    """
    if not isinstance(text, str):
        raise ValueError("Invalid weight")

    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError("Invalid weight")

    number, unit = match.group(1), match.group(2).lower()

    whole, dot, fractional = number.partition(".")
    fractional = fractional.rstrip("0") if dot else ""
    denominator = 10 ** len(fractional)
    numerator = int(whole or "0") * denominator + (int(fractional) if fractional else 0)

    unit_num, unit_den = _GRAMS_PER_UNIT[unit]
    grams = numerator * unit_num
    scale = denominator * unit_den

    # Round half up; all accepted values are non-negative.
    return (2 * grams + scale) // (2 * scale)
```
