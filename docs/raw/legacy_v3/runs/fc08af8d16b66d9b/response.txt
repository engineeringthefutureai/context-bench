"""Parsing helpers for shipping-related user input.

Warehouse staff type package weights in a variety of formats; the
functions here normalise those strings into exact integer values the
rest of the codebase can rely on.
"""

import re

# Exact conversion factors to grams, as specified by shipping.  1 lb
# = 453.59237 g and 1 oz = 28.349523125 g; both are exact by
# definition so no rounding is applied at conversion time.
_LB_TO_G = 453.59237
_OZ_TO_G = 28.349523125
_KG_TO_G = 1000.0
_G_TO_G = 1.0

# Accepted units, case-insensitive.  Only the bare unit names are
# recognised; plurals and abbreviations are deliberately rejected.
_UNITS = {"g", "kg", "lb", "oz"}

# Number part: an optional sign is not allowed by the spec (negative
# weights are invalid), digits with at most one "." as the decimal
# separator.  The "." alternative is tried before the plain-digit
# alternative so that ".5" is accepted.
_NUMBER_RE = re.compile(r"(\d+(?:\.\d+)?|\.\d+)")

# Whole pattern: optional leading whitespace, the number, any interior
# whitespace, a single unit, then optional trailing whitespace.  The
# unit is captured so it can be looked up explicitly; anything that
# does not match the pattern as a whole is rejected by the caller.
_PATTERN_RE = re.compile(
    r"\s*(\d+(?:\.\d+)?|\.\d+)\s*([A-Za-z]+)\s*$"
)


def parse_weight(text: str) -> int:
    """Parse a human-typed package weight into whole grams.

    The input is a number (integer or decimal with ``.`` separator),
    optional whitespace, then one of the units ``g``, ``kg``, ``lb``
    or ``oz`` (case-insensitive).  Leading and trailing whitespace is
    ignored.

    Args:
        text: The raw string as typed by the user.

    Returns:
        The weight in whole grams, rounded to the nearest gram.

    Raises:
        ValueError: If ``text`` is not a string, contains no number,
            has no or an unknown unit, is negative, uses ``','`` as a
            decimal separator, or is otherwise malformed.
    """
    if not isinstance(text, str):
        raise ValueError(f"weight must be a string, got {type(text).__name__}")

    # Reject negative values early: a leading "-" is not part of the
    # accepted number grammar, so the pattern match below would fail
    # anyway, but checking first gives a clearer message.  We strip
    # only leading whitespace so an interior "-" (e.g. "1-2g") is
    # still rejected by the pattern.
    if text.lstrip().startswith("-"):
        raise ValueError(f"negative weight is not allowed: {text!r}")

    match = _PATTERN_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"unrecognised weight format: {text!r}")

    number_text, unit_text = match.group(1), match.group(2)
    unit = unit_text.lower()
    if unit not in _UNITS:
        raise ValueError(f"unknown unit: {unit!r}")

    # "," is explicitly not a decimal separator; the pattern above
    # cannot match it, so this is already rejected, but a dedicated
    # check gives a clearer error for the common typo "1,5 kg".
    if "," in number_text:
        raise ValueError(f"comma is not a valid decimal separator: {text!r}")

    try:
        value = float(number_text)
    except ValueError:  # pragma: no cover - pattern already guards this
        raise ValueError(f"invalid number: {number_text!r}") from None

    factor = {
        "g": _G_TO_G,
        "kg": _KG_TO_G,
        "lb": _LB_TO_G,
        "oz": _OZ_TO_G,
    }[unit]

    grams = value * factor
    return int(round(grams))
