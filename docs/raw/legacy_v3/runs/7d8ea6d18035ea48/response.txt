"""Unit conversion and weight-entry parsing for shipping (weights are whole grams)."""
import re
from decimal import ROUND_HALF_UP, Decimal

# Exact conversion factors: 1 lb = 453.59237 g, 1 oz = 28.349523125 g (both exact by definition).
_G_PER_UNIT: dict[str, Decimal] = {
    "g": Decimal(1),
    "kg": Decimal(1000),
    "lb": Decimal("453.59237"),
    "oz": Decimal("28.349523125"),
}

_WEIGHT_RE = re.compile(r"^([+-]?(\d+(?:\.\d*)?|\.\d+))\s*([a-zA-Z]+)$")


def parse_weight(text: str) -> int:
    """Weight entered by warehouse staff as a number plus a unit -> whole grams.

    The number is an integer or a decimal with "." as separator, optionally followed by
    whitespace and one of the units g, kg, lb or oz (case-insensitive); leading and
    trailing whitespace is ignored. The result is rounded to the nearest gram.
    parse_weight("2.5 kg") -> 2500; parse_weight(" 1 LB ") -> 454.
    Raises ValueError for anything else, including a missing number or unit, a
    non-numeric or negative value and "," as the decimal separator."""
    match = _WEIGHT_RE.match(text.strip())
    if not match:
        raise ValueError(f"invalid weight: {text!r}")
    value = Decimal(match.group(1))
    if value < 0:
        raise ValueError(f"weight must be non-negative: {text!r}")
    grams = value * _G_PER_UNIT[match.group(3).lower()]
    return int(grams.quantize(Decimal(1), rounding=ROUND_HALF_UP))
