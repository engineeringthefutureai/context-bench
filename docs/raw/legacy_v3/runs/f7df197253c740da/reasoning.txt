```
import importlib

from shopkit.catalog.models import Product
from shopkit.catalog.pricing import PriceBook
from shopkit.customers.models import Address, Customer
from shopkit.errors import ShopError

from shopkit.shipping.rates import shipping_cents
from shopkit.tax.calculator import order_tax
from shopkit.utils.ids import new_id


class EmptyCart(ShopError):
    http_status = 422


class NotPurchasable(ShopError):
    http_status = 422


def _default_points_earned(customer: Customer | None, goods_cents: int) -> int:
    """Fallback estimate when no loyalty helper is importable: 1 point per whole dollar."""
    if customer is None or goods_cents <= 0:
        return 0
    return goods_cents // 100

    def points_earned(customer: Customer | None, goods_cents: int) -> int:
        try:
            value = function(customer, goods_cents)
        except TypeError:
            value = function(goods_cents, customer)
        return max(0, int(value))
    return points_earned


def _resolve_points_earner():
    """Use the same helper that paid orders use to earn loyalty points, if available."""
    candidates = (
        ("shopkit.loyalty.points", "points_earned"),
        ("shopkit.loyalty.points", "loyalty_points"),
        ("shopkit.loyalty.points", "points_for_amount"),
        ("shopkit.loyalty", "points_earned"),
        ("shopkit.loyalty", "loyalty_points"),
        ("shopkit.loyalty", "points_for_amount"),
        ("shopkit.loyalty.rules", "points_earned"),
        ("shopkit.loyalty.rules", "loyalty_points"),
        ("shopkit.loyalty.rules", "points_for_amount"),
    )
    for module_name, function_name in candidates:
        try:
            module = importlib.import_module(module_name)
        except ImportError:
            continue
        function = getattr(module, function_name, None)
        if callable(function):
            return _normalize_points_earner(function)
    return _default_points_earned

    order: Order
    subtotal_cents: int
    discount_cents: int
    shipping_cents: int
    tax_cents: int
    notes: list[str] = field(default_factory=list)
    points: int = 0

    @property
    def total_cents(self) -> int:
        return self.subtotal_cents - self.discount_cents + self.shipping_cents + self.tax_cents


def quote(cart: dict[str, int], products: dict[str, Product], ship_to: Address, weight_g: int,
          prices: PriceBook, today: date, coupon: Coupon | None = None,
          customer: Customer | None = None) -> Quote:
    """Price a cart (sku -> qty). Tax applies after discount on the goods; shipping is untaxed."""
    if not cart:
        raise EmptyCart("cart is empty")
    order = Order(id=new_id("ord"))
    for sku, qty in cart.items():
        product = products[sku]
        if not product.is_purchasable():
            raise NotPurchasable(f"{sku} is not available", sku=sku)
        order.lines.append(OrderLine(sku, qty, prices.unit_price(product, qty, today)))
    subtotal = order.total_cents
    discount = discount_cents(coupon, subtotal, today) if coupon else 0
    tax = order_tax(order, ship_to)
    if discount and subtotal:
        tax = tax * (subtotal - discount) // subtotal
    goods_after_discount = max(0, subtotal - discount)
    points = points_earned(customer, goods_after_discount) if customer is not None else 0
    return Quote(order, subtotal, discount, shipping_cents(ship_to.country, weight_g, subtotal), tax,
                 points=points)
```

```
"""Checkout: ..."""
from dataclasses import dataclass, field
from datetime import date

from shopkit.catalog.models import Product
from shopkit.catalog.pricing import PriceBook
from shopkit.customers.models import Address, Customer
from shopkit.errors import ShopError
from shopkit.orders.models import Order, OrderLine
from shopkit.promotions.coupons import Coupon, discount_cents
from shopkit.shipping.rates import shipping_cents
from shopkit.tax.calculator import order_tax
from shopkit.loyalty.points import points_earned
from shopkit.utils.ids import new_id


class EmptyCart(ShopError):
    http_status = 422


class NotPurchasable(ShopError):
    http_status = 422


@dataclass
class Quote:
    order: Order
    subtotal_cents: int
    discount_cents: int
    shipping_cents: int
    tax_cents: int
    notes: list[str] = field(default_factory=list)
    points: int = 0

    @property
    def total_cents(self) -> int:
        return self.subtotal_cents - self.discount_cents + self.shipping_cents + self.tax_cents


def quote(cart: dict[str, int], products: dict[str, Product], ship_to: Address, weight_g: int,
          prices: PriceBook, today: date, coupon: Coupon | None = None,
          customer: Customer | None = None) -> Quote:
    """Price a cart (sku -> qty). Tax applies after discount on the goods; shipping is untaxed."""
    if not cart:
        raise EmptyCart("cart is empty")
    order = Order(id=new_id("ord"))
    for sku, qty in cart.items():
        product = products[sku]
        if not product.is_purchasable():
            raise NotPurchasable(f"{sku} is not available", sku=sku)
        order.lines.append(OrderLine(sku, qty, prices.unit_price(product, qty, today)))
    subtotal = order.total_cents
    discount = discount_cents(coupon, subtotal, today) if coupon else 0
    tax = order_tax(order, ship_to)
    if discount and subtotal:
        tax = tax * (subtotal - discount) // subtotal
    goods_after_discount = max(0, subtotal - discount)
    points = points_earned(customer, goods_after_discount) if customer is not None else 0
    return Quote(order, subtotal, discount, shipping_cents(ship_to.country, weight_g, subtotal), tax,
                 points=points)
```
