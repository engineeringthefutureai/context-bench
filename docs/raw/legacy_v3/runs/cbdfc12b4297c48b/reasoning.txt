```python
import re
from decimal import Decimal, ROUND_HALF_UP

# Exact gram equivalents for the units warehouse staff can enter.
_GRAMS_PER_UNIT = {
    "g": Decimal(1),
    "kg": Decimal(1000),
    "lb": Decimal("453.59237"),
    "oz": Decimal("28.349523125"),
}

_WEIGHT_RE = re.compile(
    r"^(\d+\.?\d*|\.\d+)\s*(g|kg|lb|oz)$", re.IGNORECASE
)


def parse_weight(text: str) -> int:
    """Weight in whole grams from a warehouse entry like "2.5 kg" or "12 OZ"."""
    match = _WEIGHT_RE.match(text.strip())
    if match is None:
        raise ValueError(f"unparseable weight: {text!r}")
    number, unit = match.group(1), match.group(2).lower()
    return int((Decimal(number) * _GRAMS_PER_UNIT[unit]).quantize(Decimal(1), rounding=ROUND_HALF_UP))
```
