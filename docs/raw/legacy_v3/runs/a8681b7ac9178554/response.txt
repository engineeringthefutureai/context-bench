import csv
import json
from io import StringIO

from shopkit.orders.models import Order


def _clean_product_code(code: str) -> str:
    """Normalize a product code the same way the shop presents it elsewhere."""
    if not code:
        return ""
    return " ".join(code.split()).upper()


def _format_invoice_amount(cents: int) -> str:
    """Format integer cents the way customer invoices render amounts."""
    cents = int(cents)
    sign = "-" if cents < 0 else ""
    whole, fraction = divmod(abs(cents), 100)
    return f"{sign}{whole}.{fraction:02d}"


def export_orders_json(orders: list[Order]) -> str:
    """Raw JSON export for the warehouse integration."""
    payload = [
        {
            "id": o.id,
            "currency": o.currency,
            "lines": [{"sku": l.sku, "qty": l.qty, "unit_cents": l.unit_cents} for l in o.lines],
        }
        for o in orders
    ]
    return json.dumps(payload, indent=2)


def export_orders_csv(orders: list[Order]) -> str:
    """CSV export of order lines for finance."""
    buffer = StringIO()
    writer = csv.writer(buffer, lineterminator="\n")
    writer.writerow(["order_id", "product_code", "quantity", "line_total"])

    for order in orders:
        for line in order.lines:
            writer.writerow(
                [
                    order.id,
                    _clean_product_code(line.sku),
                    line.qty,
                    _format_invoice_amount(line.qty * line.unit_cents),
                ]
            )

    return buffer.getvalue()
