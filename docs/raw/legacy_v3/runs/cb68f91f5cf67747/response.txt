import csv
import io
import json

from shopkit.orders.models import Order


def export_orders_json(orders: list[Order]) -> str:
    """Raw JSON export for the warehouse integration."""
    payload = [
        {
            "id": o.id,
            "currency": o.currency,
            "lines": [{"sku": l.sku, "qty": l.qty, "unit_cents": l.unit_cents} for l in o.lines],
        }
        for o in orders
    ]
    return json.dumps(payload, indent=2)


def export_orders_csv(orders: list[Order]) -> str:
    """CSV export for finance, with one row per order line."""
    buffer = io.StringIO()
    writer = csv.writer(buffer, lineterminator="\n")
    writer.writerow(["order_id", "product_code", "quantity", "line_total"])
    for order in orders:
        for line in order.lines:
            writer.writerow(
                [
                    order.id,
                    _clean_product_code(line.sku),
                    line.qty,
                    _format_line_total(line.qty * line.unit_cents,
                ]
            )
    return buffer.getvalue()


def _clean_product_code(sku: str) -> str:
    """Normalize a product code the same way as other shop exports."""
    return sku.strip().upper()


def _format_line_total(cents: int) -> str:
    """Render a line total in the two-decimal style used on customer invoices."""
    sign = "-" if cents < 0 else ""
    dollars, remainder = divmod(abs(cents), 100)
    return f"{sign}{dollars}.{remainder:02d}"
