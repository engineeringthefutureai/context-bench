```
"""Weight parsing utilities."""

import re
from fractions import Fraction

__all__ = ["parse_weight"]

_WEIGHT_RE = re.compile(
    r"\s*(?P<number>[0-9-9]+(?:\.[0-9]*)?|\.[0-9]+)\s*(?P<unit>g|kg|lb|oz)\s*",
    re.IGNORECASE,
)

_GRAM_FACTORS = {
    "g": Fraction(1, 1),
    "kg": Fraction(1000, 1),
    "lb": Fraction(45359237, 100000),
    "oz": Fraction(28349523125, 1000000000),
}

def _to_fraction(number: str) -> Fraction:
    if "." in number:
        whole, frac = number.split(".", 1)
        whole = whole or "0"
        return Fraction(int(whole + frac), 10 ** len(frac))
    return Fraction(int(number), 1)

def parse_weight(text: str) -> int:
    """Parse ..."""
    if not isinstance(text, str):
        raise ValueError(f"Invalid weight: {text!r}")
    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"Invalid weight: {text!r}")
    number = _to_fraction(match.group("number"))
    factor = _GRAM_FACTORS[match.group("unit").lower()]
    grams = number * factor
    numerator, denominator = grams.numerator, grams.denominator
    return (2 * numerator + denominator) // (2 * denominator)
```

```
"""Weight parsing utilities."""

import re

_WEIGHT_RE = re.compile(
    r"\s*(?P<number>[0-9]+(?:\.[0-9]*)?|\.[0-9]+)\s*(?P<unit>g|kg|lb|oz)\s*",
    re.IGNORECASE,
)

_GRAM_FACTORS = {
    "g": (1, 1),
    "kg": (1000, 1),
    "lb": (45359237, 100000),
    "oz": (28349523125, 1000000000),
}

def _int_from_digits(digits: str) -> int:
    value = 0
    for i in range(0, len(digits), 9):
        part = digits[i:i + 9]
        value = value * (10 ** len(part)) + int(part)
    return value

def _number_parts(number: str) -> tuple:
    if "." in number:
        whole, frac = number.split(".", 1)
        whole = whole or "0"
        digits = whole + frac
        denominator = 10 ** len(frac)
    else:
        digits = number
        denominator = 1
    return _int_from_digits(digits), denominator

def parse_weight(text: str) -> int:
    """Parse a package weight string and return the weight in whole grams.

    Accepts an integer or decimal number followed by g, kg, lb, or oz
    (case-insensitive), with optional whitespace. Raises ValueError for
    anything else, including negative values and comma decimal separators.
    """
    if not isinstance(text, str):
        raise ValueError(f"Invalid weight: {text!r}")
    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"Invalid weight: {text!r}")

    number_numerator, number_denominator = _number_parts(match.group("number"))
    factor_numerator, factor_denominator = _GRAM_FACTORS[match.group("unit").lower()]
    numerator = number_numerator * factor_numerator
    denominator = number_denominator * factor_denominator
    return (2 * numerator + denominator) // (2 * denominator)
```

```
"""Weight parsing utilities."""

import re
from decimal import Decimal, ROUND_HALF_UP, localcontext

_WEIGHT_RE = re.compile(
    r"\s*(?P<number>[0-9]+(?:\.[0-9]*)?|\.[0-9]+)\s*(?P<unit>g|kg|lb|oz)\s*",
    re.IGNORECASE,
)

_GRAM_FACTORS = {
    "g": Decimal("1"),
    "kg": Decimal("1000"),
    "lb": Decimal("453.59237"),
    "oz": Decimal("28.349523125"),
}

_ONE = Decimal("1")

def parse_weight(text: str) -> int:
    """Parse a package weight string into whole grams.

    The input must be a non-negative integer or decimal number followed by
    one of the units g, kg, lb, or oz. Whitespace may surround the input and
    may separate the number and unit. Unit matching is case-insensitive.

    Raises ValueError for missing/unknown units, signs, commas, or any other
    malformed input.
    """
    if not isinstance(text, str):
        raise ValueError(f"Invalid weight: {text!r}")
    match = _WEIGHT_RE.fullmatch(text)
    if match is None:
        raise ValueError(f"Invalid weight: {text!r}")

    number = match.group("number")
    unit = match.group("unit").lower()
    with localcontext() as ctx:
        ctx.prec = max(28, len(number) + 20)
        grams = Decimal(number) * _GRAM_FACTORS[unit]
        return int(grams.quantize(_ONE, rounding=ROUND_HALF_UP))
```
